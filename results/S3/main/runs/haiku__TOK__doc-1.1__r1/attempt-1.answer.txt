```tok
tok 1.0 "Customized bicycle order"

pool company "Bicycle manufacturer"
  lane sales "Sales department"
    start received "Order received"
    xor accepted "Order accepted?"
    task reject "Reject the order"
    task accept "Accept the order"
    end rejected "Order rejected"
    task ship "Ship the bicycle to the customer"
    end shipped "Process instance finished"
  lane storehouse "Storehouse"
    and informed "Storehouse and engineering informed"
    task processParts "Process the part list of the order"
    task checkQty "Check the required quantity of each part"
    xor available "Part available in-house?"
    task reserve "Reserve the part"
    task backorder "Back-order the part"
    xor itemsDone "Every item on the part list processed?"
    and ready "Storehouse and preparation complete"
  lane engineering "Engineering department"
    task prepare "Prepare everything for the assembling"
    task assemble "Assemble the bicycle"

flow received -> accepted
flow accepted -> reject "no"
flow reject -> rejected
flow accepted -> accept "yes"
flow accept -> informed
flow informed -> processParts -> checkQty -> available
flow available -> reserve "yes"
flow available -> backorder "no"
flow reserve -> itemsDone
flow backorder -> itemsDone
flow itemsDone -> checkQty "no"
flow itemsDone -> ready "yes"
flow informed -> prepare
flow prepare -> ready
flow ready -> assemble -> ship -> shipped
```
