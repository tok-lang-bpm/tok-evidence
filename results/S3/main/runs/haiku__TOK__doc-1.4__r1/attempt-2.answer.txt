```tok
tok 1.0 "Going public with underwriters"

pool company "Company"
  lane management "Management"
    start decide "Company decides to go public"
    task select "Select the underwriters"
    task advise "Provide procedural and financial advice"
    task prepare "Prepare the registration statement"
    task register "Register the issue with the SEC"
    task checkstates "Check compliance with the blue-sky laws of each state"
    task roadshow "Arrange a road show for potential investors"
    task firm "Firm up the issue price"
    xor clearance "SEC clearance received?"
    task fix "Fix the issue price"
    task buy "Enter into a firm commitment to buy the stock"
    xor reason "Reason not to offer found?"
    task offer "Offer the stock to the public"
    end offered "Stock offered to the public"
    end abandoned "Offering abandoned"

flow decide -> select -> advise
flow advise -> prepare
flow prepare -> register
flow prepare -> checkstates
flow register -> roadshow
flow checkstates -> roadshow
flow roadshow -> firm
flow firm -> clearance
flow clearance -> fix "yes"
flow clearance -> firm "no"
flow fix -> buy
flow buy -> reason
flow reason -> abandoned "yes"
flow reason -> offer "no"
flow offer -> offered
```
