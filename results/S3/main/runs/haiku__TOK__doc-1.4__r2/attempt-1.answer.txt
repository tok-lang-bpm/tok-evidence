```tok
tok 1.0 "Going public"

pool company "Company"
  lane management "Company management"
    start decide "Company decides to go public"
    task select "Select the underwriters"
  lane underwriters "Underwriters"
    task advise "Provide procedural and financial advice"
    task prepare "Prepare the registration statement"
    and split1
    task register "Register the issue with the SEC"
    task bluesky "Check compliance with blue-sky laws"
    task price "Firm up the issue price"
    task roadshow "Arrange a road show for potential investors"
    and joined1
    task fix "Fix the issue price"
    task commit "Enter a firm commitment to buy the stock"
    xor reason "Reason not to offer found?"
    task withdraw "Withdraw the offering"
    task offer "Offer the stock to the public"
    end withdrawn "Offering withdrawn"
    end offered "Stock offered to the public"
  lane sec "SEC"
    task approve "Approve the registration statement"

flow decide -> select -> advise -> prepare -> split1
flow split1 -> register -> approve -> joined1
flow split1 -> bluesky -> joined1
flow split1 -> price -> joined1
flow split1 -> roadshow -> joined1
flow joined1 -> fix -> commit -> reason
flow reason -> withdraw "yes"
flow reason -> offer "no"
flow withdraw -> withdrawn
flow offer -> offered
```
