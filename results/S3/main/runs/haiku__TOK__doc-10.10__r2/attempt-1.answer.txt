```tok
tok 1.0 "Interference handling"

pool org "Organisation"
  lane detector "Fault detector"
    start detected "Failure reported"
    task report "Report the failure to the MPO"
  lane mpo "MPO"
    start suspect "Own fault suspected"
    xor failure "Failure to examine"
    task examine "Examine the failure"
    xor verdict "Failure confirmed?"
    task reject "Reject the failure of the fault detector"
    task confirm "Confirm the failure of the fault detector"
    task informstart "Inform the GO and the MSP about the failure"
    task fix "Fix the fault at the measuring device"
    and repairs "Repairs carried out"
    task share "Share the results of the repairs with the fault detector"
    task informgo "Inform the GO about the resolution of the interference"
    task informmsp "Inform the MSP about the resolution of the interference"
    and resolved "Interference resolved"
    end rejected "Failure rejected"
    end done "Interference resolved"

flow detected -> report -> failure
flow suspect -> failure
flow failure -> examine -> verdict
flow verdict -> reject "rejects"
flow verdict -> confirm "confirms"
flow reject -> rejected
flow confirm -> informstart -> fix -> repairs
flow repairs -> share -> resolved
flow repairs -> informgo -> resolved
flow repairs -> informmsp -> resolved
flow resolved -> done
```
