```tok
tok 1.0 "Measurement request handling"

pool org "Organisation"
  lane go "GO"
    start request_needed "Measurements needed"
    task request_measurements "Request the measurements from the MSP"
    task process_values "Process the measured values"
    task send_changed "Send the changed values to the MSP"
    task transmit_readings "Transmit the readings to the SP"
    end transmitted "Readings transmitted"
    end reading_failed "Reading failure handled"
  lane msp "MSP"
    task check_request "Check the received request"
    xor request_ok "Request valid?"
    task deny_request "Deny the request of the GO"
    task perform_measurement "Perform the measurement"
    xor measurement_ok "Measurement successful?"
    task inform_failure "Inform the GO about the failure of the reading"
    task transmit_values "Transmit the measured values to the GO"
    task receive_changed "Receive the changed values"

flow request_needed -> request_measurements -> check_request -> request_ok
flow request_ok -> deny_request "no"
flow request_ok -> perform_measurement "yes"
flow deny_request -> reading_failed
flow perform_measurement -> measurement_ok
flow measurement_ok -> inform_failure "no"
flow measurement_ok -> transmit_values "yes"
flow inform_failure -> reading_failed
flow transmit_values -> process_values -> send_changed -> receive_changed -> transmit_readings -> transmitted
```
