```tok
tok 1.0 "Master data change"

pool org "Organisation"
  lane ec "EC"
    start needed "Master data change needed"
    task tell "Tell the INQ about the change of master data"
  lane inq "INQ"
    task notify "Notify the IP of the change"
  lane ip "IP"
    task check "Check whether the master data can be changed at the desired time"
    xor possible "Change possible?"
    task confirm "Confirm the changes of the INQ"
    task reject "Reject the changes of the INQ"
    end confirmed "Changes confirmed"
    end rejected "Changes rejected"

flow needed -> tell -> notify -> check -> possible
flow possible -> confirm "yes"
flow possible -> reject "no"
flow confirm -> confirmed
flow reject -> rejected
```
