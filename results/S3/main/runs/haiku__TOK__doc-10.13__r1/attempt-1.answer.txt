```tok
tok 1.0 "Transaction data request"

pool org "Organisation"
  lane inq "INQ"
    start requested "Transaction data request needed"
    task transmit "Transmit the transaction data request"
  lane ip "IP"
    task check "Check the request of the INQ"
    xor outcome "Outcome of the examination?"
    task transmitData "Answer the question of the INQ by transmitting the data"
    task reject "Answer the question of the INQ by rejecting the request"
    end answered "Data transmitted"
    end rejected "Request rejected"

flow requested -> transmit -> check -> outcome
flow outcome -> transmitData "data"
flow outcome -> reject "rejection"
flow transmitData -> answered
flow reject -> rejected
```
