```tok
tok 1.0 "Dismissal review"

pool org "Organisation"
  lane mpon "MPON"
    start needed "Dismissal needed"
    task send "Send the dismissal to the MPOO"
  lane mpoo "MPOO"
    task review "Review the dismissal"
    xor decide "Opposes or confirms the dismissal of the MPON?"
    task oppose "Oppose the dismissal of the MPON"
    task confirm "Confirm the dismissal of the MPON"
    end opposed "Dismissal opposed"
    end confirmed "Dismissal confirmed"

flow needed -> send -> review -> decide
flow decide -> oppose "opposes"
flow decide -> confirm "confirms"
flow oppose -> opposed
flow confirm -> confirmed
```
