  lane mpon "MPON"
    start needed "Dismissal needed"
    task send "Send the dismissal to the MPOO"
  lane mpoo "MPOO"
    task review "Review the dismissal"
    xor decide "Dismissal opposed or confirmed?"
    task oppose "Oppose the dismissal"
    task confirm "Confirm the dismissal"
    end opposed "Dismissal opposed"
    end confirmed "Dismissal confirmed"

flow needed -> send -> review -> decide
flow decide -> oppose "opposed"
flow decide -> confirm "confirmed"
flow oppose -> opposed
flow confirm -> confirmed
