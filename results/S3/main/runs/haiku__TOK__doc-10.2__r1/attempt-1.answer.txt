```tok
tok 1.0 "Meter operation registration"

pool org "Meter operation registration"
  lane mpon "MPON"
    start needed "Meter operation to report"
    task report "Report the meter operation to the GO"
    or eqsplit "Equipment acquisition and/or changes?"
    task acquire "Perform the equipment acquisition"
    or eqjoin "Equipment work done"
  lane go "GO"
    task examine "Examine the application of the MPON"
    xor review "Application confirmed?"
    task reject "Reject the application of the MPON"
    task confirm "Confirm the application of the MPON"
    task informMPOO1 "Inform the MPOO about the registration confirmation of the MPON"
    task informMSPO1 "Inform the MSPO about the registration confirmation of the MPON"
    eventgw wait "Report received before the deadline?"
    catch failrep "MPON informs the GO about the failure of the entire process" message="Failure of the entire process"
    catch succrep "MPON informs the GO about the successful completion of the entire process" message="Successful completion of the entire process"
    catch timeout "Maximum time limit exceeded" timer=P30D
    task failtx "Inform the MPON about the failure of the overall transaction"
    task confirmfail "Confirm the failure of the assignment to the MPON"
    task informMPOOf "Inform the MPOO about the failure of the assignment of the MPON"
    task informMSPOf "Inform the MSPO about the failure of the assignment of the MPON"
    task assign "Assign the MPON"
    task confirmassign "Confirm the assignment to the MPON"
    task informMPOOa "Inform the MPOO about the assignment of the MPON"
    task informMSPOa "Inform the MSPO about the assignment of the MPON"
    task informSP "Inform the SP about the assignment of the MPON"
    end endRej "Application rejected"
    end endFail "Registration failed"
    end endTimeout "Overall transaction failed by deadline"
    end endOk "Registration assigned"
  lane mpoo "MPOO"
    task changes "Perform the equipment changes"

flow needed -> report -> examine -> review
flow review -> reject "nie"
flow review -> confirm "tak"
flow reject -> endRej
flow confirm -> informMPOO1 -> informMSPO1 -> eqsplit
flow eqsplit -> acquire -> eqjoin
flow eqsplit -> changes -> eqjoin
flow eqjoin -> wait
flow wait -> failrep
flow wait -> succrep
flow wait -> timeout
flow failrep -> confirmfail -> informMPOOf -> informMSPOf -> endFail
flow timeout -> failtx -> endTimeout
flow succrep -> assign -> confirmassign -> informMPOOa -> informMSPOa -> informSP -> endOk
```
