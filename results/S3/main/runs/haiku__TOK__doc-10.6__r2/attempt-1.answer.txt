```tok
tok 1.0 "Dismissal"

pool org "MSP"
  lane msn "MSPN"
    start needed "Dismissal needed"
    task send "Send the dismissal"
  lane mso "MSPO"
    task review "Review the dismissal"
    xor decide "Decision?"
    task reject "Reject the dismissal"
    task confirm "Confirm the dismissal"
    end rejected "Dismissal rejected"
    end confirmed "Dismissal confirmed"

flow needed -> send -> review -> decide
flow decide -> reject "rejects"
flow decide -> confirm "confirms"
flow reject -> rejected
flow confirm -> confirmed
```
