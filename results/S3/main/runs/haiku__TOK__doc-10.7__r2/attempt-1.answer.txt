```tok
tok 1.0 "Measurement and assignment"

pool org "Organisation"
  lane mspn "MSPN"
    start needed "Measurement needed"
    task register "Register the measurement at the GO"
  lane go "GO"
    task examine "Examine the application of the MSPN"
    xor decide "Application confirmed?"
    task reject "Reject the application of the MSPN"
    task confirm "Confirm the application of the MSPN"
    task assign "Assign the MSPN"
    task informMspo "Inform the MSPO about the assignment of the MSPN"
    task informMpo "Inform the MPO about the assignment of the MSPN"
    task informSp "Inform the SP about the assignment of the MSPN"
    end rejected "Application rejected"
    end done "Assignment informed"

flow needed -> register -> examine -> decide
flow decide -> reject "no"
flow decide -> confirm "yes"
flow reject -> rejected
flow confirm -> assign -> informMspo -> informMpo -> informSp -> done
```
