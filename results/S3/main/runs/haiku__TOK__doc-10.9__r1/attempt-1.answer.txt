```tok
tok 1.0 "Measuring point change"

pool org "Organisation"
  lane requester "SP / PU / GO"
    start requested "Change requested"
    task request "Request the change to the MPO"
  lane mpo "MPO"
    start mpoStart "Change caused by the MPO"
    task cause "Cause a change of the measuring point"
    xor merged
    task review "Review the change request"
    xor decide "Change request confirmed?"
    task confirm "Confirm the request of the SP / PU / GO"
    task reject "Reject the change of the measuring point"
    end rejected "Change rejected"
    task perform "Perform the measuring point change"
    xor outcome "Implementation successful?"
    task report "Report the implementation to the SP / PU / GO"
    task notifyFail "Notify the SP / PU / GO about the failure of the changes"
    end implemented "Change implemented"
    end failed "Changes failed"

flow requested -> request -> merged
flow mpoStart -> cause -> merged
flow merged -> review -> decide
flow decide -> confirm "yes"
flow decide -> reject "no"
flow confirm -> perform
flow reject -> rejected
flow perform -> outcome
flow outcome -> report "yes"
flow outcome -> notifyFail "no"
flow report -> implemented
flow notifyFail -> failed
```
