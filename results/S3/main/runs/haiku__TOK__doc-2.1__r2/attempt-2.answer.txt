tok 1.0 "Service degradation handling"

pool teleco "TELECO"
  lane customer "Customer"
    start degr "Service degradation perceived"
    task send_params "Send list of problem parameters to Customer Service"
    catch cust_recv "Report received" link="cust-report"
    end cust_end "Customer process ended"
  lane cs "Customer Service"
    task enter_report "Enter problem report into system T"
    task compare_sla "Compare problem report with customer SLA"
    task det_counter "Determine counter measures and their priorities"
    xor premium "Premium customer?"
    call fix_proc "Run extra problem fix process" process=extra-problem-fix
    end fix_end "Handled in extra problem fix process"
    xor signif "Customer significance affects counter measures?"
    task reprio "Re-prioritize counter measures"
    xor prio_merge
    task create_detail "Create detailed problem report"
    task send_detail "Send detailed problem report to Service Management"
    xor cs_in
    and cs_fork
    task create_cust_report "Create report for the customer on service performance and problem resolution"
    task send_cust_report "Send report to the customer"
    task rebate_report "Report SLA violation rebate to Billing & Collections"
    and cs_join
    throw cs_out "Forward report to customer" link="cust-report"
  lane esvc "Electronic service"
    service det_signif "Determine significance of customer based on contract history"
  lane sm "Service Management"
    task investigate "Investigate whether problem is analyzable at Service Management level"
    xor an_gw "Analyzable by Service Management?"
    task analyze_sm "Analyze the problem"
    task trouble_sm "Create trouble report indicating problem type"
    task send_rp "Send detailed problem report to Resource Provisioning"
    task sm_recv_status "Receive status report"
    xor sm_merge1
    task prep_final "Prepare final status report"
    task det_cm "Determine counter measures based on final status report"
    xor cm_gw "Counter measures required?"
    task send_perf "Send actual service performance to Customer Service"
    task corr_sm "Undertake minor corrective actions"
    task sm_recv_ts "Receive trouble-shooting report"
    xor merge2
    task resol "Create problem resolution report"
    task send_resol "Send problem resolution report to Customer Service"
    task req_restore "Create request for automatic resource restoration"
    task send_req "Send restoration request to Resource Provisioning"
  lane rp "Resource Provisioning"
    task rp_recv "Receive detailed problem report"
    task check_prob "Check for possible problems"
    xor prob_gw "Problem detected?"
    task notify_normal "Create notification about normal service execution"
    task analyze_rp "Analyze the problem"
    task trouble_rp "Create trouble report"
    xor rp_merge
    task include_status "Include trouble report or normal execution notification in status report"
    task send_status "Send status report to Service Management"
    eventgw rp_wait "Wait for restoration request"
    catch restore_rcv "Restoration request received" message="Restoration request"
    catch rp_timeout "Two days elapsed" timer=P2D
    end rp_end "Resource Provisioning terminates the process"
    task track_err "Track all possible errors"
    task corr_rp "Undertake all necessary corrective actions"
    task trouble_shoot "Create trouble-shooting report"
    task send_ts "Send trouble-shooting report to Service Management"
  lane bc "Billing & Collections"
    task adjust_billing "Adjust the billing"

flow degr -> send_params -> enter_report -> compare_sla -> det_counter -> det_signif -> premium
flow premium -> fix_proc "yes"
flow fix_proc -> fix_end
flow premium -> signif "no"
flow signif -> reprio "yes"
flow reprio -> prio_merge
flow signif -> prio_merge "no"
flow prio_merge -> create_detail -> send_detail -> investigate -> an_gw
flow an_gw -> analyze_sm "yes"
flow analyze_sm -> trouble_sm -> sm_merge1
flow an_gw -> send_rp "no"
flow send_rp -> rp_recv -> check_prob -> prob_gw
flow prob_gw -> notify_normal "no"
flow notify_normal -> rp_merge
flow prob_gw -> analyze_rp "yes"
flow analyze_rp -> trouble_rp -> rp_merge
flow rp_merge -> include_status -> send_status -> sm_recv_status -> sm_merge1
flow sm_merge1 -> prep_final -> det_cm -> cm_gw
flow cm_gw -> send_perf "no problem detected"
flow send_perf -> cs_in
flow cm_gw -> corr_sm "minor corrective actions"
flow corr_sm -> merge2
flow cm_gw -> req_restore "automatic resource restoration"
flow req_restore -> send_req -> rp_wait
flow merge2 -> resol -> send_resol -> cs_in
flow rp_wait -> restore_rcv "request received"
flow restore_rcv -> track_err -> corr_rp -> trouble_shoot -> send_ts -> sm_recv_ts -> merge2
flow rp_wait -> rp_timeout "no request"
flow rp_timeout -> rp_end
flow cs_in -> cs_fork
flow cs_fork -> create_cust_report -> send_cust_report -> cs_join
flow cs_fork -> rebate_report -> adjust_billing -> cs_join
flow cs_join -> cs_out
flow cust_recv -> cust_end
