```tok
tok 1.0 "Customer switch-over"

pool supplier "New supplier"
  lane customer "Customer"
    start request "Switch-over request"
    task transmit "Transmit customer data to customer service"
    task checkDetails "Check contract details"
    eventgw decide "Decide within 7 days?"
    catch withdraw "Withdraw from contract" message="withdrawal"
    catch confirm "Confirm contract" message="confirmation"
    catch timer7 "7 days elapsed" timer=P7D
    end withdrawn "Switch contract withdrawn"
    task receiveBillCust "Receive final billing"
    end custEnd "Switch process ended for customer"
  lane cs "Customer service"
    task receiveData "Receive customer data"
    task enter "Enter customer data object into CRM system"
    task compare "Compare customer data with internal customer database"
    task check "Check completeness and plausibility of customer data"
    xor errors "Errors found?"
    task correct "Correct errors based on simple error list"
    xor errorsMerged
    xor exists "Customer exists in customer database?"
    task createCust "Create new customer object"
    xor existsMerged
    task prepare "Prepare contract documents with customer object, bank connection, rate and requested switch-over date"
    task analyze "Analyze and resolve causing issues"
    task confirmDoc "Send confirmation document to customer"
    task createContract "Create CIS contract"
    xor acceptMerged
    task importMeter "Import meter data into systems that require the information"
    end won "New customer won"
  lane cis "CIS"
    task checkDocs "Automatically check contract documents"
    xor docsOk "Documents generated successfully?"
    task regenerate "Generate contract documents again"
    task sendRequest "Automatically send switch-over request to grid operator"
  lane grid "Grid operator"
    task checkRequest "Check switch-over request for supplier concurrence"
    task respond "Transmit response comment"
    xor concur "Supplier concurrence?"
    task informSup "Inform all involved suppliers"
    task demand "Demand resolution of the conflict"
    task communicate "Communicate with old supplier"
    task terminate "Carry out termination of sales agreement"
    xor concurMerged
    and meterSplit
    task transmitMeter "Transmit power meter data consumption report"
    task computeBill "Compute final billing based on meter data"
    task sendBill "Send final billing to old supplier"
    and billSplit
    and meterOut
    end gridEnd "Switch process ended for grid operator"
  lane old "Old supplier"
    task resolve "Resolve the conflict"
    task receiveMeter "Receive power meter data"
    task receiveBill "Receive final billing from grid operator"
    and oldJoin
    task createBill "Create final billing for customer"
    task sendBillCust "Send final billing to customer"
    and oldSplit
    end oldEnd "Final billing sent"

flow request -> transmit -> receiveData -> enter -> compare -> check -> errors
flow errors -> correct "yes"
flow errors -> errorsMerged "no"
flow correct -> errorsMerged
flow errorsMerged -> exists
flow exists -> createCust "no"
flow exists -> existsMerged "yes"
flow createCust -> existsMerged
flow existsMerged -> prepare -> checkDocs -> docsOk
flow docsOk -> confirmDoc "yes"
flow docsOk -> analyze "no"
flow analyze -> regenerate -> checkDocs
flow confirmDoc -> sendRequest -> checkRequest -> respond -> concur
flow concur -> informSup "yes"
flow concur -> concurMerged "no"
flow informSup -> demand -> resolve -> communicate -> terminate -> concurMerged
flow concurMerged -> createContract -> checkDetails -> decide
flow decide -> withdraw
flow decide -> confirm
flow decide -> timer7
flow withdraw -> withdrawn
flow confirm -> acceptMerged
flow timer7 -> acceptMerged
flow acceptMerged -> meterSplit
flow meterSplit -> transmitMeter -> meterOut
flow meterSplit -> computeBill -> sendBill -> billSplit
flow meterOut -> importMeter -> won
flow meterOut -> receiveMeter -> oldJoin
flow billSplit -> gridEnd
flow billSplit -> receiveBill -> oldJoin
flow oldJoin -> createBill -> sendBillCust -> oldSplit
flow oldSplit -> oldEnd
flow oldSplit -> receiveBillCust -> custEnd
```
