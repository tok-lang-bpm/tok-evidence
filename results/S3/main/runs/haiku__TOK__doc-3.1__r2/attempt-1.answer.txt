```tok
tok 1.0 "Warrant possession request handling"

pool party "Party"
pool police "Queensland Police"
pool staff "MC internal staff"

pool office "Small Claims Registry Operations"
  lane back "Client Service Back Office"
    start requested "Warrant possession request received" message="Warrant possession request"
    task receive "Receive the request"
    task retrieve "Retrieve the SCT file"
    task forward "Forward the SCT Warrant Possession to Queensland Police"
    task store "Store the SCT physical file awaiting the police report"
    catch report "Police report received" message="Police report"
    task retrieve-report "Retrieve the respective SCT file"
    task attach "Attach the new SCT document"
    task store-expanded "Store the expanded SCT physical file"
    task handover "Hand over the physical SCT file to other MC internal staff"
    end done "SCT file handed over"

message party -> requested "Warrant possession request"
message forward -> police "Warrant possession"
message police -> report "Police report"
message handover -> staff "Physical SCT file"

flow requested -> receive -> retrieve -> forward -> store -> report -> retrieve-report -> attach -> store-expanded -> handover -> done
```
