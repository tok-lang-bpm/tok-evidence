```tok
tok 1.0 "Preparing files for the court hearing"

pool court "Court"
  lane registry "Registry clerk"
    start morning "Each morning"
    task check "Check the files yet to be processed"
    xor missing "Files missing?"
    task search "Initiate a search"
    task track "Track the files physically to the intended location"
    xor ready
    and split
    task hand "Hand the files to the Associate"
    task distribute "Distribute the Judge's Lawlist to the relevant people"
    and joined
  lane judge "Court"
    task conduct "Conduct the directions hearings"
    end finished "Directions hearings conducted"

flow morning -> check -> missing
flow missing -> search "yes"
flow missing -> track "no"
flow search -> ready
flow track -> ready
flow ready -> split
flow split -> hand
flow split -> distribute
flow hand -> joined
flow distribute -> joined
flow joined -> conduct -> finished
```
