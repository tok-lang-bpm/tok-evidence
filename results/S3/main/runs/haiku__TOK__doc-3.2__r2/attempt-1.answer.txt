```tok
tok 1.0 "Court file preparation before directions hearings"

pool court "Court"
  lane registry "Registry"
    start morning "Each morning"
    task check "Check the files yet to be processed"
    xor missing "Files missing?"
    task search "Initiate a search for the missing files"
    xor merged
    task track "Physically track the files to the intended location"
    and ready
    task handover "Hand the files over to the Associate"
    task distribute "Distribute the Judge's Lawlist to the relevant people"
    and joined
  lane bench "Court hearing"
    task hearings "Conduct the directions hearings"
    end done "Directions hearings conducted"

flow morning -> check -> missing
flow missing -> search "yes"
flow missing -> merged "no"
flow search -> merged
flow merged -> track -> ready
flow ready -> handover -> joined
flow ready -> distribute -> joined
flow joined -> hearings -> done
```
