```tok
tok 1.0 "Car insurance claim handling"

pool insurer "Car insurer"
  lane customer "Customer"
    start claimed "Claim to be submitted"
    task submit "Submit the claim with the relevant documentation"
  lane notification "Notification department"
    task checkdocs "Check the documents for completeness"
    task register "Register the claim"
  lane handling "Handling department"
    task pickup "Pick up the claim"
    task checkins "Check the insurance"
    task assess "Perform the assessment"
    xor positive "Assessment positive?"
    task phone "Phone the garage to authorise the repairs"
    task schedule "Schedule the payment"
    task reject "Reject the claim"
    xor outcome
    task letter "Send a letter to the customer"
    end done "Claim handling complete"

flow claimed -> submit -> checkdocs -> register -> pickup -> checkins -> assess -> positive
flow positive -> phone "positive"
flow phone -> schedule -> outcome
flow positive -> reject "negative"
flow reject -> outcome
flow outcome -> letter -> done
```
