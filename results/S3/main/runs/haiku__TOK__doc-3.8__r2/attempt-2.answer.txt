```tok
tok 1.0 "Car insurance claim handling"

pool insurer "Car insurer"
  lane customer "Customer"
    start submitted "Claim submitted"
    task submit "Submit a claim with relevant documentation"
    end complete "Process complete"
  lane notification "Notification department"
    task check_docs "Check the documents for completeness"
    xor complete_docs "Documents complete?"
    task register "Register the claim"
    task reject_docs "Request the missing documents"
  lane handling "Handling department"
    task check_ins "Check the insurance"
    task assess "Perform an assessment"
    xor positive "Assessment positive?"
    task phone_garage "Phone the garage to authorise the repairs"
    task schedule_pay "Schedule the payment"
    task reject "Reject the claim"
    xor outcome "Outcome known"
    task letter "Send a letter to the customer"

flow submitted -> submit -> check_docs -> complete_docs
flow complete_docs -> register "yes"
flow complete_docs -> reject_docs "no"
flow reject_docs -> check_docs
flow register -> check_ins -> assess -> positive
flow positive -> phone_garage "yes"
flow positive -> reject "no"
flow phone_garage -> schedule_pay -> outcome
flow reject -> outcome
flow outcome -> letter -> complete
```
