```tok
tok 1.0 "Loan approval"

pool bank "Loan approval process"
  lane customer "Customer"
    start needed "Loan request needed"
    task submit "Submit the request for a loan amount"
    task feedback "Receive the feedback"
    end informed "Customer informed"
  lane system "Loan system"
    task receive "Receive the customer request"
    xor highrisk "Customer high risk?"
    xor small "Loan small and customer low risk?"
    task approve "Approve the loan"
    task deny "Deny the loan"
    task sendapp "Send the request to the approver service"
    xor merged "Decision reached?"
    task sendfb "Send the feedback to the customer"
  lane risk "Risk assessment service"
    service assess "Assess the loan request"
  lane approver "Approver service"
    service decide "Decide on the loan request"

flow needed -> submit -> receive -> assess -> highrisk
flow highrisk -> deny "tak" when="customer is high risk"
flow highrisk -> small "nie" default
flow small -> approve "tak" when="loan is small and customer is low risk"
flow small -> sendapp "nie" default
flow approve -> merged
flow deny -> merged
flow sendapp -> decide -> merged
flow merged -> sendfb -> feedback -> informed
```
