```tok
tok 1.0 "Loan approval"

pool bank "Bank"
  lane system "Loan system"
    start started "Loan approval process started"
    receive request "Receive customer request for a loan amount" message="Loan request"
    service assess "Invoke risk assessment Web service to assess the request"
    xor decide "Loan small and low risk, high risk, or needs review?"
    task approve "Approve the loan"
    task deny "Deny the loan"
    service approver "Send the request to the approver Web service"
    xor merged
  lane customer "Customer"
    receive feedback "Receive feedback from the assessor or approver"
    end done "Loan process ended"

flow started -> request -> assess -> decide
flow decide -> approve "small and low risk" when="loanAmount < 10000 and lowRisk"
flow decide -> deny "high risk" when="highRisk"
flow decide -> approver "further review or amount 10,000 or more" when="needsReview or loanAmount >= 10000"
flow approve -> merged
flow deny -> merged
flow approver -> merged
flow merged -> feedback -> done
```
