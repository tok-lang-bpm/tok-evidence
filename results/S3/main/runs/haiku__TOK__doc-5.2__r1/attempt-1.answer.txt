```tok
tok 1.0 "Vacations Request"

pool org "Organization"
  lane employee "Employee"
    start needed "Vacation needed"
    task submit "Submit the vacation request"
    task register "Register the vacation request"
    task review "Review the rejection reasons"
    end returned "Application returned to the employee"
  lane supervisor "Immediate supervisor"
    task receive "Receive the vacation request"
    xor decide "Request approved?"
    task approve "Approve the vacation request"
    task reject "Reject the vacation request"
    task returnapp "Return the application to the employee"
    task notify "Generate the notification to the Human Resources Representative"
  lane hr "Human Resources Representative"
    task complete "Complete the management procedures"
    end done "Vacation request processed"

flow needed -> submit -> register -> receive -> decide
flow decide -> approve "yes"
flow decide -> reject "no"
flow approve -> notify -> complete -> done
flow reject -> returnapp -> review -> returned
```
