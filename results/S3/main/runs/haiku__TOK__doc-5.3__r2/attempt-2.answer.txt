```tok
tok 1.0 "Office supply request"

pool org "Organization"
  lane employee "Employee"
    start needed "Office supply needed"
    task submit "Submit an office supply request"
    task review "Review the comments for the change"
    end changed "Change requested"
  lane system "System"
    task register "Register the requirement"
    task notifyChange "Notify the user of the change request"
    task notifyReject "Notify the user of the rejection"
    task notifyApproval "Notify the user of the approval"
    task genOrder "Generate the purchase order"
    task sendOrder "Send the purchase order"
    task notifyResult "Notify the user of the result"
    end rejected "Request rejected"
    end completed "Request completed"
  lane supervisor "Immediate supervisor"
    task receive "Receive the office supply request"
    xor decide "Approve, request a change or reject?"
    task approve "Approve the request"
    task requestChange "Request a change"
    task reject "Reject the request"
  lane purchase "Purchase department"
    sub quotes "Make the quotations"
      start quoteNeeded "Quotations needed"
      task requestQuotes "Request the vendor quotations"
      end quotesDone "Quotations made"
    task select "Select a vendor"
    xor valid "Vendor valid in the system?"
    task choose "Choose a different vendor"
    task confirm "Confirm the selected vendor"
    and dispatch
    task deliver "Receive the delivered product"
    task invoice "Receive the invoice"
    and received

flow needed -> submit -> register -> receive -> decide
flow decide -> approve "approved"
flow decide -> requestChange "change required"
flow decide -> reject "rejected"
flow approve -> notifyApproval -> quotes
flow requestChange -> notifyChange -> review -> changed
flow reject -> notifyReject -> rejected
flow quoteNeeded -> requestQuotes -> quotesDone
flow quotes -> select -> valid
flow valid -> confirm "yes"
flow valid -> choose "no"
flow choose -> valid
flow confirm -> genOrder -> sendOrder -> dispatch
flow dispatch -> deliver -> received
flow dispatch -> invoice -> received
flow received -> notifyResult -> completed
```
