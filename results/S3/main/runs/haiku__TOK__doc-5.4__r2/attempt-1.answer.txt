```tok
tok 1.0 "Expense report approval"

pool org "Company"
  lane employee "Employee"
    start needed "Product or service required"
    task purchase "Purchase the required product or service"
    task submit "Submit the expense report with the item list and receipts"
    task edit "Edit the expense report"
    xor resubmitted
  lane supervisor "Supervisor"
    xor rules "Expense rules decide on first inspection?"
    task inspect "Accept or reject the report on first inspection"
    task review "Review the expense report"
    task decide "Approve or reject the expense report"
    xor decided
    xor approved "Report approved?"
  lane treasurer "Treasurer"
    task check "Check that all receipts are submitted and match the items"
    xor inorder "Receipts complete and matching?"
    task accept "Accept the expenses for processing"
    task return "Send the report back to the employee"
  lane payments "Payment system"
    service pay "Link the report to the payment system"
    catch confirmed "Payment confirmed" message="Payment confirmed"
    end paid "Expenses paid"

flow needed -> purchase -> submit -> resubmitted -> rules
flow rules -> inspect "yes"
flow rules -> review "no"
flow inspect -> decided
flow review -> decide -> decided
flow decided -> approved
flow approved -> check "yes"
flow approved -> edit "no"
flow edit -> resubmitted
flow check -> inorder
flow inorder -> accept "yes"
flow inorder -> return "no"
flow return -> edit
flow accept -> pay -> confirmed -> paid
```
