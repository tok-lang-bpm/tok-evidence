```tok
tok 1.0 "Monthly product parts ordering"

pool assembler "Assembler AG"
  lane buyer "Assembler AG"
    start monthly "Process starts on the first of each month" timer=P1M
    task place "Place the order with the supplier"
    send order "Send the order to the supplier" message="Order"
    receive invoice "Receive the invoice" message="Invoice"
    end done "Order cycle completed"
  lane supplier "Supplier"
    task process "Process the order"
    send sendinvoice "Send the invoice to Assembler AG" message="Invoice"

flow monthly -> place -> order -> process -> sendinvoice -> invoice -> done
```
