```tok
tok 1.0 "Sales process for potential customers"

pool office "Company"
  lane company "Company"
    start needed "Contact details of potential customers needed"
    xor source "Which way to get the contact details?"
    task buy "Buy contact details for cold calls"
    task acquaintance "Get contact details of an acquaintance interested in the product"
    xor found "Contact details obtained"
    xor presplit "Which presentation?"
    xor presdone "Presentation given"
    eventgw wait "Customer comes back?"
    catch cameback "Customer comes back" message="Customer contacts us"
    catch tooLong "No contact within 2 weeks" timer=P14D
    xor quotemerge "Customer contacted"
    task quote "Create a quotation"
    end finished "Quotation created"
  lane marketing "Marketing staff"
    task exhibit "Participate in exhibitions"
  lane callcenter "Call center staff"
    task call "Call the potential customer"
    task determine "Determine the contact person and the budget"
    task ask "Ask whether the customer is generally interested"
    xor gen "Generally interested?"
    xor huge "Potential budget huge?"
    task leave "Leave the customer alone"
    xor ready "Customer interested?"
    end gone "Customer left alone"
  lane head "Head of development"
    task acquire "Acquire the customer personally"
    xor endint "Customer interested in the end?"
  lane sales "Sales representative"
    task salespres "Give detailed online presentation"
    task callback "Call the customer"
  lane presales "Pre-sales employee"
    task techpres "Give detailed technical online presentation"

flow needed -> source
flow source -> buy "cold calls"
flow source -> exhibit "exhibitions"
flow source -> acquaintance "acquaintance"
flow buy -> found
flow exhibit -> found
flow acquaintance -> found
flow found -> call -> determine -> ask -> gen
flow gen -> ready "yes"
flow gen -> huge "no"
flow huge -> acquire "yes"
flow huge -> leave "no"
flow acquire -> endint
flow endint -> ready "yes"
flow endint -> leave "no"
flow leave -> gone
flow ready -> presplit
flow presplit -> salespres "standard"
flow presplit -> techpres "technical"
flow salespres -> presdone
flow techpres -> presdone
flow presdone -> wait
flow wait -> cameback
flow wait -> tooLong
flow cameback -> quotemerge
flow tooLong -> callback
flow callback -> quotemerge
flow quotemerge -> quote -> finished
```
