```tok
tok 1.0 "Customer acquisition"

pool sales "Sales organisation"
  lane marketing "Marketing staff"
    start needed "Potential customer needed"
    xor source "Contact source?"
    task buy "Buy contact details for cold calls"
    task exhibit "Participate in exhibitions"
    task known "Obtain contact via acquaintance"
    xor sourced "Contact details obtained"
  lane callcenter "Call center staff"
    task call "Call the customer"
    task determine "Determine contact person and available budget"
    task ask "Ask whether the customer is generally interested"
    xor interested "Customer interested?"
    xor budget "Potential project budget huge?"
    task leave "Leave the customer alone"
  lane headdev "Head of development"
    task acquire "Personally acquire the customer"
  lane presales "Sales representative / pre-sales employee"
    xor kind "Technical presentation?"
    task online "Give detailed online presentation (sales representative)"
    task techonline "Give detailed technical online presentation (pre-sales employee)"
    xor back "Customer contacted us within 2 weeks?"
    task recall "Call the customer"
    task quote "Create a quotation"
    end done "Quotation created"
    end lost "Customer not pursued"

flow needed -> source
flow source -> buy "cold calls" 
flow source -> exhibit "exhibitions"
flow source -> known "acquaintance"
flow buy -> sourced
flow exhibit -> sourced
flow known -> sourced
flow sourced -> call
flow call -> determine -> ask -> interested
flow interested -> leave "no" when="false"
flow interested -> kind "yes" when="true"
flow leave -> budget
flow budget -> acquire "yes" when="true"
flow budget -> lost "no" default
flow acquire -> kind
flow kind -> online "sales representative"
flow kind -> techonline "pre-sales employee"
flow online -> back
flow techonline -> back
flow back -> quote "yes" when="true"
flow back -> recall "no" default
flow recall -> quote
flow quote -> done
```
