```tok
tok 1.0 "Lead processing"

pool sales "Sales department"
  lane manager "Manager"
    start review "Lead review needed"
    task check "Check the open leads"
    task select "Select the top five leads"
    task instruct "Tell the Sales Assistant to call the contacts"
    task process "Process the lead"
    end processed "Lead processed"
    end next "Next customer called"
  lane assistant "Sales Assistant"
    task call "Call the customer"
    xor interested "Customer interested?"
    task note "Send a note to the Manager"

flow review -> check -> select -> instruct -> call -> interested
flow interested -> note "yes"
flow note -> process -> processed
flow interested -> next "no"
```
