```tok
tok 1.0 "Filling a post"

pool org "Organisation"
  lane dept "Functional department"
    start demand "Demand to fill a post"
  lane hr "HR department"
    task advertise "Advertise the post"
    task check "Check the applications"
    task fill "Fill the post"
    task contract "Conclude the contract of employment"
    end filled "Post filled"
  lane applicant "Applicant"
    task apply "Apply for the post"

flow demand -> advertise -> apply -> check -> fill -> contract -> filled
```
