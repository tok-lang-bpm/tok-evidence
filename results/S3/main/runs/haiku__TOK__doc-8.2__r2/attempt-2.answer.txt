```tok
tok 1.0 "Job posting"

pool org "Organisation"
  lane dept "Department"
    start reported "Vacancy reported"
    task report "Report the vacancy"
    task answer "Answer the question"
    task consider "Consider the job description"
    xor approved "Approved?"
    task reject "Reject the job description and request a correction"
  lane hr "HR clerk"
    task create "Create the job description"
    xor confused "Confusion in the message?"
    task ask "Ask the Department again"
    xor clarified
    task submit "Submit the job description for consideration"
    task correct "Correct the job description"
    task post "Post the job"
    end posted "Job posted"

flow reported -> report -> create -> confused
flow confused -> ask "yes" when="confusion = true"
flow confused -> clarified "no" default
flow ask -> answer -> clarified
flow clarified -> submit -> consider -> approved
flow approved -> post "yes" when="approved = true"
flow approved -> reject "no" default
flow reject -> correct -> submit
flow post -> posted
```
