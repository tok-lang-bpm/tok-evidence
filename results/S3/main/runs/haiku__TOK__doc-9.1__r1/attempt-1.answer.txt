```tok
tok 1.0 "Daily defaulter processing"

pool org "Organisation"
  lane sys "Reporting system"
    start morning "Weekday morning"
    task backup "Back up the database"
    task checkTable "Check the Account Defaulter table for new records"
    xor newRecords "New records found?"
    end recordsDone "Records processed"
    task checkCrm "Check the CRM system for new returns"
    xor newReturns "New returns filed?"
    end noReturns "No new returns"
    task register "Register all defaulting accounts and customers"
    xor advised "Defaulting client codes previously advised?"
    task produceTable "Produce a table of defaulting accounts"
    task sendTable "Send the table to account management"
    xor merge1
    xor deadline1 "Completed by 2:30 pm?"
    task alert "Send an alert to the supervisor"
    xor merge2
    task checkCrm2 "Check the CRM system for new returns"
    xor newReturns2 "New returns filed?"
    end noReturns2 "No new returns"
    task reconcile "Reconcile with the existing Account Defaulter table"
    xor deadline2 "Completed by 4:00 pm?"
    task messageSup "Send a message to the supervisor"
    end done "Process completed"
    end late "Supervisor informed of late completion"

flow morning -> backup -> checkTable -> newRecords
flow newRecords -> recordsDone "yes"
flow newRecords -> checkCrm "no"
flow checkCrm -> newReturns
flow newReturns -> noReturns "no"
flow newReturns -> register "yes"
flow register -> advised
flow advised -> produceTable "yes"
flow advised -> merge1 "no"
flow produceTable -> sendTable -> merge1
flow merge1 -> deadline1
flow deadline1 -> merge2 "yes"
flow deadline1 -> alert "no"
flow alert -> merge2
flow merge2 -> checkCrm2 -> newReturns2
flow newReturns2 -> reconcile "yes"
flow newReturns2 -> noReturns2 "no"
flow reconcile -> deadline2
flow deadline2 -> done "yes"
flow deadline2 -> messageSup "no"
flow messageSup -> late
```
