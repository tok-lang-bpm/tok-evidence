```tok
tok 1.0 "Annual meeting schedule"

pool authority "Town Planning Authority"
  lane coord "Coordination Unit"
    start begin "November of each year"
    task draft "Draft the schedule of meetings for the next calendar year"
    task addcal "Add draft dates to all calendars"
    task recheck "Recheck all dates"
    task conflicts "Look for potential conflicts"
    task sendsched "Send the final schedule of meeting dates to all Committee Members by email"
  lane support "Support Officer"
    task check "Check the dates"
    xor modq "Modifications suggested?"
    task suggest "Suggest modifications"
    xor modmerge
  lane members "Committee Members"
    task diaries "Check their diaries"
    xor confq "Conflicts found?"
    task advise "Advise the Coordination Unit of conflicts"
    end noconf "No conflicts"
    end done "Conflicts reported to the Coordination Unit"

flow begin -> draft -> addcal -> check -> modq
flow modq -> suggest "yes" when="modifications_suggested = true"
flow suggest -> modmerge
flow modq -> modmerge "no" default
flow modmerge -> recheck -> conflicts -> sendsched -> diaries -> confq
flow confq -> advise "yes" when="conflicts_found = true"
flow advise -> done
flow confq -> noconf "no" default
```
