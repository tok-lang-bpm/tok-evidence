```tok
tok 1.0 "Committee meeting process"

pool org "Organisation"
  lane cu "Coordination Unit"
    start s "Meeting dates to be finalized"
    task finalize "Finalize the dates"
    task qa "Conduct a QA check on the minutes within 5 working days"
  lane so "Support Officer"
    and prep_split
    task calendars "Update all group calendars"
    task folders "Create meeting folders for each meeting"
    task upload "Upload all appropriate documents to the system"
    and prep_join
    task advise "Advise Committee Members a week before each meeting"
    task produce "Produce the minutes including any Action Points for each Committee Member"
    task send "Send the minutes to all Committee Members"
    task records "Update all departmental records"
    end done "Process completed"
  lane cm "Committee Members"
    task read "Read all related documents"
    task hold "Hold the meeting"

flow s -> finalize -> prep_split
flow prep_split -> calendars
flow prep_split -> folders
flow prep_split -> upload
flow calendars -> prep_join
flow folders -> prep_join
flow upload -> prep_join
flow prep_join -> advise -> read -> hold -> produce -> qa -> send -> records -> done
```
