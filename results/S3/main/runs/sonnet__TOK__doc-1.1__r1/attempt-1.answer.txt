```tok
tok 1.0 "Customized bicycle order"

pool company "Bicycle manufacturer"
  lane sales "Sales department"
    start received "Order received"
    task review "Review the order"
    xor accepted "Order accepted?"
    task reject "Reject the order"
    end rejectedEnd "Order rejected"
    and inform "Inform"
    task informStore "Inform the storehouse"
    task informEng "Inform the engineering department"
    task ship "Ship the bicycle to the customer"
    end shipped "Process finished"
  lane storehouse "Storehouse"
    task partList "Process the part list"
    task checkQty "Check the required quantity of the part"
    xor available "Part available in-house?"
    task reserve "Reserve the part"
    task backorder "Back-order the part"
    xor merged
    xor more "More items on the part list?"
  lane engineering "Engineering department"
    task prepare "Prepare the assembling of the bicycle"
    and joined
    task assemble "Assemble the bicycle"

flow received -> review -> accepted
flow accepted -> reject "reject"
flow accepted -> inform "accept"
flow reject -> rejectedEnd
flow inform -> informStore
flow inform -> informEng
flow informStore -> partList -> checkQty -> available
flow available -> reserve "yes"
flow available -> backorder "no"
flow reserve -> merged
flow backorder -> merged
flow merged -> more
flow more -> checkQty "yes"
flow more -> joined "no"
flow informEng -> prepare -> joined
flow joined -> assemble -> ship -> shipped
```
