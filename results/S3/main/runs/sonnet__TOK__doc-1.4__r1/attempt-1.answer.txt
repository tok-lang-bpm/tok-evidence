```tok
tok 1.0 "Going public"

pool ipo "Going public"
  lane company "Company"
    start decided "Decision to go public"
    task select "Select the underwriters"
  lane uw "Underwriters"
    task prepare "Prepare the registration statement"
    and split
    task register "Register the issue with the SEC"
    task check "Check compliance with the blue-sky laws"
    task firm "Firm up the issue price"
    task roadshow "Arrange a road show"
    and join
    task fix "Fix the issue price"
    task commit "Enter into a firm commitment to buy the stock"
    xor reason "Reason not to offer the stock?"
    task offer "Offer the stock to the public"
    end sold "Stock offered to the public"
    end stopped "Stock not offered"
  lane sec "SEC"
    task approve "Approve the registration statement"

flow decided -> select -> prepare -> split
flow split -> register
flow split -> check
flow split -> firm
flow register -> approve
flow approve -> join
flow check -> join
flow firm -> roadshow
flow roadshow -> join
flow join -> fix -> commit -> reason
flow reason -> stopped "yes"
flow reason -> offer "no"
flow offer -> sold
```
