```tok
tok 1.0 "Fault handling"

pool org "Organisation"
  lane detector "Fault detector"
    start reported "Failure detected"
    task report "Report the failure to the MPO"
  lane mpo "MPO"
    start suspected "Own fault suspected"
    task suspect "Suspect own fault"
    xor merge
    task examine "Examine the failure"
    xor valid "Failure confirmed?"
    task reject "Reject the failure of the fault detector"
    task confirm "Confirm the failure of the fault detector"
    and inform1 
    task informgo "Inform the GO"
    task informmsp "Inform the MSP"
    and informed1
    task fix "Fix the fault at the measuring device"
    task share "Share the results of the repairs with the fault detector"
    and inform2
    task resgo "Inform the GO about the resolution of the interference"
    task resmsp "Inform the MSP about the resolution of the interference"
    and informed2
    end rejected "Failure rejected"
    end resolved "Fault resolved"

flow reported -> report -> merge
flow suspected -> suspect -> merge
flow merge -> examine -> valid
flow valid -> reject "no"
flow valid -> confirm "yes"
flow reject -> rejected
flow confirm -> inform1
flow inform1 -> informgo
flow inform1 -> informmsp
flow informgo -> informed1
flow informmsp -> informed1
flow informed1 -> fix -> share -> inform2
flow inform2 -> resgo
flow inform2 -> resmsp
flow resgo -> informed2
flow resmsp -> informed2
flow informed2 -> resolved
```
