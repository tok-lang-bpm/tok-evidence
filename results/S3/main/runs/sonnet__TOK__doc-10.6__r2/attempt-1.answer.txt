```tok
tok 1.0 "Dismissal handling"

pool org "Organisation"
  lane mspn "MSPN"
    start begin "Dismissal to send"
    task send "Send the dismissal to the MSPO"
  lane mspo "MSPO"
    task review "Review the dismissal"
    xor decision "Dismissal confirmed?"
    task confirm "Confirm the dismissal of the MSPN"
    task reject "Reject the dismissal of the MSPN"
    end confirmed "Dismissal confirmed"
    end rejected "Dismissal rejected"

flow begin -> send -> review -> decision
flow decision -> confirm "yes"
flow decision -> reject "no"
flow confirm -> confirmed
flow reject -> rejected
```
