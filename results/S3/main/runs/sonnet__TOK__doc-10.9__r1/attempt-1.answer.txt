```tok
tok 1.0 "Measuring point change"

pool org "Organisation"
  lane sp "SP / PU / GO"
    start requested "Change requested"
    task request "Request changes to the MPO"
  lane mpo "MPO"
    start caused "Change caused by the MPO"
    xor entry
    task review "Review the change request"
    xor decision "Request confirmed?"
    task reject "Reject the change of the measuring point"
    task confirm "Confirm the request"
    task perform "Perform the measuring point change"
    xor result "Change successful?"
    task report "Report the implementation to the SP / PU / GO"
    task notify "Notify the SP / PU / GO about the failure of the changes"
    end rejected "Change rejected"
    end done "Implementation reported"
    end failed "Failure notified"

flow requested -> request -> entry
flow caused -> entry
flow entry -> review -> decision
flow decision -> reject "no"
flow decision -> confirm "yes"
flow confirm -> perform -> result
flow result -> report "yes"
flow result -> notify "no"
flow reject -> rejected
flow report -> done
flow notify -> failed
```
