tok 1.0 "Court hearing file preparation"

pool court "Court"
  lane registry "Court registry"
    start morning "Morning of court hearing"
    task check "Check the files to be processed"
    xor missing "Files missing?"
    task search "Initiate a search for the missing files"
    task track "Physically track the files to the intended location"
    xor ready
    and split
    task handover "Hand the files to the Associate"
    task distribute "Distribute the Judge's Lawlist to the relevant people"
    and joined
    task hearings "Conduct the directions hearings"
    end done "Directions hearings conducted"

flow morning -> check -> missing
flow missing -> search "yes" when="filesMissing = true"
flow missing -> track "no" default
flow search -> ready
flow track -> ready
flow ready -> split
flow split -> handover
flow split -> distribute
flow handover -> joined
flow distribute -> joined
flow joined -> hearings -> done
