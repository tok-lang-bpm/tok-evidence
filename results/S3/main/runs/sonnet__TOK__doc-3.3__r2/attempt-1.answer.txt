tok 1.0 "Claim handling"

pool org "Insurance company"
  lane officer "Claims officer"
    start registered "Claim registered"
    task examine "Examine the claim"
    task recommend "Write a settlement recommendation"
  lane senior "Senior claims officer"
    task check "Check the recommendation"
    xor ok "Claim OK?"
    end proceed "Claim handling proceeds"

flow registered -> examine -> recommend -> check -> ok
flow ok -> proceed "OK" when="ok = true"
flow ok -> recommend "Not OK" default
