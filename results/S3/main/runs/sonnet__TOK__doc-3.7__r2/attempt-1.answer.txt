```tok
tok 1.0 "Car accident claim handling"

pool org "Insurance company"
  lane clerk "Claims clerk"
    start begin "Car accident claim received"
    task search "Search the Police Report in the Police Report database"
    task file "Put the Police Report in a file together with the Claim Documentation"
  lane handler "Claims handler"
    task estimate "Calculate the initial claim estimate"
    task plan "Create the Action Plan based on the Action Plan Checklist"
  lane manager "Claims manager"
    task negotiate "Negotiate a settlement on the claim estimate"
    task inform "Inform the claimant of the outcome"
    end done "Claim handled"

flow begin -> search -> file -> estimate -> plan -> negotiate -> inform -> done
```
