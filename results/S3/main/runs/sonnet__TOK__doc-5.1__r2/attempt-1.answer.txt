```tok
tok 1.0 "Loan approval process"

pool bank "Bank"
  lane system "Loan approval system"
    start requested "Loan request needed"
    task receive "Receive the customer request for a loan amount"
    xor decision "Which outcome?"
    task approve "Approve the loan"
    task deny "Deny the loan"
    xor joined
    task feedback "Send feedback to the customer"
    end done "Customer informed"
  lane assessor "Risk assessment Web service"
    service assess "Assess the request"
  lane approver "Approver Web service"
    service review "Review the request"

flow requested -> receive -> assess -> decision
flow decision -> approve "small loan and low risk" when="amount < 10000 and risk = \"low\""
flow decision -> deny "high risk" when="risk = \"high\""
flow decision -> review "further review needed or amount of $10,000 or more" default
flow approve -> joined
flow deny -> joined
flow review -> joined
flow joined -> feedback -> done
```
