```tok
tok 1.0 "Vacations Request"

pool org "Organization"
  lane employee "Employee"
    start needed "Vacation request needed"
    task submit "Submit the vacation request"
    task review "Review the rejection reasons"
    end reviewed "Request returned to employee"
  lane supervisor "Immediate supervisor"
    task receive "Receive the request"
    task decide "Approve or reject the request"
    xor approved "Request approved?"
  lane hr "Human Resources Representative"
    task manage "Complete the management procedures"
    end done "Vacation request handled"

flow needed -> submit -> receive -> decide -> approved
flow approved -> manage "yes"
flow approved -> review "no"
flow manage -> done
flow review -> reviewed
```
