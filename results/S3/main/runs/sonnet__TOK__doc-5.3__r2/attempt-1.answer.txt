```tok
tok 1.0 "Office Supply Request"

pool org "Organisation"
  lane employee "Employee"
    start needed "Office supplies needed"
    task submit "Submit the office supply request"
    task review "Review the comments for the change request"
  lane supervisor "Supervisor"
    task evaluate "Evaluate the request"
    xor decision "Decision?"
  lane purchase "Purchase department"
    sub quotations "Make quotations"
      start qstart "Quotations needed"
      task qrequest "Request quotations from vendors"
      task qcollect "Collect the quotations"
      end qend "Quotations made"
      flow qstart -> qrequest -> qcollect -> qend
    task select "Select a vendor"
    xor valid "Vendor valid in the system?"
  lane system "System"
    task register "Register the request"
    task notifyreject "Send the rejection notification"
    task notifychange "Send the change request notification"
    task notifyapprove "Send the approval notification"
    task order "Generate and send the purchase order"
    and wait
    task delivery "Wait for the product to be delivered"
    task invoice "Wait for the invoice to be received"
    and received
    end rejected "Request rejected"
    end completed "Order completed"

flow needed -> submit -> register -> evaluate -> decision
flow decision -> notifyreject "rejected"
flow decision -> notifychange "change required"
flow decision -> notifyapprove "approved"
flow notifyreject -> rejected
flow notifychange -> review
flow review -> submit
flow notifyapprove -> quotations -> select -> valid
flow valid -> select "no"
flow valid -> order "yes"
flow order -> wait
flow wait -> delivery
flow wait -> invoice
flow delivery -> received
flow invoice -> received
flow received -> completed
```
