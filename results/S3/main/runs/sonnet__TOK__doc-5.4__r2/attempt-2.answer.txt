tok 1.0 "Expense report handling"

pool company "Company"
  lane employee "Employee"
    start needed "Product or service required"
    task purchase "Purchase the product or service"
    task submit "Submit the expense report"
    task edit "Edit the expense report"
  lane supervisor "Supervisor"
    task review "Review the expense report"
    xor approved "Report approved?"
  lane treasurer "Treasurer"
    task check "Check the receipts against the items"
    xor order "All in order?"
    task accept "Accept the expenses for processing"
  lane payment "Payment system"
    service pay "Link to the payment system" type="payment"
    catch confirmed "Payment confirmation received" message="Payment confirmation"
    end done "Expense report processed"

flow needed -> purchase -> submit -> review -> approved
flow approved -> check "yes" when="approved = true"
flow approved -> edit "no" default
flow edit -> review
flow check -> order
flow order -> accept "yes" when="inOrder = true"
flow order -> edit "no" default
flow accept -> pay -> confirmed -> done
