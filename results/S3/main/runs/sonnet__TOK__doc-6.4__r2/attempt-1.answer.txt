```tok
tok 1.0 "Customer acquisition"

pool company "Company"
  lane marketing "Marketing"
    start begin "Contact details needed"
    task determine "Determine contact details of potential customers"
    xor how "Which way?"
    task buy "Buy details for cold calls"
    task exhibit "Participate in exhibitions"
    task know "Use personal contacts"
    xor found "Details determined"
  lane callcenter "Call center"
    task call "Call the customer"
    task contact "Determine the contact person and the budget"
    task ask "Ask whether the customer is interested"
    xor interested "Customer interested?"
    xor huge "Project budget huge?"
    end left "Customer left alone"
  lane head "Head of development"
    task acquire "Acquire the customer personally"
    end acquired "Customer approached by head of development"
  lane presales "Pre-sales"
    task presentpre "Give the technical online presentation"
  lane sales "Sales representative"
    xor technical "More technical presentation?"
    task presentsales "Give the online presentation"
    xor presented "Presentation given"
    eventgw wait "Customer reaction?"
    catch back "Customer comes back" message="Customer comes back"
    catch twoweeks "2 weeks without contact" timer=P14D
    task callagain "Call the customer again"
    xor reacted "Customer contacted"
    task quote "Create the quotation"
    end done "Quotation created"

flow begin -> determine -> how
flow how -> buy "buy details"
flow how -> exhibit "exhibition"
flow how -> know "personal contact"
flow buy -> found
flow exhibit -> found
flow know -> found
flow found -> call -> contact -> ask -> interested
flow interested -> technical "yes"
flow interested -> huge "no"
flow huge -> acquire "yes"
flow huge -> left "no"
flow acquire -> acquired
flow technical -> presentpre "yes"
flow technical -> presentsales "no"
flow presentpre -> presented
flow presentsales -> presented
flow presented -> wait
flow wait -> back
flow wait -> twoweeks
flow twoweeks -> callagain
flow back -> reacted
flow callagain -> reacted
flow reacted -> quote -> done
```
