```tok
tok 1.0 "Lead processing"

pool sales "Sales"
  lane manager "Manager"
    start begin "Leads to be handled"
    task check "Check the open leads"
    task select "Select the top five leads"
    task tell "Tell the Sales Assistant to call the contact persons"
    task process "Process the lead"
    end done "Lead processed"
  lane assistant "Sales Assistant"
    task call "Call the customer"
    xor interested "Is the customer interested?"
    task note "Send a note to the Manager"
    task next "Call the next customer"

flow begin -> check -> select -> tell -> call -> interested
flow interested -> note "yes" when="interested = true"
flow interested -> next "no" default
flow next -> interested
flow note -> process -> done
```
