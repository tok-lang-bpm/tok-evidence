```tok
tok 1.0 "Filling a post"

pool org "Organisation"
  lane dept "Functional department"
    start demand "Demand to fill a post received"
  lane hr "Human resources"
    task advertise "Advertise the post"
    task receive "Receive the applications"
    task check "Check the applications"
    task conclude "Conclude the contract of employment"
    end filled "Post filled"

flow demand -> advertise -> receive -> check -> conclude -> filled
```
