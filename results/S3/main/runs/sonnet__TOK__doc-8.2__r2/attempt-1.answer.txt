```tok
tok 1.0 "Job vacancy handling"

pool hr "HR"
  lane clerk "HR clerk"
    start reported "Vacancy reported"
    xor info_merge
    task create "Create the job description"
    xor clear "Information clear?"
    task ask "Ask the Department again"
    xor submit_merge
    task submit "Submit the job description for consideration"
    receive wait "Wait for the approval"
    xor approved "Description approved?"
    task correct "Correct the description"
    task post "Post the job"
    end posted "Job posted"

flow reported -> info_merge -> create -> clear
flow clear -> submit_merge "yes"
flow clear -> ask "no"
flow ask -> info_merge
flow submit_merge -> submit -> wait -> approved
flow approved -> post "yes"
flow approved -> correct "no"
flow correct -> submit_merge
flow post -> posted
```
