```tok
tok 1.0 "Committee meeting preparation and follow-up"

pool org "Organisation"
  lane coord "Coordination Unit"
    start begin "Dates finalized"
    task qa "Conduct a QA check on the minutes"
  lane support "Support Officer"
    task calendars "Update all group calendars"
    task folders "Create meeting folders for each meeting"
    task upload "Upload all appropriate documents to system"
    task advise "Advise Committee Members to read all related documents"
    task minutes "Produce minutes including Action Points"
    task send "Send the minutes to all Committee Members"
    task records "Update all departmental records"
    end done "Records updated"
  lane members "Committee Members"
    task meeting "Hold the meeting"

flow begin -> calendars -> folders -> upload -> advise -> meeting -> minutes -> qa -> send -> records -> done
```
