```tok
tok 1.0 "Expense report handling"

pool company "Company"
  lane employee "Employee"
    start received "Expense report received"
  lane finance "Finance system"
    xor hasaccount "Employee has an account?"
    task create "Create a new account"
    xor accountok
    task review "Review the expense report for automatic approval"
    xor amount "Amount under $200?"
    xor approvedjoin
    task reimburse "Reimburse to the direct deposit bank account"
    task reject "Send a rejection notice by email"
    task progress "Send an approval in progress email"
    task cancel "Send a cancellation notice by email"
    end paid "Reimbursement paid"
    end rejected "Rejection notice sent"
    end informed "Approval in progress email sent"
    end cancelled "Expense report cancelled, to be re-submitted"
  lane supervisor "Supervisor"
    task approve "Approve the expense report"
    xor decision "Approved?"
    boundary day7 "7 days passed" on=approve timer=P7D noninterrupting
    boundary day30 "30 days passed" on=approve timer=P30D

flow received -> hasaccount
flow hasaccount -> create "no"
flow hasaccount -> accountok "yes"
flow create -> accountok
flow accountok -> review -> amount
flow amount -> approvedjoin "yes"
flow amount -> approve "no"
flow approve -> decision
flow decision -> approvedjoin "yes"
flow decision -> reject "no"
flow approvedjoin -> reimburse -> paid
flow reject -> rejected
flow day7 -> progress -> informed
flow day30 -> cancel -> cancelled
```
