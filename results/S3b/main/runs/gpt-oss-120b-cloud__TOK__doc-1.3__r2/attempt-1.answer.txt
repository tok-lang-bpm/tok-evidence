  lane guest "Guest"
    start s1 "guest calls room service"
  lane manager "Room-service manager"
    task t1 "take down the order"
    task t2 "submit order ticket to kitchen"
    task t3 "give order to sommelier"
    task t4 "assign order to waiter"
  lane kitchen "Kitchen"
    task k1 "prepare food"
  lane sommelier "Sommelier"
    task s1 "fetch wine and prepare alcoholic beverages"
  lane waiter "Waiter"
    task w1 "ready cart and prepare nonalcoholic drinks"
    and pg1 "Parallel split"
    and pg2 "Parallel join"
    task w2 "deliver food, wine and cart to guest's room"
    task w3 "return to room-service station"
    xor g1 "another order pending?"
    task w4 "wait to do billing"
    task w5 "debit guest's account"
    end e1 "order completed"

flow s1 -> t1 -> t2 -> t3 -> t4 -> pg1
flow pg1 -> k1
flow pg1 -> s1
flow pg1 -> w1
flow k1 -> pg2
flow s1 -> pg2
flow w1 -> pg2
flow pg2 -> w2 -> w3 -> g1
flow g1 -> w4 "yes"
flow w4 -> w5

flow w5 -> e1
