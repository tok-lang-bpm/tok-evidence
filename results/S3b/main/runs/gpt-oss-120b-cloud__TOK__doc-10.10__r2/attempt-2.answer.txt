```tok
tok 1.0 "Fault handling process"

pool org "Organization"
  lane fault_detector "Fault Detector"
    start start_failure "Failure reported or suspected"
  lane mpo "MPO"
    task examine_failure "Examine the failure"
    xor confirm_failure "Failure confirmed?"
    task reject_failure "Reject the failure"
    and start_initial_inform
    task inform_GO_initial "Inform GO"
    task inform_MSP_initial "Inform MSP"
    and end_initial_inform
    task fix_fault "Fix the fault at the measuring device"
    task share_results "Share repair results with fault detector"
    and start_resolution_inform
    task inform_GO_resolution "Inform GO about resolution"
    task inform_MSP_resolution "Inform MSP about resolution"
    and end_resolution_inform
    end resolved "Resolution handled"
    end rejected "Failure rejected"

flow start_failure -> examine_failure
flow examine_failure -> confirm_failure
flow confirm_failure -> reject_failure "reject"
flow confirm_failure -> start_initial_inform "confirm"
flow reject_failure -> rejected
flow start_initial_inform -> inform_GO_initial
flow start_initial_inform -> inform_MSP_initial
flow inform_GO_initial -> end_initial_inform
flow inform_MSP_initial -> end_initial_inform
flow end_initial_inform -> fix_fault
flow fix_fault -> share_results
flow share_results -> start_resolution_inform
flow start_resolution_inform -> inform_GO_resolution
flow start_resolution_inform -> inform_MSP_resolution
flow inform_GO_resolution -> end_resolution_inform
flow inform_MSP_resolution -> end_resolution_inform
flow end_resolution_inform -> resolved
```
