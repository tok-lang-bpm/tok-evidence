```tok
tok 1.0 "Dismissal Process"

pool org "Organisation"
  lane MPON "MPON"
    start start "Start"
    task send "Send dismissal"
    flow start -> send

  lane MPOO "MPOO"
    task review "Review dismissal"
    xor decision "Oppose or confirm dismissal?"
    task oppose "Oppose dismissal"
    task confirm "Confirm dismissal"
    end opposed "Dismissal opposed"
    end confirmed "Dismissal confirmed"

    flow send -> review
    flow review -> decision
    flow decision -> oppose "oppose"
    flow decision -> confirm "confirm"
    flow oppose -> opposed
    flow confirm -> confirmed
```
