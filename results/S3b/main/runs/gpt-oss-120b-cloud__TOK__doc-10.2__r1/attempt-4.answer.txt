```tok
tok 1.0 "Meter Operation Process"

pool org "Organisation"
  lane MPON "MPON"
    start s_start "Process start"
    send s_report "Report the meter operation to the GO" message="Report"
    task t_equip_mpon "Perform equipment acquisition and changes"
    receive r_failure "Receive failure from MPON" message="Failure"
    receive r_success "Receive success from MPON" message="Success"
    end e_end_failure "Process completed with failure"
    end e_end_success "Process completed successfully"

  lane GO "GO"
    task t_examine "Examine the application of the MPON"
    xor g_app_decision "Application decision?"
    task t_reject "Reject the application of the MPON"
    end e_reject "Application rejected"
    task t_confirm_app "Confirm the application of the MPON"
    and g_inform_reg
    task t_inform_mpoo_reg "Inform MPOO about registration confirmation of the MPON"
    task t_inform_mspo_reg "Inform MSPO about registration confirmation of the MPON"
    and g_reg_done
    and g_equip_split
    and g_equip_join
    xor g_wait "Wait for MPON outcome or timeout"
    catch c_deadline "Deadline waiting" timer=PT48H
    task t_inform_mpon_deadline "Inform MPON about failure of the overall transaction by deadline"
    task t_confirm_failure "Confirm failure of the assignment to the MPON"
    and g_inform_failure
    task t_inform_mpoo_fail "Inform MPOO about failure of the assignment of the MPON"
    task t_inform_mspo_fail "Inform MSPO about failure of the assignment of the MPON"
    and g_failure_done
    task t_assign "Assign the MPON"
    task t_confirm_assign "Confirm the assignment to the MPON"
    and g_inform_success
    task t_inform_mpoo_assign "Inform MPOO about the assignment of the MPON"
    task t_inform_mspo_assign "Inform MSPO about the assignment of the MPON"
    task t_inform_sp_assign "Inform SP about the assignment of the MPON"
    and g_success_done

  lane MPOO "MPOO"
    task t_equip_mpoo "Perform equipment acquisition and changes"

  lane MSPO "MSPO"
    # no internal tasks

  lane SP "SP"
    # no internal tasks

# Sequence flows
flow s_start -> s_report -> t_examine
flow t_examine -> g_app_decision
flow g_app_decision -> t_reject "reject"
flow g_app_decision -> t_confirm_app "confirm"
flow t_reject -> e_reject
flow t_confirm_app -> g_inform_reg
flow g_inform_reg -> t_inform_mpoo_reg
flow g_inform_reg -> t_inform_mspo_reg
flow t_inform_mpoo_reg -> g_reg_done
flow t_inform_mspo_reg -> g_reg_done
flow g_reg_done -> g_equip_split
flow g_equip_split -> t_equip_mpon
flow g_equip_split -> t_equip_mpoo
flow t_equip_mpon -> g_equip_join
flow t_equip_mpoo -> g_equip_join
flow g_equip_join -> g_wait
flow g_wait -> r_success "success"
flow g_wait -> r_failure "failure"
flow g_wait -> c_deadline "timeout"
flow c_deadline -> t_inform_mpon_deadline -> t_confirm_failure
flow r_success -> t_assign
flow t_assign -> t_confirm_assign -> g_inform_success
flow g_inform_success -> t_inform_mpoo_assign
flow g_inform_success -> t_inform_mspo_assign
flow g_inform_success -> t_inform_sp_assign
flow t_inform_mpoo_assign -> g_success_done
flow t_inform_mspo_assign -> g_success_done
flow t_inform_sp_assign -> g_success_done
flow g_success_done -> e_end_success
flow r_failure -> t_confirm_failure
flow t_confirm_failure -> g_inform_failure
flow g_inform_failure -> t_inform_mpoo_fail
flow g_inform_failure -> t_inform_mspo_fail
flow t_inform_mpoo_fail -> g_failure_done
flow t_inform_mspo_fail -> g_failure_done
flow g_failure_done -> e_end_failure
```
