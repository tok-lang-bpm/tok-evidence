tok 1.0 "Equipment Change Process"

pool org "Organization"
  lane MPON "MPON"
    start s "Start"
    task n1 "Notify MPOO about equipment change intentions"
    xor g1 "Self dismounting decision?"
    xor gj1 "Self dismounting merge"
    xor g2 "Who performs final reading?"
    xor gj2 "Final reading merge"
    xor g3 "Who dismounts old equipment?"
    xor gj3 "Old equipment dismount merge"
    task f1 "Perform final reading"
    task d1 "Dismount old equipment"
    task m1 "Mount new device"
    task r1 "Read meter count from installed meter"
    task s1 "Send final reading values to GO"
    task t1 "Tell GO about device changes, master data and meter count at installation"
  lane MPOO "MPOO"
    task a1 "Announce self dismounting to MPON"
    task a2 "Notify MPON about no self-dismounting of MPOO"
    task f2 "Perform final reading"
    task d2 "Dismount old equipment"
  lane GO "GO"
    task notif "Notify MSP about device changes, master data, meter count at dismounting and installation"
    end e "End"
  lane MSP "MSP"
    # no activities, only recipient of notification

  flow s -> n1
  flow n1 -> g1
  flow g1 -> a1 "self dismounting"
  flow g1 -> a2 "no self dismounting"
  flow a1 -> gj1
  flow a2 -> gj1
  flow gj1 -> g2

  flow g2 -> f1 "MPON"
  flow g2 -> f2 "MPOO"
  flow f1 -> gj2
  flow f2 -> gj2
  flow gj2 -> g3

  flow g3 -> d1 "MPON"
  flow g3 -> d2 "MPOO"
  flow d1 -> gj3
  flow d2 -> gj3
  flow gj3 -> m1
  flow m1 -> r1
  flow r1 -> s1
  flow s1 -> t1
  flow t1 -> notif
  flow notif -> e
