tok 1.0 "Dismissal process"

pool org "Organisation"
  lane MSPN "MSPN"
    start start "Start"
    task send "Send dismissal"
  lane MSPO "MSPO"
    task review "Review dismissal"
    xor decision "Dismissal rejected?"
    task reject "Reject dismissal"
    end rejected "Dismissal rejected"
    task confirm "Confirm dismissal"
    end confirmed "Dismissal confirmed"

flow start -> send -> review -> decision
flow decision -> reject "Reject"
flow decision -> confirm "Confirm"
flow reject -> rejected
flow confirm -> confirmed
