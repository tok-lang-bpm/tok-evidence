```tok
tok 1.0 "Measuring point change process"

pool organisation "Organisation"
  lane requester "SP / PU / GO"
    start start "Change request initiated"
    xor initiator "Initiator"
    task request_change "Request change"
    end rejected "Change rejected"
  lane mpo "MPO"
    task cause_change "Cause change"
    task review "Review change request"
    xor approval "Approval?"
    task perform_change "Perform measuring point change"
    xor outcome "Implementation outcome?"
    task report_implementation "Report implementation to requester"
    task notify_failure "Notify failure to requester"
    end success "Change succeeded"
    end failure "Change failed"

flow start -> initiator
flow initiator -> request_change "SP/PU/GO request"
flow initiator -> cause_change "MPO causes"
flow request_change -> review
flow cause_change -> review
flow review -> approval
flow approval -> perform_change "confirm"
flow approval -> rejected "reject"
flow perform_change -> outcome
flow outcome -> report_implementation "success"
flow outcome -> notify_failure "failure"
flow report_implementation -> success
flow notify_failure -> failure
```
