```tok
tok 1.0 "Warrant Possession Process"

pool organization "Organization"
  lane party "Party"
    start send "Party sends warrant possession request"
  lane backoffice "Client Service Back Office (Small Claims Registry Operations)"
    task receive "Receive warrant possession request"
    task retrieve "Retrieve SCT file"
    task forward "Forward SCT Warrant Possession to Queensland Police"
    task storeawait "Store SCT physical file awaiting report"
    task receiveReport "Receive report from Queensland Police"
    task retrieveSCT "Retrieve respective SCT file"
    task attach "Attach new SCT document"
    task storeExpanded "Store expanded SCT physical file"
    end done "Process completed"
  lane police "Queensland Police"
    task receiveWarrant "Receive SCT Warrant Possession"
    task sendReport "Send report"

flow send -> receive
flow receive -> retrieve
flow retrieve -> forward
flow forward -> storeawait
flow storeawait -> receiveWarrant
flow receiveWarrant -> sendReport
flow sendReport -> receiveReport
flow receiveReport -> retrieveSCT
flow retrieveSCT -> attach
flow attach -> storeExpanded
flow storeExpanded -> done
```
