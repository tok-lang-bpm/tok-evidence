```tok
tok 1.0 "Morning file processing and hearings"

pool court "Court"
  lane clerk "Clerk"
    start start "Morning start"
    task check "Check unprocessed files"
    xor missing "Missing files?"
    task search "Search for missing files"
    task track "Track files to intended location"
    and parallel_split "Parallel split"
    task hand "Hand files to Associate"
    and parallel_join "Parallel join"
    end end "Process completed"
  lane system "System"
    task distribute "Distribute Judge's Lawlist to relevant people"
  lane judge "Judge"
    task hearings "Conduct directions hearings"

flow start -> check -> missing
flow missing -> search "yes"
flow missing -> track "no"
flow search -> parallel_split
flow track -> parallel_split
flow parallel_split -> hand
flow parallel_split -> distribute
flow hand -> parallel_join
flow distribute -> parallel_join
flow parallel_join -> hearings
flow hearings -> end
```
