```tok
tok 1.0 "Claims Notification Process"

pool organization "Organization"
  lane claimant "Claimant"
    start start1 "Claim received"
    task return_forms "Return forms"
    task return_updated_forms "Return updated forms"

  lane claims_dept "Claims Department"
    task check_insured "Check claimant insured"
    xor insured "Claimant insured?"
    task inform_rejection "Inform claimant claim must be rejected"
    task evaluate_severity "Evaluate claim severity"
    xor severity "Simple or complex claim?"
    task send_simple_forms "Send simple claim forms"
    task send_complex_forms "Send complex claim forms"
    xor merge_forms "Forms sent"
    task check_forms "Check forms for completeness"
    xor forms_complete "Forms complete?"
    task register_claim "Register claim in Claims Management system"
    task inform_update "Inform claimant to update forms"
    end end_rejected "Claim rejected"
    end end_registered "Claim registered"

    # Sequence flows
    flow start1 -> check_insured
    flow check_insured -> insured
    flow insured -> evaluate_severity  # insured = yes
    flow insured -> inform_rejection   # insured = no
    flow inform_rejection -> end_rejected

    flow evaluate_severity -> severity
    flow severity -> send_simple_forms   # outcome = simple
    flow severity -> send_complex_forms   # outcome = complex
    flow send_simple_forms -> merge_forms
    flow send_complex_forms -> merge_forms
    flow merge_forms -> return_forms

    flow return_forms -> check_forms
    flow check_forms -> forms_complete
    flow forms_complete -> register_claim   # outcome = yes
    flow register_claim -> end_registered

    flow forms_complete -> inform_update    # outcome = no
    flow inform_update -> return_updated_forms
    flow return_updated_forms -> check_forms
```
