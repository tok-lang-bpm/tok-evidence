```tok
tok 1.0 "Claims handling process"

pool org "Insurance Company"
  lane system_police "Police Report System"
    start start "Process start"
    task search "Search Police Report in database"
    task file "Create file with Police Report and Claim Documentation"
  lane handler "Claims Handler"
    task calculate "Calculate initial claim estimate"
    task create_plan "Create Action Plan based on Action Plan Checklist"
  lane manager "Claims Manager"
    task negotiate "Negotiate settlement on claim estimate"
  lane claimant "Claimant"
    task inform "Inform claimant of outcome"
    end end "Process end"

flow start -> search
flow search -> file
flow file -> calculate
flow calculate -> create_plan
flow create_plan -> negotiate
flow negotiate -> inform
flow inform -> end
```
