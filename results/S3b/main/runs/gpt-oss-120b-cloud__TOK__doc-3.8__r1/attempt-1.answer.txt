```tok
tok 1.0 "Claim handling process"

pool insurer "Car Insurer"
  lane customer "Customer"
    start start "Customer submits claim"
  lane notification "Notification Department"
    task check "Check documents"
    task register "Register claim"
  lane handling "Handling Department"
    task pick "Pick up claim"
    task insure "Check insurance"
    task assess "Perform assessment"
    xor decision "Assessment positive?"
    task phone "Phone garage to authorise repairs"
    task schedule "Schedule payment"
    task reject "Reject claim"
    xor join "Outcome processed"
    task letter "Send letter to customer"
    end end "Process complete"

flow start -> check -> register -> pick -> insure -> assess -> decision
flow decision -> phone "yes"
flow decision -> reject "no"
flow phone -> schedule -> join
flow reject -> join
flow join -> letter -> end
```
