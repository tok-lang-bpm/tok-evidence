tok 1.0 "Claim handling process"

pool insurer "Car Insurer"
  lane cust "Customer"
    start s1 "Process start"
    task t_submit "Submit claim"
  lane notif "Notification department"
    task t_checkDocs "Check documents"
    task t_registerClaim "Register claim"
  lane hand "Handling department"
    task t_pickup "Pick up claim"
    task t_checkInsurance "Check insurance"
    task t_assess "Perform assessment"
    xor d_assessment "Assessment positive?"
    task t_phoneGarage "Phone garage to authorise repairs"
    task t_schedulePayment "Schedule payment"
    task t_reject "Reject claim"
    task t_sendLetter "Send letter to customer"
    end e_done "Process complete"

flow s1 -> t_submit -> t_checkDocs
flow t_checkDocs -> t_registerClaim
flow t_registerClaim -> t_pickup
flow t_pickup -> t_checkInsurance
flow t_checkInsurance -> t_assess
flow t_assess -> d_assessment
flow d_assessment -> t_phoneGarage "yes"
flow d_assessment -> t_reject "no"
flow t_phoneGarage -> t_schedulePayment
flow t_schedulePayment -> t_sendLetter
flow t_reject -> t_sendLetter
flow t_sendLetter -> e_done
