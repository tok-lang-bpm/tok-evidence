```tok
tok 1.0 "Expense Reimbursement Process"

pool org "Organization"
  lane employee "Employee"
    start s "Start"
    task pu "Purchase product or service"
    task sub "Submit expense report"
    task edit "Edit expense report"
  lane supervisor "Supervisor"
    task rev "Review expense report"
    xor sup_dec "Supervisor approves?"
  lane treasurer "Treasurer"
    task chk "Check receipts"
    xor tre_dec "All receipts match?"
  lane system "Payment System"
    task link "Link to payment system"
    task wait "Wait for payment confirmation"
    end e "Process end"

flow s -> pu -> sub -> rev
flow rev -> sup_dec
flow sup_dec -> chk "yes"
flow sup_dec -> edit "no"
flow edit -> rev
flow chk -> tre_dec
flow tre_dec -> link "yes"
flow tre_dec -> edit "no"
flow link -> wait -> e
```
