```tok
tok 1.0 "Invoice Processing"

pool org "ACME AG" [executable]
  lane secretariat "Secretariat"
    start start_evt "Invoice received (paper or fax)"
    task receive_invoice "Receive invoice on paper or fax"
    task forward_invoice "Forward invoice after visual inspection"
  lane accounting "Accounting Employee"
    task identify_create "Identify suppliers and create invoice instance"
    task check_items "Check invoice items and note cost centers on docket"
    task send_internal_mail "Send docket and copy of invoice to internal mail"
    task clarify_vendor "Clarify with vendor and consult cost centre managers"
    task forward_to_commercial "Forward copy of invoice to commercial manager"
    task give_payment "Give payment instructions and close invoice"
  lane manager "Cost Centre Manager"
    task receive_copy "Receive copy of invoice and docket"
    xor accuracy_gate "Is position accurate?"
      task mark_ap "Mark position as accurate (AP) on docket"
      task reject_ap "Reject AP with note on docket"
    xor all_accurate_gate "All positions marked accurate?"
      task return_to_accounting_all "Return copy to accounting (all accurate)"
      task return_to_accounting_rej "Return copy to accounting (rejection)"
  lane commercial "Commercial Manager"
    task perform_audit "Perform commercial audit"
    xor amount_gate "Invoice amount > EUR 20?"
      task board_check "Board checks invoice"
    xor complaint_gate "Complaint during commercial audit?"
      task resolve_complaint "Resolve complaint with supplier"
  lane board "Board"
    task board_review "Review invoice"
  lane end_lane "End"
    end end_evt "Process completed"

  # Sequence flows
  flow start_evt -> receive_invoice -> forward_invoice -> identify_create -> check_items -> send_internal_mail -> receive_copy
  flow receive_copy -> accuracy_gate
  flow accuracy_gate -> mark_ap -> all_accurate_gate
  flow accuracy_gate -> reject_ap -> return_to_accounting_rej
  flow all_accurate_gate -> return_to_accounting_all
  flow all_accurate_gate -> receive_copy
  flow return_to_accounting_all -> forward_to_commercial
  flow return_to_accounting_rej -> clarify_vendor -> send_internal_mail -> receive_copy
  flow forward_to_commercial -> perform_audit -> amount_gate
  flow amount_gate -> board_check -> board_review -> complaint_gate
  flow amount_gate -> complaint_gate
  flow complaint_gate -> resolve_complaint -> perform_audit
  flow complaint_gate -> give_payment -> end_evt
```
