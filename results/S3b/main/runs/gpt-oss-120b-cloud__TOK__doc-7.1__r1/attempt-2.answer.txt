```tok
tok 1.0 "Lead handling process"

pool org "Organization"
  lane manager "Manager"
    start start "Start"
    task check "Check open leads"
    task select "Select top five leads"
    task tell "Tell Sales Assistant to call contacts"
    task process "Process lead"
    end processed "Lead processed"
  lane assistant "Sales Assistant"
    task call "Call customers"
    xor interested "Customer interested?"
    task send "Send note to Manager"

flow start -> check -> select -> tell -> call
flow call -> interested
flow interested -> send "yes" when="interested"
flow send -> process -> processed
flow interested -> call "no" default
```
