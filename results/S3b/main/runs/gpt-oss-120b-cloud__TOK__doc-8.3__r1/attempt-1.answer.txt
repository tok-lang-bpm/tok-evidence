```tok
tok 1.0 "Job description processing"

pool org "Organization"
  lane head "Head of functional department"
    start start "Process start"
    task detect "Detect personnel requirements"
    task report "Report vacancy to Personnel Department"
    task wait "Wait to receive job description"
    xor hrq "HR has questions?"
    task clarify "Provide clarifications to HR"
    xor corr "Corrections needed?"
    task ask_corr "Ask for corrections"
    task approve "Approve job description"
    end finish "Process end"

flow start -> detect -> report -> wait -> hrq
flow hrq -> clarify "yes"
flow hrq -> corr "no"
flow clarify -> corr
flow corr -> ask_corr "yes"
flow corr -> approve "no"
flow ask_corr -> wait
flow approve -> finish
```
