```tok
tok 1.0 "Vacancy handling process"

pool organization "Organization"
  lane head "Head of Functional Department"
    start S "Start"
    task d1 "Detect personnel requirements"
    task r1 "Report vacancy to Personnel Department"
    task w1 "Wait for job description"
    xor g1 "HR has questions?"
    task clar "Provide clarifications"
    xor g2 "Job description needs corrections?"
    task corr "Ask for corrections again"
    task appr "Approve job description"
    end e "Process completed"

  lane personnel "Personnel Department"
    task prep "Prepare job description"

  lane hr "HR Department"
    task hrq "Ask questions about tasks and requirements"

flow S -> d1
flow d1 -> r1
flow r1 -> w1
flow w1 -> g1

flow g1 -> hrq "yes" when="hasQuestions"
flow hrq -> clar
flow clar -> g2

flow g1 -> g2 "no"

flow g2 -> corr "yes" when="needsCorrections"
flow corr -> prep
flow prep -> w1

flow g2 -> appr "no"

flow appr -> e
```
