```tok
tok 1.0 "Annual meeting schedule drafting"

pool authority "Town Planning Authority"
  lane coord "Coordination Unit"
    start start "Start November scheduling"
    task draft "draft schedule of meetings for the next calendar year and add draft dates to all calendars"
    task recheck "recheck all dates and look for potential conflicts"
    task send "send final schedule of meeting dates to all the independent Committee Members by email"
    end done "Process completed"
  lane support "Support Officer"
    task check "check the dates and suggest modifications"
  lane committee "Committee Members"
    task advise "check diaries and advise the Coordination Unit of any conflicts"

flow start -> draft
flow draft -> check
flow check -> recheck
flow recheck -> send
flow send -> advise
flow advise -> done
```
