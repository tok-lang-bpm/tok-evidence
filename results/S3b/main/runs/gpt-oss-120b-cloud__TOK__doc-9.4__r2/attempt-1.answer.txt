```tok
tok 1.0 "Committee meeting process"

pool org "Organization"
  lane coordination "Coordination Unit"
  lane support "Support Officer"
  lane committee "Committee Members"

  start start "Dates finalized"

  task updateCalendars "Update group calendars, create meeting folders, upload documents"   # support
    on= support

  task adviseMembers "Advise Committee Members to read documents"   # support
    on= support

  task holdMeeting "Hold the meeting"   # committee
    on= committee

  task produceMinutes "Produce minutes with Action Points"   # support
    on= support

  task qaCheck "Conduct QA check on minutes"   # coordination
    on= coordination

  task sendMinutes "Send minutes to Committee Members"   # support
    on= support

  task updateRecords "Update departmental records"   # support
    on= support

  end end "Process completed"

flow start -> updateCalendars -> adviseMembers -> holdMeeting -> produceMinutes -> qaCheck -> sendMinutes -> updateRecords -> end
```
