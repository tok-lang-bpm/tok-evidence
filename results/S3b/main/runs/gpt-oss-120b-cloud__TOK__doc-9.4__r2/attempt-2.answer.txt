```tok
tok 1.0 "Committee meeting process"

pool org "Organization"
  lane coordination "Coordination Unit"
    start start "Dates finalized"
    task qaCheck "Conduct QA check on the minutes"
  lane support "Support Officer"
    task updateCalendars "Update group calendars and create meeting folders and upload documents"
    task adviseMembers "Advise Committee Members a week before each meeting to read all related documents"
    task produceMinutes "Produce minutes including any Action Points for each Committee Member"
    task sendMinutes "Send minutes to all Committee Members"
    task updateRecords "Update departmental records"
    end end "Process completed"
  lane committee "Committee Members"
    task holdMeeting "Hold the meeting"

flow start -> updateCalendars -> adviseMembers -> holdMeeting -> produceMinutes -> qaCheck -> sendMinutes -> updateRecords -> end
```
