```tok
tok 1.0 "Customized bicycle manufacturing"

pool company "Company"
  lane sales "Sales department"
    start received "Order received"
    task reject "Reject order"
    task accept "Accept order"
    and split_parallel "Start preparation"
    task ship "Ship bicycle"
    end rejected "Order rejected"
    end finished "Process finished"
    xor decide_order "Reject or accept order?"
  lane storehouse "Storehouse"
    task process_list "Process part list"
    task check_qty "Check required quantity"
    task reserve "Reserve part"
    task backorder "Back-order part"
    xor available "Part available?"
    xor more_parts "More parts?"
  lane engineering "Engineering department"
    task prepare "Prepare assembling"
    task assemble "Assemble bicycle"
    and join_parallel "Preparation finished"

flow received -> decide_order
flow decide_order -> reject "reject"
flow decide_order -> accept "accept"
flow reject -> rejected
flow accept -> split_parallel
flow split_parallel -> process_list
flow split_parallel -> prepare
flow process_list -> check_qty
flow check_qty -> available
flow available -> reserve "available"
flow available -> backorder "not available"
flow reserve -> more_parts
flow backorder -> more_parts
flow more_parts -> process_list "more parts"
flow more_parts -> join_parallel "all parts done" default
flow prepare -> join_parallel
flow join_parallel -> assemble
flow assemble -> ship
flow ship -> finished
```
