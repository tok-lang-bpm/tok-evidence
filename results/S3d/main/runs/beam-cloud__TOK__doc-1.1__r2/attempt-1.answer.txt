```tok
tok 1.0 "Customized bicycle manufacturing"

pool company "Company"
  lane sales "Sales department"
    start received "Order received"
    xor choice "Reject or accept?"
    task reject "Reject the order"
    task accept "Accept the order"
    and split
    task inform_store "Inform the storehouse"
    task inform_eng "Inform the engineering department"
    task ship "Ship the bicycle"
    end rejected "Order rejected"
    end completed "Order completed"
  lane storehouse "Storehouse"
    task process_list "Process the part list"
    task check_qty "Check the required quantity"
    xor avail "Part available in-house?"
    task reserve "Reserve the part"
    task backorder "Back-order the part"
    xor more "More items to process?"
  lane engineering "Engineering department"
    task prepare "Prepare everything for the assembling"
    task assemble "Assemble the bicycle"
    and join

flow received -> choice
flow choice -> reject "Reject"
flow choice -> accept "Accept"
flow reject -> rejected
flow accept -> split
flow split -> inform_store
flow split -> inform_eng
flow inform_store -> process_list
flow inform_eng -> prepare
flow process_list -> check_qty
flow check_qty -> avail
flow avail -> reserve "Yes"
flow avail -> backorder "No"
flow reserve -> more
flow backorder -> more
flow more -> check_qty "Yes"
flow more -> join "No"
flow prepare -> join
flow join -> assemble
flow assemble -> ship
flow ship -> completed
```
