```tok
tok 1.0 "Room service at The Evanstonian"

pool hotel "The Evanstonian"
  lane manager "Room-service manager"
    start call "Guest calls room service"
    task take "Take down the order"
    task submit "Submit an order ticket to the kitchen"
    task assign "Assign the order to the waiter"
    xor alcohol "Alcoholic beverage included?"
    task give "Give an order to the sommelier"
    and fork "Ready tasks"
    and join "All tasks complete"
    task deliver "Deliver the order to the guest's room"
    task debit "Debit the guest's account"
    end done "Order completed"
  lane kitchen "Kitchen"
    task prepare "Prepare the food"
  lane sommelier "Sommelier"
    task fetch "Fetch wine from the cellar and prepare alcoholic beverages"
  lane waiter "Waiter"
    task cart "Ready a cart"
    task drinks "Prepare nonalcoholic drinks"

flow call -> take -> submit -> assign -> fork
flow fork -> prepare
flow fork -> alcohol
flow fork -> cart
flow fork -> drinks
flow alcohol -> give "yes" when="alcoholic = true"
flow alcohol -> join "no" default
flow give -> fetch -> join
flow prepare -> join
flow cart -> join
flow drinks -> join
flow join -> deliver -> debit -> done
```
