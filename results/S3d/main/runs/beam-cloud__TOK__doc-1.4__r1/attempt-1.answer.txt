```tok
tok 1.0 "Company going public"

pool company "Company"
  lane company "Company"
    start decision_to_go_public "Decision to go public"
    task select_underwriters "Select underwriters"
  lane underwriters "Underwriters"
    task provide_advice "Provide procedural and financial advice"
    xor facts_fair "Facts presented fairly?"
    task give_seal "Give seal of approval"
    task prepare_registration "Prepare registration statement"
    and split_reg
    task register_issue "Register issue with SEC"
    task check_blue_sky "Check compliance with blue-sky laws"
    task firm_up_price "Firm up issue price"
    and split_price
    task arrange_roadshow "Arrange road show"
    task receive_clearance "Receive clearance from SEC"
    and join_reg
    task fix_price "Fix issue price"
    xor reason_not_to_offer "Any reason not to offer?"
    task firm_commitment "Enter firm commitment to buy stock"
    task offer_public "Offer to public"
    end offered "Issue offered to public"
    end not_offered "Issue not offered to public"
    end not_handled "Issue not handled"
  lane sec "Securities and Exchange Commission"
    task approve_registration "Approve registration statement"

flow decision_to_go_public -> select_underwriters
flow select_underwriters -> provide_advice
flow provide_advice -> facts_fair
flow facts_fair -> give_seal "yes"
flow facts_fair -> not_handled "no" default
flow give_seal -> prepare_registration
flow prepare_registration -> split_reg
flow split_reg -> register_issue
flow split_reg -> check_blue_sky
flow split_reg -> firm_up_price
flow register_issue -> approve_registration
flow approve_registration -> receive_clearance
flow receive_clearance -> join_reg
flow check_blue_sky -> join_reg
flow firm_up_price -> split_price
flow split_price -> arrange_roadshow
flow arrange_roadshow -> join_reg
flow join_reg -> fix_price
flow fix_price -> reason_not_to_offer
flow reason_not_to_offer -> firm_commitment "No" default
flow reason_not_to_offer -> not_offered "Yes"
flow firm_commitment -> offer_public
flow offer_public -> offered
```
