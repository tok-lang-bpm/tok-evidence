```tok
tok 1.0 "Fault handling"

pool organisation "Organisation"
  lane faultdetector "Fault detector"
    task report "Report the failure to the MPO"
  lane mpo "MPO"
    start occurred "Failure occurred"
    xor source "Failure source?"
    task suspect "Have a suspicion of own fault"
    task examine "Examine the failure"
    xor confirmed "Failure of fault detector confirmed?"
    task reject "Reject the failure of the fault detector"
    task confirm "Confirm the failure of the fault detector"
    task informGO "Inform the GO"
    task informMSP "Inform the MSP"
    task fix "Fix the fault at the measuring device"
    task share "Share the results of the repairs carried out"
    task informGORes "Inform the GO about the resolution of the interference"
    task informMSPRes "Inform the MSP about the resolution of the interference"
    end rejected "Failure rejected"
    end resolved "Interference resolved"

flow occurred -> source
flow source -> report "Fault detector reports a failure"
flow source -> suspect "MPO has a suspicion"
flow report -> examine
flow suspect -> examine
flow examine -> confirmed
flow confirmed -> reject "MPO rejects the failure"
flow confirmed -> confirm "MPO confirms the failure"
flow reject -> rejected
flow confirm -> informGO
flow informGO -> informMSP
flow informMSP -> fix
flow fix -> share
flow share -> informGORes
flow informGORes -> informMSPRes
flow informMSPRes -> resolved
```
