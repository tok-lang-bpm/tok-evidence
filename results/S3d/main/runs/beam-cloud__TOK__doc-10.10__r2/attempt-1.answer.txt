```tok
tok 1.0 "Fault handling"

pool organisation "Organisation"
  lane faultdetector "Fault Detector"
    start reported "Failure reported"
    task report "Report the failure"
  lane mpo "MPO"
    start suspected "Suspicion of own fault"
    xor source "Failure source?"
    task examine "Examine the failure"
    xor decision "Failure confirmed?"
    task reject "Reject the failure of the fault detector"
    and informInitial "Inform GO and MSP"
    task informGO "Inform the GO"
    task informMSP "Inform the MSP"
    and informedInitial "GO and MSP informed"
    task fix "Fix the fault at the measuring device"
    task share "Share the results of the repairs with the fault detector"
    task informGORes "Inform the GO about the resolution of the interference"
    task informMSPRes "Inform the MSP about the resolution of the interference"
    end rejected "Failure rejected"
    end resolved "Failure resolved"

flow reported -> report
flow suspected -> source
flow report -> source
flow source -> examine
flow examine -> decision
flow decision -> reject "Rejects"
flow decision -> informInitial "Confirms"
flow reject -> rejected
flow informInitial -> informGO
flow informInitial -> informMSP
flow informGO -> informedInitial
flow informMSP -> informedInitial
flow informedInitial -> fix
flow fix -> share
flow share -> informGORes
flow informGORes -> informMSPRes
flow informMSPRes -> resolved
```
