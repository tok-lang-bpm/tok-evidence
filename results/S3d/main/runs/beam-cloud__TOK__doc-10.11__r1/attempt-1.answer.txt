```tok
tok 1.0 "Measurements handling"

pool organisation "Organisation"
  lane go "GO"
    start s "Start"
    task request "Request the measurements"
    task process "Process the measured values"
    task sendChanged "Send the changed values to the MSP"
    task transmitReadings "Transmit the readings to the SP"
    end eHandled "Measurements handled"
  lane msp "MSP"
    task check "Check the received request"
    xor denyOrPerform "Deny or perform?"
    task deny "Deny the request of the GO"
    task perform "Perform the measurement"
    task inform "Inform the GO about the failure of the reading"
    task transmit "Transmit the measured values to the GO"
    end eDenied "Request denied"

flow s -> request -> check -> denyOrPerform
flow denyOrPerform -> deny "Deny" when="request denied"
flow denyOrPerform -> perform "Perform" default
flow deny -> inform -> eDenied
flow perform -> transmit -> process -> sendChanged -> transmitReadings -> eHandled
```
