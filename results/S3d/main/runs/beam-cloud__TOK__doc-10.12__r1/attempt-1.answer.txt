```tok
tok 1.0 "Master data change"

pool organisation "Organisation"
  lane ec "EC"
    start s "Start"
    task tell "Tell the INQ about the change of his master data"
  lane inq "INQ"
    task notify "Notify the IP of the change"
  lane ip "IP"
    task check "Check whether the master data can be changed at the desired time"
    xor can_change "Can be changed at desired time?"
    task confirm "Confirm the changes of the INQ"
    task reject "Reject the changes of the INQ"
    end done "Changes processed"

flow s -> tell -> notify -> check -> can_change
flow can_change -> confirm "yes"
flow can_change -> reject "no"
flow confirm -> done
flow reject -> done
```
