```tok
tok 1.0 "Transaction data request processing"

pool org "Organisation"
  lane inq "INQ"
    start start "Start"
    task transmit "Transmit the transaction data request"
  lane ip "IP"
    task check "Check the request"
    xor outcome "Outcome of the examination?"
    task transmit_data "Transmission of data"
    task reject "Rejection"
    xor answered "Answered?"
    end done "Question answered"

flow start -> transmit -> check -> outcome
flow outcome -> transmit_data "Transmission of data"
flow outcome -> reject "Rejection" default
flow transmit_data -> answered
flow reject -> answered
flow answered -> done
```
