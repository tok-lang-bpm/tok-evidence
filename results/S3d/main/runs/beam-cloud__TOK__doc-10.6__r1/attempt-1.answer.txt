```tok
tok 1.0 "MSPN dismissal"

pool organisation "Organisation"
  lane mspn "MSPN"
    start start "Start"
    task send_dismissal "Send a dismissal"
  lane mspo "MSPO"
    task review_dismissal "Review the dismissal"
    xor decision "Reject or confirm?"
    task reject_dismissal "Reject the dismissal"
    task confirm_dismissal "Confirm the dismissal"
    end rejected "Dismissal rejected"
    end confirmed "Dismissal confirmed"

flow start -> send_dismissal -> review_dismissal -> decision
flow decision -> reject_dismissal "Rejects"
flow decision -> confirm_dismissal "Confirms" default
flow reject_dismissal -> rejected
flow confirm_dismissal -> confirmed
```
