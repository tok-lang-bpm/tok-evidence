tok 1.0 "Dismissal handling"

pool organisation "Organisation"
  lane MSPN "MSPN"
    start start "Start"
    task sendDismissal "Send dismissal"
  lane MSPO "MSPO"
    task reviewDismissal "Review dismissal"
    xor outcome "Dismissal outcome?"
    task rejectDismissal "Reject dismissal"
    task confirmDismissal "Confirm dismissal"
    end done "Dismissal handled"

flow start -> sendDismissal -> reviewDismissal -> outcome
flow outcome -> rejectDismissal
flow outcome -> confirmDismissal
flow rejectDismissal -> done
flow confirmDismissal -> done
