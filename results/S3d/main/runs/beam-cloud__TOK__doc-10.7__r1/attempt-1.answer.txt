```tok
tok 1.0 "MSPN registration process"

pool organisation "Organisation"
  lane MSPN "MSPN"
    start start "Start"
    task register "Register the measurement at the GO"
  lane GO "GO"
    task examine "Examine the application of the MSPN"
    xor decision "Reject or confirm the application?"
    task reject "Reject the application of the MSPN"
    task confirm "Confirm the application of the MSPN"
    task assign "Assign the MSPN"
    task inform_mspo "Inform the MSPO about the assignment of the MSPN"
    task inform_mpo "Inform the MPO about the assignment of the MSPN"
    task inform_sp "Inform the SP about the assignment of the MSPN"
    end rejected "Application rejected"
    end completed "Application completed"

flow start -> register -> examine -> decision
flow decision -> reject "reject"
flow decision -> confirm "confirm"
flow reject -> rejected
flow confirm -> assign -> inform_mspo -> inform_mpo -> inform_sp -> completed
```
