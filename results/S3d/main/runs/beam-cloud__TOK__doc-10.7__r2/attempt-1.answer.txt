```tok
tok 1.0 "MSPN process"

pool organisation "Organisation"
  lane mspn "MSPN"
    start start "Start"
    task register "Register the measurement"
  lane go "GO"
    task examine "Examine the application"
    xor result "Reject or confirm?"
    task reject "Reject the application"
    task confirm "Confirm the application"
    task assign "Assign the MSPN"
    task inform_mspos "Inform the MSPO about the assignment of MSPN"
    task inform_mpo "Inform the MPO about the assignment of the MSPN"
    task inform_sp "Inform the SP about the assignment of MSPN"
    end rejected "Application rejected"
    end completed "Process completed"

flow start -> register -> examine -> result
flow result -> reject "Reject"
flow result -> confirm "Confirm"
flow reject -> rejected
flow confirm -> assign -> inform_mspos -> inform_mpo -> inform_sp -> completed
```
