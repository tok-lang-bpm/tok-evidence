```tok
tok 1.0 "Measuring point change"

pool organisation "Organisation"
  lane sp "SP / PU / GO"
    start start "Change arises"
    task request "Request changes"
  lane mpo "MPO"
    task cause "Cause a change"
    task review "Review the change request"
    xor source "Change source?"
    xor decision "Reject or confirm?"
    task reject "Reject the change of the measuring point"
    task confirm "Confirm the request of the SP / PU / GO"
    task perform "Perform the measuring point change"
    xor outcome "Implementation or failure?"
    task report "Report the implementation to the SP / PU / GO"
    task notify "Notify the SP / PU / GO about the failure"
    end rejected "Change rejected"
    end reported "Implementation reported"
    end notified "Failure notified"
  lane merge
    xor mergeSource

flow start -> source
flow source -> request "SP / PU / GO request"
flow source -> cause "MPO causes"
flow request -> mergeSource
flow cause -> mergeSource
flow mergeSource -> review
flow review -> decision
flow decision -> reject "Reject"
flow decision -> confirm "Confirm"
flow reject -> rejected
flow confirm -> perform
flow perform -> outcome
flow outcome -> report "Implementation"
flow outcome -> notify "Failure"
flow report -> reported
flow notify -> notified
```
