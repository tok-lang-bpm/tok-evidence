```tok
tok 1.0 "Warrant possession process"

pool organisation "Small Claims Registry Operations"
  lane party "Party"
    start requested "Warrant possession requested"
    task send_request "Send warrant possession request"
  lane back_office "Client Service Back Office"
    task receive_request "Receive the request"
    task retrieve_sct "Retrieve the SCT file"
    task forward_possession "Forward SCT Warrant Possession to Queensland Police"
    task store_physical "Store SCT physical file"
    task receive_report "Receive the report"
    task retrieve_respective "Retrieve the respective SCT file"
    task attach_document "Attach the new SCT document"
    task store_expanded "Store the expanded SCT physical file"
  lane police "Queensland Police"
    task send_report "Send report"
  lane other_staff "Other MC internal staff"
    task receive_physical "Receive the physical SCT file"
    end completed "Physical SCT file received"

flow requested -> send_request
flow send_request -> receive_request
flow receive_request -> retrieve_sct
flow retrieve_sct -> forward_possession
flow forward_possession -> store_physical
flow store_physical -> send_report
flow send_report -> receive_report
flow receive_report -> retrieve_respective
flow retrieve_respective -> attach_document
flow attach_document -> store_expanded
flow store_expanded -> receive_physical
flow receive_physical -> completed
```
