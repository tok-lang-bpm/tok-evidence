```tok
tok 1.0 "Court hearing preparation"

pool court "Court"
  lane staff "Court Staff"
    start morning "Files to be checked"
    task check "Check the files"
    xor missing "Some files missing?"
    task search "Initiate a search"
    task track "Physically track the files to the intended location"
    xor ready "All files ready"
    and split "Files ready"
    task hand "Hand the files to the Associate"
    task distribute "Distribute the Judge's Lawlist to the relevant people"
    and join "Ready"
  lane judge "Judge"
    task hearings "Conduct the directions hearings"
    end done "Directions hearings conducted"

flow morning -> check -> missing
flow missing -> search "yes" when="someFilesMissing"
flow missing -> track "no" default
flow search -> ready
flow track -> ready
flow ready -> split
flow split -> hand
flow split -> distribute
flow hand -> join
flow distribute -> join
flow join -> hearings
flow hearings -> done
```
