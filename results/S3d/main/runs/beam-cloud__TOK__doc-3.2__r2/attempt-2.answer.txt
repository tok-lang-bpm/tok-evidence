```tok
tok 1.0 "Court hearing file preparation"

pool court "Court"
  lane officer "Court Officer"
    start morning "Each morning" timer=P1D
    task check "Check files"
    xor missing "Files missing?"
    task search "Initiate search"
    task track "Physically track files"
    xor ready
    and split "All files ready?"
  lane associate "Associate"
    task hand "Hand files to Associate"
  lane judge "Judge"
    task distribute "Distribute Judge's Lawlist"
    and join
    task conduct "Conduct directions hearings"
    end done "Directions hearings conducted"

flow morning -> check -> missing
flow missing -> search "yes" when=""
flow missing -> track "no" default
flow search -> ready
flow track -> ready
flow ready -> split
flow split -> hand
flow split -> distribute
flow hand -> join
flow distribute -> join
flow join -> conduct -> done
```
