```tok
tok 1.0 "Mail handling"

pool organisation "Organisation"
  lane mail_processing_unit "Mail Processing Unit"
    start start "Start"
    task collect "Collect mail"
    task distribute "Distribute mail"
  lane mail_clerk "Mail Clerk"
    task sort_unopened "Sort unopened mail"
  lane registry "Registry"
    task open "Open mail"
    task sort_groups "Sort mail into groups"
    task register "Register mail in incoming Mail Register"
  lane assistant_registry_manager "Assistant Registry Manager"
    task quality "Perform quality check"
    xor compliant "Mail compliant?"
    task compile "Compile list of requisition"
    task send_back "Send back to party"
    task capture_matter "Capture matter details"
    task put_post "Put receipt and copied documents into envelope and post to party"
    end returned "Mail returned"
  lane cashier "Cashier"
    task take_fees "Take applicable fees"
    and split
    task capture_party "Capture Party Details"
    task print_file "Print Physical Court File"
    and join
    end handled "Mail handled"

flow start -> collect -> sort_unopened -> distribute -> open -> sort_groups -> register -> quality -> compliant
flow compliant -> compile "no"
flow compliant -> capture_matter "yes"
flow compile -> send_back -> returned
flow capture_matter -> take_fees -> split
flow split -> put_post
flow split -> capture_party
flow capture_party -> print_file
flow put_post -> join
flow print_file -> join
flow join -> handled
```
