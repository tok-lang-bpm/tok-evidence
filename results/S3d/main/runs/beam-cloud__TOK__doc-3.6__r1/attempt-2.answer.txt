```tok
tok 1.0 "Claims Notification"

pool organisation "Organisation"
  lane claimant "Claimant"
    task return_forms "Return forms"
    task update_forms "Update forms"
  lane org "Organisation"
    start received "Claim received"
    task check_insured "Check whether claimant is insured"
    xor insured "Claimant insured?"
    task inform_rejected "Inform claimant that claim must be rejected"
    task evaluate_severity "Evaluate severity of claim"
    xor severity "Simple or complex claim?"
    task send_forms "Send relevant forms to claimant"
    task check_forms "Check forms for completeness"
    xor complete "Forms complete?"
    task inform_update "Inform claimant to update forms"
    end rejected "Claim rejected"
  lane claims_system "Claims Management system"
    task register "Register claim in Claims Management system"
    end done "Claims Notification ended"

flow received -> check_insured
flow check_insured -> insured
flow insured -> inform_rejected "no"
flow insured -> evaluate_severity "yes"
flow inform_rejected -> rejected
flow evaluate_severity -> severity
flow severity -> send_forms "simple"
flow severity -> send_forms "complex"
flow send_forms -> return_forms
flow return_forms -> check_forms
flow check_forms -> complete
flow complete -> register "yes"
flow complete -> inform_update "no"
flow register -> done
flow inform_update -> update_forms
flow update_forms -> check_forms
```
