```tok
tok 1.0 "Claim handling"

pool insurer "Car insurer"
  lane customer "Customer"
    start s "Start"
    task submit "Submit the claim"
  lane notification "Notification department"
    task check_documents "Check the documents upon completeness"
    task register "Register the claim"
    task send_letter "Send a letter to the customer"
    end completed "Process complete"
  lane handling "Handling department"
    task pick_up "Pick up the claim"
    task check_insurance "Check the insurance"
    task assess "Perform the assessment"
    xor assessment "Assessment positive?"
    task phone_garage "Phone the garage to authorise the repairs"
    task schedule_payment "Schedule the payment"
    task reject "Reject the claim"
    xor outcome "Outcome merged"

flow s -> submit -> check_documents -> register -> pick_up -> check_insurance -> assess -> assessment
flow assessment -> phone_garage "Yes" when="positive"
flow assessment -> reject "No" default
flow phone_garage -> schedule_payment -> outcome
flow reject -> outcome
flow outcome -> send_letter -> completed
```
