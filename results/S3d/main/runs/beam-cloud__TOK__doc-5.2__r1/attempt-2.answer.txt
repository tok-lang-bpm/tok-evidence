```tok
tok 1.0 "Vacations Request"

pool organisation "Organisation"
  lane employee "Employee"
    start submitted "Vacation request submitted"
    task submit "Submit vacation request"
    task review "Review rejection reasons"
    end rejected "Vacation request rejected"
  lane system "System"
    task register "Register requirement"
    task notify "Generate notification"
  lane supervisor "Immediate Supervisor"
    task receive "Receive vacation request"
    xor approve "Vacation request approved?"
  lane hr "Human Resources Representative"
    task procedures "Complete management procedures"
    end completed "Vacation request completed"

flow submitted -> submit -> register -> receive -> approve
flow approve -> notify "approved" when="approved"
flow approve -> review "rejected" default
flow review -> rejected
flow notify -> procedures -> completed
```
