```tok
tok 1.0 "Lead handling"

pool organisation "Organisation"
  lane manager "Manager"
    start s "Start"
    task check "Check the open leads"
    task select "Select the top five ones"
    task tell "Tell Sales Assistant to call the contact person of the leads"
    task process "Process the lead"
    end done "Lead processed"
  lane sales "Sales Assistant"
    task call "Call each customer"
    xor interested "Someone is interested?"
    task sendnote "Send a note to the Manager"
    task callnext "Call the next customer"
    end finished "No interest"

flow s -> check -> select -> tell -> call
flow call -> interested
flow interested -> sendnote "yes"
flow interested -> callnext "no" default
flow sendnote -> process
flow process -> done
flow callnext -> finished
```
