```tok
tok 1.0 "Fill a post"

pool organisation "Organisation"
  lane functional "Functional department"
    start demand "Demand to fill a post"
    task advertise "Advertise the post"
    task check "Check the applications"
    task fill "Fill the post"
    end filled "Post filled"
  lane applicants "Applicants"
    task apply "Apply for the post"

flow demand -> advertise -> apply -> check -> fill -> filled
```
