```tok
tok 1.0 "Job description handling"

pool organisation "Organisation"
  lane clerk "HR clerk"
    start reported "Vacancy reported" message="Vacancy reported"
    task create "Create job description"
    xor confusion "Confusion in message?"
    task ask "Ask the Department"
    task submit "Submit job description"
    task wait "Wait for approval"
    xor approved "Approved?"
    task correct "Correct job description"
    task post "Post job"
    end posted "Job posted"
  lane department "Department"
    task reject "Reject and request correction"

flow reported -> create -> confusion
flow confusion -> ask "yes" when="confusion"
flow confusion -> submit "no" default
flow ask -> create
flow submit -> wait -> approved
flow approved -> post "yes" when="approved"
flow approved -> reject "no" default
flow reject -> correct -> submit
flow post -> posted
```
