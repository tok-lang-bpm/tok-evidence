```tok
tok 1.0 "Personnel requirements and job description"

pool organisation "Organisation"
  lane head "Head of functional department"
    start detected "Personnel requirements detected"
    task report "Report vacancy"
    xor qsplit "Questions from HR?"
    task clarify "Clarify tasks and requirements"
    xor qjoin "Ready to describe"
    task wait "Wait for job description"
    task review "Review job description"
    xor csplit "Corrections needed?"
    task ask "Ask for corrections"
    task approve "Approve job description"
    end approved "Job description approved"
  lane hr "Personnel Department"
    task ask_questions "Ask about tasks and requirements"
    task describe "Describe the job"

flow detected -> report
flow report -> qsplit
flow qsplit -> ask_questions "Yes" when="hrHasQuestions"
flow qsplit -> qjoin "No" default
flow ask_questions -> clarify
flow clarify -> qjoin
flow qjoin -> wait
flow wait -> review
flow describe -> wait
flow review -> csplit
flow csplit -> ask "Yes" when="correctionsNeeded"
flow csplit -> approve "No" default
flow ask -> describe
flow approve -> approved
```
