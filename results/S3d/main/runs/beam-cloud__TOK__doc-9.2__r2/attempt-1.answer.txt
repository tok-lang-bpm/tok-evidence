```tok
tok 1.0 "Mortgage offer handling"

pool organisation "Organisation"
  lane csr "Customer Service Representative"
    start start "Start"
    task send_offer "Send mortgage offer to customer"
    task wait_reply "Wait for reply"
    xor reply "Reply received?"
    task update_decline "Update case details"
    task archive_decline "Archive work"
    end cancel_decline "Case cancelled"
    xor prereq "All prerequisite documents provided?"
    task generate_msg "Generate message to customer requesting outstanding documents"
    end requested "Outstanding documents requested"
    task update_timeout "Update case details"
    task archive_timeout "Archive work"
    end cancel_timeout "Case cancelled"
  lane customer "Customer"
    task decline "Call or write back declining the mortgage"
    task send_docs "Send back completed offer documents"
    task attach_docs "Attach prerequisite documents"
  lane administration "Administration"
    task move_admin "Move case to administration for completion"
    end completed "Case moved to administration for completion"

flow start -> send_offer -> wait_reply -> reply
flow reply -> decline "Customer declines"
flow decline -> update_decline -> archive_decline -> cancel_decline
flow reply -> send_docs "Completed offer documents"
flow send_docs -> attach_docs -> prereq
flow prereq -> move_admin "Yes"
flow move_admin -> completed
flow prereq -> generate_msg "No"
flow generate_msg -> requested
flow reply -> update_timeout "No answer after 2 weeks"
flow update_timeout -> archive_timeout -> cancel_timeout
```
