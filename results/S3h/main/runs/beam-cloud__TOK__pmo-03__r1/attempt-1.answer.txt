```tok
tok 1.0 "Inventory restocking"

pool organisation "Organisation"
  lane warehouse "Warehouse or store"
    start s "Start"
    task monitor "Monitor inventory levels"
    xor signal "Signal need to reorder?"
    task manual "Perform manual check"
    task receive "Receive stock"
    task inspect "Inspect stock for quality"
    task shelves "Place stock on shelves"
    task storeStorage "Place stock in storage"
    xor mergeSignal
    xor storage "Storage location?"
    xor mergeStorage
  lane procurement "Procurement team"
    task order "Place order with suppliers"
  lane inventory "Inventory system"
    task auto "Trigger automated alert"
    task updateDates "Update system with expected delivery dates"
    task record "Record stock in system"
    task updateLevels "Update inventory levels"
    end e "Inventory levels updated"

flow s -> monitor
flow monitor -> signal
flow signal -> manual "manual check"
flow signal -> auto "automated alert"
flow manual -> mergeSignal
flow auto -> mergeSignal
flow mergeSignal -> order
flow order -> updateDates
flow updateDates -> receive
flow receive -> inspect
flow inspect -> record
flow record -> storage
flow storage -> shelves "on shelves"
flow storage -> storeStorage "in storage"
flow shelves -> mergeStorage
flow storeStorage -> mergeStorage
flow mergeStorage -> updateLevels
flow updateLevels -> e
```
