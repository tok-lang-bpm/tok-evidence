```tok
tok 1.0 "Order fulfilment"

pool org "Organisation"
  lane customer "Customer"
    start order_placed "Order placed"
    task place_order "Place order"
  lane system "System"
    task generate_confirmation "Generate order confirmation"
    task send_confirmation "Send order confirmation"
    task generate_label "Generate shipping label"
    task send_tracking "Send tracking information"
  lane warehouse "Warehouse team"
    and prepare "Prepare order"
    task pick_items "Pick items"
    task pack_items "Pack items"
    and prepared "Order prepared"
    task hand_over "Hand over order to logistics provider"
  lane logistics "Logistics provider"
    task monitor_shipment "Monitor shipment"
    task deliver_order "Deliver order"
    end order_delivered "Order delivered"

flow order_placed -> place_order -> generate_confirmation -> send_confirmation -> prepare
flow prepare -> pick_items
flow prepare -> generate_label
flow pick_items -> pack_items
flow pack_items -> prepared
flow generate_label -> prepared
flow prepared -> hand_over -> send_tracking -> monitor_shipment -> deliver_order -> order_delivered
```
