```tok
tok 1.0 "Incident management"

pool org "Organisation"
  lane reporter "Employee or Customer"
    start start_event "Start"
    task report_incident "Report the incident"
  lane tracking "Tracking System"
    task log_report "Log the report"
    task assign_team "Assign to the team"
    task close_report "Close the incident report"
    task notify_stakeholders "Notify stakeholders"
    end done "Incident report closed"
  lane team "Investigation Team"
    task gather_info "Gather necessary information"
    task identify_cause "Identify the cause"
    task propose_actions "Propose corrective actions"
    task implement_solution "Implement the solution"
    task follow_up "Follow up"

flow start_event -> report_incident -> log_report -> assign_team -> gather_info -> identify_cause -> propose_actions -> implement_solution -> follow_up -> close_report -> notify_stakeholders -> done
```
