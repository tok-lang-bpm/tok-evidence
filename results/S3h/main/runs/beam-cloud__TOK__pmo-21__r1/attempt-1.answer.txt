```tok
tok 1.0 "Shipment process"

pool organisation "Organisation"
  lane secretary "Secretary"
    start goods "Goods shall be shipped"
    task clarify "Clarify who will do the shipping"
    xor large "Large amounts?"
    task invite "Invite three logistic companies to make offers"
    task select "Select one of them"
    task post "Use normal post shipment"
    xor method "Shipping method determined"
    and startPrep "Start preparation"
    task writeLabel "Write package label"
    xor insurance "Parcel insurance necessary?"
    and ready "Everything ready?"
  lane logisticsHead "Logistics department head"
    task takeInsurance "Take parcel insurance"
  lane warehousemen "Warehousemen"
    task package "Package goods"
    task prepare "Prepare packaged goods for being picked up"
    end prepared "Packaged goods prepared for pickup"

flow goods -> clarify -> large
flow large -> invite "large amounts"
flow large -> post "small amounts" default
flow invite -> select
flow select -> method
flow post -> method
flow method -> startPrep
flow startPrep -> writeLabel
flow startPrep -> insurance
flow startPrep -> package
flow insurance -> takeInsurance "necessary"
flow insurance -> ready "not necessary" default
flow takeInsurance -> ready
flow writeLabel -> ready
flow package -> ready
flow ready -> prepare -> prepared
```
