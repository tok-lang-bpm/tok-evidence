```tok
tok 1.0 "Restaurant service"

pool restaurant "Restaurant"
  lane guest "Guest"
    start hungry "Feeling hungry"
    task enter "Enter the restaurant"
    task choose "Choose a dish from the changing meal range"
    task wait_turn "Wait until it is his turn"
    task place_order "Place his order with the employee"
    task pick_up "Pick up his meal"
    task eat "Eat his meal"
    end done "Meal finished"
  lane employee "Employee"
    task enter_order "Enter the order into the POS system"
    task collect_money "Collect the money from the guest"
    task setup_buzzer "Set up a buzzer"
    task pass_buzzer "Pass the buzzer to the guest"
    task inform_chef "Inform the chef of the new meal order"
    task set_off_buzzer "Set off the guest's buzzer"
    xor reacts "Guest reacts to the buzzer?"
    task hand_over "Hand over the meal"
    catch wait_five "5 minutes" timer=PT5M
    task call_guest "Call for the guest"
  lane chef "Chef"
    task prepare "Prepare the meal"
    task place_hatch "Place the meal in the service hatch"
    task inform_ready "Inform the employee that the meal is in the service hatch"

  flow hungry -> enter -> choose -> wait_turn -> place_order
  flow place_order -> enter_order -> collect_money -> setup_buzzer -> pass_buzzer -> inform_chef
  flow inform_chef -> prepare -> place_hatch -> inform_ready -> set_off_buzzer -> reacts
  flow reacts -> hand_over "Reacted"
  flow reacts -> wait_five "No reaction" default
  flow wait_five -> call_guest
  flow call_guest -> reacts
  flow hand_over -> pick_up -> eat -> done
```
