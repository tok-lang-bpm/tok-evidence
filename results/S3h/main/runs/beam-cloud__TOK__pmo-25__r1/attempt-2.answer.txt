```tok
tok 1.0 "Find a Job"

pool jobmarket "Job market"
  lane applicant "Applicant"
    start start "Start job search"
    task report "Report job applications"
    task negotiate "Negotiate job interview"
    task enter "Enter probation phase"
    task ratecompany "Rate company"
    task receive "Receive job offers"
    and rates "Ratings"
    and ratedone "Ratings done"
    xor want "Company wants you?"
    xor permanent "Job becomes permanent?"
    xor rating "Rated C or less?"
    end found "Job found"
    end receiving "Continue receiving job offers"
  lane company "Company"
    task send "Send potential job offers"
    task confirm "Confirm receipt of job applications"
    task rateapp "Rate application"
    task rateapplicant "Rate applicant"

flow start -> report -> send -> confirm -> rateapp -> negotiate -> want
flow want -> enter "yes"
flow want -> report "no"
flow enter -> rates
flow rates -> ratecompany
flow rates -> rateapplicant
flow ratecompany -> ratedone
flow rateapplicant -> ratedone
flow ratedone -> permanent
flow permanent -> report "no"
flow permanent -> rating "yes"
flow rating -> receive "C or less"
flow rating -> found "better"
flow receive -> receiving
```
