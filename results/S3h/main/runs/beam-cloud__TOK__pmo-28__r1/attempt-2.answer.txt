tok 1.0 "Internship"

pool internship "Internship"
  lane you "You"
    start s "Start"
    task enter_topic "Enter topic and money"
    task enter_experience "Enter experience and hobbies"
    task receive_offer "Receive offer"
    xor accept_offer "Accept offer?"
    task accept "Accept offer"
    task deny "Deny offer"
    task invalidate "Invalidate other offers"
    and update_start "Start status updates"
    task write_updates "Write status updates" loop sequential loop-max=3
    and update_join "Status updates finished"
    task recommend "Recommend company to friends"
    task tweet "Send tweet" parallel
    end e "End"
  lane company "Company"
    task write_company_updates "Write status updates" loop sequential loop-max=3

flow s -> enter_topic -> enter_experience -> receive_offer -> accept_offer
flow accept_offer -> accept "accept" when="accepted"
flow accept_offer -> deny "deny" default
flow deny -> receive_offer
flow accept -> invalidate -> update_start
flow update_start -> write_updates
flow update_start -> write_company_updates
flow write_updates -> update_join
flow write_company_updates -> update_join
flow update_join -> recommend -> tweet -> e
