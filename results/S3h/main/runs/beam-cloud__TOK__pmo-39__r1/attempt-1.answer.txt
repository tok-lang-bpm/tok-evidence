```tok
tok 1.0 "DIY Repair of a Broken Smartphone Screen"

pool repair "DIY Repair of a Broken Smartphone Screen"
  lane you "You"
    start begin "Start repair"
    task enter "Enter the model and what is broken"
    and split "Split order sources"
    task contact "Contact friends"
    task buy "Buy the rest"
    and merge "Merge order sources"
    task receive "Receive instructions how to fix your phone"
    task fix "Fix your phone"
    xor worked "Did it work?"
    task review "Send a review and/or video"
    task expert "Send it to an expert"
    end success "Repair reviewed"
    end failed "Sent to expert"
  lane tool "Online Tool"
    task provide "Give the list of materials, the list of tools, and ordering choices"

flow begin -> enter -> provide -> split
flow split -> contact
flow split -> buy
flow contact -> merge
flow buy -> merge
flow merge -> receive -> fix -> worked
flow worked -> review "yes" when="worked"
flow worked -> expert "no" default
flow review -> success
flow expert -> failed
```
