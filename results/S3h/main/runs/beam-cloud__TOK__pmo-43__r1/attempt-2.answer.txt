```tok
tok 1.0 "Car Service"

pool car_service "Car Service"
  lane system "System"
    start service_needed "Service necessary"
    task check_registration "Check if your car is still registered"
    xor registered "Car still registered?"
    task notify "Notify you"
    catch wait30 "30 days" timer=P30D
    task fine "Fine you"
    end not_registered "Car not registered"
    end fined "You are fined"
  lane mechanic "Mechanic"
    task enter_problems "Enter the problems of your car for reference"
    task enter_success "Enter that the repair was successful and you got your Pickerl"
    task enter_next "Enter the time for your next service"
  lane customer "Customer"
    xor attend "Go to the service?"
    task get_updates "Get status updates via email"
    task pay "Pay through the app"
    end completed "Service completed"

flow service_needed -> check_registration -> registered
flow registered -> notify "yes" when="registered"
flow registered -> not_registered "no" default
flow notify -> attend
flow attend -> enter_problems "yes" when="go_to_service"
flow attend -> wait30 "no" default
flow wait30 -> fine
flow fine -> fined
flow enter_problems -> get_updates
flow get_updates -> pay
flow pay -> enter_success
flow enter_success -> enter_next
flow enter_next -> completed
```
