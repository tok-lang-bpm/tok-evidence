```tok
tok 1.0 "Ultrasound-guided venous catheter insertion"

pool org "Organisation"
  lane doctor "Doctor"
    start s "Start"
    task prepare "Prepare implements"
    task wash "Wash hands"
    task sterile_clothes "Get in sterile clothes"
    task clean "Clean puncture area"
    task drape "Drape puncture zone"
    task config "Configure ultrasound"
    task gel_probe "Put gel in probe"
    task cover_probe "Cover probe"
    task sterile_gel "Put sterile gel"
    task pos_probe "Position probe"
    task pos_patient "Position patient"
    xor ident "Identification method?"
    task id_anat "Identify vein by anatomic identification"
    task id_dopp "Identify vein by doppler identification"
    task id_comp "Identify vein by compression identification"
    xor join_ident
    task anest "Anesthetize patient"
    task puncture "Puncture patient"
    task check_blood "Check blood return"
    xor blood "Blood return correct?"
    task drop_probe "Drop probe"
    task remove_syringe "Remove syringe"
    task install_wire "Install guidewire"
    task remove_trocar "Remove trocar"
    xor wire_check "Wire check method?"
    task check_long "Check wire in long axis"
    task check_short "Check wire in short axis"
    xor join_wire
    task check_wirepos "Check wire position"
    xor wirepos "Wire in good position?"
    task widen "Widen pathway"
    task advance "Advance catheter"
    task remove_wire "Remove guidewire"
    task verify_flow "Verify flow"
    task verify_reflow "Verify reflow"
    task check_position "Check catheter position"
    end done "End"

flow s -> prepare -> wash -> sterile_clothes -> clean -> drape -> config -> gel_probe -> cover_probe -> sterile_gel -> pos_probe -> pos_patient -> ident
flow ident -> id_anat "anatomic"
flow ident -> id_dopp "doppler"
flow ident -> id_comp "compression" default
flow id_anat -> join_ident
flow id_dopp -> join_ident
flow id_comp -> join_ident
flow join_ident -> anest -> puncture -> check_blood -> blood
flow blood -> puncture "not correct" default
flow blood -> drop_probe "correct"
flow drop_probe -> remove_syringe -> install_wire -> remove_trocar -> wire_check
flow wire_check -> check_long "long axis"
flow wire_check -> check_short "short axis" default
flow check_long -> join_wire
flow check_short -> join_wire
flow join_wire -> check_wirepos -> wirepos
flow wirepos -> puncture "not in good position" default
flow wirepos -> widen "in good position"
flow widen -> advance -> remove_wire -> verify_flow -> verify_reflow -> check_position -> done
```
