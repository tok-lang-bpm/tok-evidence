```tok
tok 1.0 "Sales inquiry process"

pool org "Organization"
  lane customer "Customer"
    start inquiry "Customer inquiry"

  lane sales "Sales"
    task collect "Collect information and address concerns"
    xor interested "Customer interested?"
    task guide "Guide product selection"
    task quote "Provide quote"
    xor approved "Quote approved by customer?"
    task place "Place order"
    task record "Record order in system"
    task confirm "Send order confirmation"
    end done "Order placed and confirmed"
    end not_interested "Process ended – not interested"
    end not_approved "Process ended – quote not approved"

flow inquiry -> collect
flow collect -> interested
flow interested -> guide "yes"
flow interested -> not_interested "no"
flow guide -> quote
flow quote -> approved
flow approved -> place "yes"
flow approved -> not_approved "no"
flow place -> record
flow record -> confirm
flow confirm -> done
```
