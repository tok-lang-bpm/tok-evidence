tok 1.0 "Hiring Process"

pool org "Organization"
  lane department "Department"
    start start "Process start"
    task identify_need "Identify need for new hire"
    end end_terminate "Process terminated"
  lane hr "HR"
    task create_jd "Create job description"
    task post_job "Post job on job boards"
    task collect_resumes "Collect resumes"
    task screen_resumes "Screen resumes"
    task phone_interviews "Conduct initial phone interviews"
    xor interview_type "Interview type?"
      task invite_inperson "Invite candidate for in-person interview"
      task invite_virtual "Invite candidate for virtual interview"
    task conduct_interview "Conduct interview"
    xor candidate_chosen "Candidate chosen?"
      task extend_offer "Extend offer"
    xor salary_negotiation "Salary negotiations?"
      task negotiate_salary "Negotiate salary"
    task offer_accepted "Offer accepted"
    and onboarding_split
    task paperwork "Complete paperwork"
    task orientation "Conduct orientation"
    task training "Provide training"
    and onboarding_join
    task integrate "Integrate new hire into team"
    end end_complete "Hiring process completed"
  lane newhire "New Hire"
    # No activities defined for the new hire lane

flow start -> identify_need -> create_jd -> post_job -> collect_resumes -> screen_resumes -> phone_interviews -> interview_type
flow interview_type -> invite_inperson
flow interview_type -> invite_virtual
flow invite_inperson -> conduct_interview
flow invite_virtual -> conduct_interview
flow conduct_interview -> candidate_chosen
flow candidate_chosen -> extend_offer
flow candidate_chosen -> end_terminate
flow extend_offer -> salary_negotiation
flow salary_negotiation -> negotiate_salary
flow salary_negotiation -> offer_accepted
flow negotiate_salary -> offer_accepted
flow offer_accepted -> onboarding_split
flow onboarding_split -> paperwork
flow onboarding_split -> orientation
flow onboarding_split -> training
flow paperwork -> onboarding_join
flow orientation -> onboarding_join
flow training -> onboarding_join
flow onboarding_join -> integrate -> end_complete
