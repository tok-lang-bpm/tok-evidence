```tok
tok 1.0 "Inventory Reorder Process"

pool org "Organization"
  lane warehouse "Warehouse"
    start S "Start"
    task mon "Monitor inventory levels"
    xor thresh "Stock reaches threshold?"
    xor alert "Alert or manual check?"
    task auto "Generate automated alert"
    task manual "Perform manual check"
    xor join_sig "Merge alert paths"
    task signal "Signal need to reorder"
    task receive "Receive stock"
    task inspect "Inspect stock for quality"
    task record "Record stock in system"
    task store "Place stock on shelves or in storage"
    end E "Inventory levels updated after restock"
  lane procurement "Procurement Team"
    task place "Place order with suppliers"
    task update "Update inventory system with expected delivery dates"

flow S -> mon
flow mon -> thresh
flow thresh -> alert
flow alert -> auto
flow alert -> manual
flow auto -> join_sig
flow manual -> join_sig
flow join_sig -> signal
flow signal -> place
flow place -> update
flow update -> receive
flow receive -> inspect
flow inspect -> record
flow record -> store
flow store -> E
```
