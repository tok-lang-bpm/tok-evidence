```tok
tok 1.0 "Marketing Campaign Process"

pool org "Organization"
  lane marketing "Marketing Team"
    start s1 "Define campaign objectives"
    and g1
    task t_create "Create content"
    task t_design "Design visuals"
    task t_select "Select promotion channels"
    and g2
    task t_launch "Launch campaign"
    task t_end "End campaign period"
    task t_analyze "Analyze performance"
    end e_end "Process completed"

  lane system "CRM System"
    task t_track "Track performance in real-time"
    task t_collect "Collect leads in CRM system"

  lane sales "Sales Team"
    task t_follow "Follow up leads"

flow s1 -> g1
flow g1 -> t_create
flow g1 -> t_design
flow g1 -> t_select
flow t_create -> g2
flow t_design -> g2
flow t_select -> g2
flow g2 -> t_launch
flow t_launch -> t_track
flow t_track -> t_collect
flow t_collect -> t_follow
flow t_follow -> t_end
flow t_end -> t_analyze
flow t_analyze -> e_end
```
