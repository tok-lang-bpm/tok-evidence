```tok
tok 1.0 "Supplier onboarding process"

pool org "Organization"
  lane company "Company"
    start s "Process start"
    task identify "Identify need for new supplier"
    flow s -> identify

  lane procurement "Procurement Team"
    task issue "Issue request for proposals (RFP)"
    task receive "Receive proposals from suppliers"
    task evaluate "Evaluate proposals"
    xor visits "Conduct site visits or interviews?"
    task conduct "Conduct site visits or interviews"
    task select "Select supplier"
    task negotiate "Begin contract negotiations"
    task agree "Agree on terms"
    task sign "Sign contract"
    and split
    task execute "Execute contract"
    and join
    end e "Process end"

    flow identify -> issue
    flow issue -> receive
    flow receive -> evaluate
    flow evaluate -> visits
    flow visits -> conduct "yes"
    flow visits -> select "no"
    flow conduct -> select
    flow select -> negotiate
    flow negotiate -> agree
    flow agree -> sign
    flow sign -> split
    flow split -> execute
    flow split -> onboard
    flow execute -> join
    flow onboard -> join
    flow join -> e

  lane supplier "Supplier"
    task onboard "Onboard supplier"
```
