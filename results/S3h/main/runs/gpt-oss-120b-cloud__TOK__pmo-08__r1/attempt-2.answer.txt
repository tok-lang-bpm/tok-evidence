```tok
tok 1.0 "Incident Management Process"

pool org "Organization"
  lane reporter "Reporter"
    start start1 "Process start"
    xor reporter_type "Reporter type?"
    task emp_report "Employee reports incident"
    task cust_report "Customer reports incident"
  lane team "Incident Response Team"
    task log_report "Log incident report"
    task assign_team "Assign incident to team"
    task gather_info "Gather necessary information"
    task identify_cause "Identify cause of incident"
    task propose_actions "Propose corrective actions"
    task implement_solution "Implement solution"
    task perform_followup "Perform follow-up"
    task close_and_notify "Close incident report and notify stakeholders"
    end end1 "Incident closed"

flow start1 -> reporter_type
flow reporter_type -> emp_report "employee"
flow reporter_type -> cust_report "customer"
flow emp_report -> log_report
flow cust_report -> log_report
flow log_report -> assign_team
flow assign_team -> gather_info
flow gather_info -> identify_cause
flow identify_cause -> propose_actions
flow propose_actions -> implement_solution
flow implement_solution -> perform_followup
flow perform_followup -> close_and_notify
flow close_and_notify -> end1
```
