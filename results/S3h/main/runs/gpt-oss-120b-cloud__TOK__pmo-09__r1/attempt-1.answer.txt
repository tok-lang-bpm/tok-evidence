```tok
tok 1.0 "Product Development Process"

pool org "Organisation"
  lane r_and_d "R&D Team"
    start start "Start"
    task identify "Identify idea for new product or improvement"
    task research "Conduct initial research and feasibility studies"
    task design "Draft design concepts"
    xor select "Promising design selected?"
    task prototype "Build prototype using available materials and resources"
    task test "Prototype undergoes various tests"
    task feedback "Collect feedback from testing phase"
    xor refine_gw "Is refinement needed?"
    task refine_task "Refine prototype"
    xor outcome_gw "Prototype approved for further development?"
    end approved "Prototype approved"
    end discarded "Prototype discarded"

    flow start -> identify
    flow identify -> research
    flow research -> design
    flow design -> select
    flow select -> prototype "yes"
    flow prototype -> test
    flow test -> feedback
    flow feedback -> refine_gw
    flow refine_gw -> refine_task "yes"
    flow refine_gw -> outcome_gw "no"
    flow refine_task -> test
    flow outcome_gw -> approved "yes"
    flow outcome_gw -> discarded "no"
```
