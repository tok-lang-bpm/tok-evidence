tok 1.0 "Subscription Service Process"

pool org "Organization"
  lane customer "Customer"
    start S1 "Customer signs up for subscription"
    task T1 "Enter personal and payment information"
    xor G1 "Customer decides to cancel?"
    task T5 "Submit cancellation request"
    end E1 "Subscription active"
  lane system "System"
    task T2 "Generate account"
    task T3 "Assign access"
    task T4 "Trigger automated billing cycles"
  lane support "Support Team"
    task T6 "Process cancellation request"
    xor G2 "Refund or charge applicable?"
    task T7 "Apply refund or charge"
    task T8 "Deactivate subscription"
    task T9 "Settle final account balance"
    end E2 "Subscription cancelled"

flow S1 -> T1 -> T2 -> T3 -> T4 -> G1
flow G1 -> E1 "no"
flow G1 -> T5 -> T6 -> G2
flow G2 -> T7 -> T8 -> T9 -> E2 "yes"
flow G2 -> T8 -> T9 -> E2 "no"
