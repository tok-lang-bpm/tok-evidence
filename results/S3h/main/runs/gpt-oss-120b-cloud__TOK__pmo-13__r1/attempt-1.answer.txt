```tok
tok 1.0 "Complaint handling process"

pool organization "Organization"
  lane customer "Customer"
    start S "Customer files complaint"
    task F "File complaint"
    end E "Process completed"
  lane custservice "Customer Service"
    task L "Log complaint and assign"
  lane investigation "Investigation Team"
    task R "Review complaint details"
    xor G "Refund approved?"
    task N1 "Notify customer"
    task N2 "Notify customer"
    task M "Mark complaint as resolved"
  lane finance "Financial Team"
    task P "Process reimbursement"
  lane customer "Customer"
    task FB "Provide feedback"

flow S -> F -> L -> R -> G
flow G -> N1 "yes"
flow G -> N2 "no"
flow N1 -> P
flow P -> M
flow N2 -> M
flow M -> FB
flow FB -> E
```
