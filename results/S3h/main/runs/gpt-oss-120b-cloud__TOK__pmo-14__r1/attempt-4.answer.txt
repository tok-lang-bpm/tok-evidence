```tok
tok 1.0 "Budget Planning Process"

pool org "Organization"
  lane department "Department"
    start S "Start budgeting process"
    task O "Outline objectives"
    end E "Budget distributed and implementation starts"
  
  lane team "Team"
    task D "Draft plan with cost estimates and justifications"
  
  lane strategy "Strategy"
    task SR "Strategic alignment meeting review"
  
  lane finance "Finance"
    task FR "Finance detailed review"
    xor ADJ "Adjustments needed?"
    task AD "Document adjustments"
    task AL "Allocate budget to department"
  
  lane stakeholders "Stakeholders"
    task AA "Approve adjustments"

flow S -> O -> D -> SR -> FR -> ADJ
flow ADJ -> AD "yes"
flow ADJ -> AL default
flow AD -> AA
flow AA -> FR
flow AL -> E
```
