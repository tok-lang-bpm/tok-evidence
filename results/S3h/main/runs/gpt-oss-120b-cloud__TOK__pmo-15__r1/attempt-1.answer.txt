```tok
tok 1.0 "Compliance Audit Process"

pool organization "Company"
  lane scheduler "Scheduler"
    start start "Start audit scheduling"
    task schedule "Schedule compliance audit"
  lane internal "Internal Team"
    and prep_start
    task prepare "Prepare documentation"
    task gather "Gather evidence"
    task assess "Conduct self-assessment"
    and prep_end
  lane auditor "Auditor"
    task review "Review processes"
    task final_audit "Conduct final audit"
  lane company "Company"
    task correct "Make corrections or improvements"
    xor decision "All criteria met?"
    and certify_start
    task award "Award certification"
    task issue "Issue official documents"
    and certify_end
    end success "Process completed"
    end failure "Certification not awarded"

flow start -> schedule -> prep_start
flow prep_start -> prepare -> prep_end
flow prep_start -> gather -> prep_end
flow prep_start -> assess -> prep_end
flow prep_end -> review -> correct -> decision
flow decision -> certify_start "yes"
flow decision -> failure "no"
flow certify_start -> award -> certify_end
flow certify_start -> issue -> certify_end
flow certify_end -> success
```
