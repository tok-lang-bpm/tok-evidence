```tok
tok 1.0 "University enrollment process"

pool university "University"

  lane prospective_student "Prospective Student"
    start start "Start enrollment process"
    task submit_app "Submit application online"
    task provide_missing "Provide missing documents"
    task confirm_enroll "Confirm enrollment by deadline"
    task receive_orientation "Receive orientation materials"
    task obtain_id "Obtain student ID card"
    task select_courses "Select courses"
    task resolve_conflicts "Resolve any schedule conflicts"
    task begin_classes "Begin attending classes"
    task add_drop "Add or drop courses within the add/drop period"
    task post_grades "Post grades"
    task review_grades "Review grades online"
    task submit_appeal "Submit appeal form"
    task await_decision "Await decision on appeal"
    task graduate "Graduate"
    task withdraw "Withdraw"
    end end_reject "Application rejected"
    end end_cancel "Application cancelled"
    end end_graduate "Graduated"
    end end_withdraw "Withdrawn"

  lane admissions_office "Admissions Office"
    task review_app "Review application and supporting documents"
    task notify_missing "Notify applicant to provide missing items"
    task send_accept "Send acceptance letter"
    task send_reject "Send rejection letter"
    task cancel_app "Cancel application"

  lane admissions_committee "Admissions Committee"
    task eval_app "Evaluate application by admissions committee"

  lane finance "Finance Department"
    task process_fees "Process application fees or waivers"

  lane it_department "IT Department"
    task setup_accounts "Set up student accounts for email, online portals, and library access"

  lane international_office "International Student Office"
    task assist_visa "Assist with visa processing"

  lane academic_advisor "Academic Advisor"
    task meet_advisor "Meet with academic advisor"
    task meet_appeals "Meet with appeals committee"

  xor missing "Are all documents present?"
  and parallel_eval "Parallel evaluation and fee processing"
  and join_eval "Join evaluation and fee"
  xor accept "Is the application accepted?"
  xor confirm "Is enrollment confirmed?"
  and join_post "Join orientation and account setup"
  xor international "Is the student international?"
  xor add_drop "Add or drop courses?"
  xor grievance "File an appeal?"
  xor continue "Next step?"
  and loop_continue "Loop back to next semester"

  flow start -> submit_app -> review_app -> missing
  flow missing -> notify_missing "no"
  flow missing -> parallel_eval "yes"

  flow notify_missing -> provide_missing -> review_app

  flow parallel_eval -> eval_app
  flow parallel_eval -> process_fees
  flow eval_app -> join_eval
  flow process_fees -> join_eval

  flow join_eval -> accept

  flow accept -> send_reject "no"
  flow accept -> send_accept "yes"

  flow send_reject -> end_reject
  flow send_accept -> confirm

  flow confirm -> cancel_app "no"
  flow confirm -> confirm_enroll "yes"

  flow cancel_app -> end_cancel
  flow confirm_enroll -> and_post_accept

  # parallel post‑acceptance activities
  and and_post_accept
  flow and_post_accept -> receive_orientation
  flow and_post_accept -> setup_accounts
  flow receive_orientation -> join_post
  flow setup_accounts -> join_post
  flow join_post -> international

  flow international -> assist_visa "yes"
  flow international -> obtain_id "no"
  flow assist_visa -> obtain_id

  flow obtain_id -> meet_advisor -> select_courses -> resolve_conflicts -> begin_classes -> add_drop

  flow add_drop -> post_grades "yes"
  flow add_drop -> post_grades "no"

  flow post_grades -> review_grades -> grievance
  flow grievance -> submit_appeal "yes"
  flow grievance -> continue "no"

  flow submit_appeal -> meet_appeals -> await_decision -> review_grades

  flow continue -> loop_continue "continue"
  flow continue -> graduate "graduate"
  flow continue -> withdraw "withdraw"

  flow loop_continue -> begin_classes

  flow graduate -> end_graduate
  flow withdraw -> end_withdraw
```
