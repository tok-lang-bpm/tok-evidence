```tok
tok 1.0 "Online shop purchase process"

pool online_shop "Online Shop"
  lane user "User"
    start start "Start order"
    task login "log in to account"
    and split1 "Select and set payment"
    task select "select items to purchase"
    task set_payment "set payment method"
    and join1 "After selection and payment method"
    and split2 "Payment and reward parallel"
    xor pay_dec "Payment decision?"
    task pay "pay"
    task installment "complete installment agreement"
    task reward "choose reward option"
    and join2 "After payment and reward"
    xor return_dec "Return items?"
    task return "return items for exchange"
    end done "order completed"
  lane shop "Shop"
    task deliver "deliver items"

# Sequence flows
flow start -> login
flow login -> split1
flow split1 -> select
flow split1 -> set_payment
flow select -> join1
flow set_payment -> join1
flow join1 -> split2
flow split2 -> pay_dec
flow split2 -> reward
flow pay_dec -> pay
flow pay_dec -> installment
flow pay -> join2
flow installment -> join2
flow reward -> join2
flow join2 -> deliver
flow deliver -> return_dec
flow return_dec -> done
flow return_dec -> return
flow return -> deliver
```
