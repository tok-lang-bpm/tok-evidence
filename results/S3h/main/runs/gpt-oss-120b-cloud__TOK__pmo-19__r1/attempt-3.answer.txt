```tok
tok 1.0 "Online shop purchase process"

pool online_shop "Online Shop"
  lane user "User"
    start start "Start order"
    task login "Log in to account"
    and split_select_set "Select items and set payment method"
    task select "Select items to purchase"
    task set_payment "Set payment method"
    and join_select_set "After selection and payment method"
    and split_payment_reward "Payment and reward parallel"
    xor payment_dec "Payment decision?"
    task pay "Pay"
    task installment "Complete installment agreement"
    task reward "Choose reward option"
    and join_payment_reward "After payment and reward"
    xor return_dec "Return items?"
    task return "Return items for exchange"
    end completed "Order completed"
  lane shop "Shop"
    task deliver "Deliver items"

# Sequence flows
flow start -> login
flow login -> split_select_set
flow split_select_set -> select
flow split_select_set -> set_payment
flow select -> join_select_set
flow set_payment -> join_select_set
flow join_select_set -> split_payment_reward
flow split_payment_reward -> payment_dec
flow split_payment_reward -> reward
flow payment_dec -> pay "pay"
flow payment_dec -> installment "installment"
flow pay -> join_payment_reward
flow installment -> join_payment_reward
flow reward -> join_payment_reward
flow join_payment_reward -> deliver
flow deliver -> return_dec
flow return_dec -> completed "no return"
flow return_dec -> return "return"
flow return -> deliver
```
