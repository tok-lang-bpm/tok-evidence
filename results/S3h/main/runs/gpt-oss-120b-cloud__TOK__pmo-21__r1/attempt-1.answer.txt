```tok
tok 1.0 "Shipping Process" [executable]

pool org "Organisation"
  lane secretary "Secretary"
    start start "Start shipping process"
    xor ship_decision "Ship goods?"
    task clarify "Clarify who will do the shipping"
    xor amount_decision "Large amounts?"
    task invite "Invite logistic companies"
    task select "Select logistic company"
    xor amount_join "Amount join"
    and split "Parallel activities"
    task label "Write package label"
    and join "Parallel join"
    task prepare "Prepare goods for pickup"
    end done "Shipping process completed"
    end not_shipped "No shipping required"

  lane logistics_head "Logistics Department Head"
    task insurance "Take parcel insurance"

  lane warehousemen "Warehousemen"
    task package "Package goods"

# Sequence flows
flow start -> ship_decision
flow ship_decision -> clarify "yes"
flow ship_decision -> not_shipped "no"

flow clarify -> amount_decision

flow amount_decision -> invite "yes"
flow amount_decision -> amount_join "no"

flow invite -> select
flow select -> amount_join

flow amount_join -> split

flow split -> label
flow split -> insurance
flow split -> package

flow label -> join
flow insurance -> join
flow package -> join

flow join -> prepare
flow prepare -> done
```
