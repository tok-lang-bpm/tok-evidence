```tok
tok 1.0 "Internship"

pool organization "Organization"
  lane intern "Intern"
    start start "Start internship"
    task e1 "Enter topic and desired money"
    task e2 "Enter experience and hobbies"
    task recv "Receive offer"
    xor decide "Offer decision?"
    task accept "Accept offer"
    task deny "Deny offer"
    and updates_split
    task iu "Write intern status update" loop while="intern_updates < 3"
    and updates_join
    task rec "Recommend company via Twitter"
    and tweets_split
    task tweet1 "Send tweet to friend 1"
    task tweet2 "Send tweet to friend 2"
    task tweet3 "Send tweet to friend 3"
    and tweets_join
    end end "Internship completed"

  lane company "Company"
    task cu "Write company status update" loop while="company_updates < 3"

flow start -> e1 -> e2 -> recv -> decide
flow decide -> accept "accept"
flow decide -> deny "deny"
flow accept -> updates_split
flow deny -> recv
flow updates_split -> iu
flow updates_split -> cu
flow iu -> updates_join
flow cu -> updates_join
flow updates_join -> rec
flow rec -> tweets_split
flow tweets_split -> tweet1
flow tweets_split -> tweet2
flow tweets_split -> tweet3
flow tweet1 -> tweets_join
flow tweet2 -> tweets_join
flow tweet3 -> tweets_join
flow tweets_join -> end
```
