```tok
tok 1.0 "LAN Party"

pool org "LAN Party Organisation"
  lane organizer "Organizer"
    start start "Start planning"
    task send_inv "Send invitations to 10 friends"
    task find_games "Find out which games friends want to play"
    task appoint_date "Appoint a date for the LAN party"
    xor agreement "8 or more people agree on date?"
    task date_confirm "Date confirmed"
    and split
    and join
    task check_games "Check if you have all the games"
    xor games_available "All required games are available?"
    task download_games "Download missing games"
    task find_beer "Find out how much beer friends will require"
    xor beer_enough "Enough beer available?"
    task buy_beer "Buy missing beer"
    task enjoy "Enjoy the LAN party"
    end end "LAN party enjoyed"

    flow start -> send_inv -> find_games -> appoint_date -> agreement
    flow agreement "yes" -> split
    flow agreement "no" -> appoint_date

    flow split -> date_confirm -> join
    flow split -> check_games -> games_available

    flow games_available "yes" -> join
    flow games_available "no" -> download_games -> join

    flow join -> find_beer -> beer_enough
    flow beer_enough "yes" -> enjoy
    flow beer_enough "no" -> buy_beer -> enjoy

    flow enjoy -> end
```
