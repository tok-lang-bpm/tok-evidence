```tok
tok 1.0 "Farming Bot"

pool org "Organisation"
  lane player "Player"
    start s "Start"
    task t_find "Find resources to farm"
    xor g_tool "Is a tool needed?"
    task t_create "Create tool"
  lane bot "Bot"
    and g_collect_split "Start collecting"
    task t_collect "Collect resources"
    and g_collect_join "Collect done"
    and g_farm_split "Start farming"
    task t_farm "Farm resources"
    task t_notify "Notify milestones"
    and g_farm_join "Farming sync"
    xor g_disaster "Natural disaster occurred?"
    task t_handle "Handle disaster"
    xor g_update "Update resource list?"
    task t_update "Update list of farmed materials"
    end e_finish "Farming finished"
    and g_brag_split "Post‑farming split"
    task t_brag "Brag to selected friends"
    task t_send "Send materials to friends"
    and g_brag_join "Post‑farming done"
    end e_end "Process completed"

# Flows
flow s -> t_find -> g_tool
flow g_tool -> t_create -> g_collect_split
flow g_tool -> g_collect_split
flow g_collect_split -> t_collect -> g_collect_join
flow g_collect_join -> g_farm_split
flow g_farm_split -> t_farm -> g_farm_join
flow g_farm_split -> t_notify -> g_farm_join
flow g_farm_join -> g_disaster
flow g_disaster -> t_handle -> g_farm_split
flow g_disaster -> g_update
flow g_update -> t_update -> g_farm_split
flow g_update -> e_finish
flow e_finish -> g_brag_split
flow g_brag_split -> t_brag -> g_brag_join
flow g_brag_split -> t_send -> g_brag_join
flow g_brag_join -> e_end
```
