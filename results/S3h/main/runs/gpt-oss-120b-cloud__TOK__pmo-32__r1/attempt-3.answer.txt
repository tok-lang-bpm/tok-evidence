```tok
tok 1.0 "Farming Bot"

pool org "Organisation"
  lane player "Player"
    start s "Start"
    task t_find "Find desired resources"
    xor g_tool_needed "Tool needed?"
    task t_create "Create required tool"
  lane bot "Bot"
    and g_farm_split "Start farming"
    task t_farm "Farm resources"
    task t_notify "Notify milestones"
    and g_farm_join "Farming sync"
    xor g_disaster "Natural disaster occurred?"
    task t_handle "Handle disaster"
    xor g_update "Update resource list?"
    task t_update "Update list of farmed materials"
    and g_bragsplit "Post‑farming actions"
    task t_brag "Brag to selected friends"
    task t_send "Send materials to friends"
    and g_bragjoin "Post‑farming sync"
    end e_end "Process completed"

# Flows
flow s -> t_find
flow t_find -> g_tool_needed
flow g_tool_needed -> t_create "yes" when="toolNeeded"
flow g_tool_needed -> g_farm_split "no" when="!toolNeeded"
flow t_create -> g_farm_split

flow g_farm_split -> t_farm
flow g_farm_split -> t_notify
flow t_farm -> g_farm_join
flow t_notify -> g_farm_join

flow g_farm_join -> g_disaster

flow g_disaster -> t_handle "yes" when="disaster"
flow g_disaster -> g_update "no" when="!disaster"

flow t_handle -> g_farm_split

flow g_update -> t_update "yes" when="update"
flow g_update -> g_bragsplit "no" when="!update"

flow t_update -> g_farm_split

flow g_bragsplit -> t_brag
flow g_bragsplit -> t_send
flow t_brag -> g_bragjoin
flow t_send -> g_bragjoin

flow g_bragjoin -> e_end
```
