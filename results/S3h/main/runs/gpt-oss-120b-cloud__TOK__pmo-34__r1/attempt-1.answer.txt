```tok
tok 1.0 "Building a custom machine out of Lego bricks"

pool org "Lego Workshop"
  lane builder "Builder"
    start start1 "Start building custom machine"
    task design "Develop the basic design"
    task order "Order LEGO brick sets"
    task give "Give LEGO sets to children"
    task build_sub "Build subcomponent"
    xor parts_check "Parts available?"
    task reorder "Reorder individual parts"
    task test "Test subcomponent"
    xor test_result "Test successful?"
    task redesign "Redesign subcomponent"
    task assemble "Assemble subcomponent"
    xor more_sub "More subcomponents?"
    end end_done "Machine completed"

  lane children "Children"
    task sort "Sort parts into containers"

flow start1 -> design
flow design -> order
flow order -> give
flow give -> sort
flow sort -> build_sub
flow build_sub -> parts_check
flow parts_check -> reorder "no"
flow parts_check -> test "yes"
flow reorder -> build_sub
flow test -> test_result
flow test_result -> redesign "no"
flow test_result -> assemble "yes"
flow redesign -> build_sub
flow assemble -> more_sub
flow more_sub -> build_sub "yes"
flow more_sub -> end_done "no"
```
