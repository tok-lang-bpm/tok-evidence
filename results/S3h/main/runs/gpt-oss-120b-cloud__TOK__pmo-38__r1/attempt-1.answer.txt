```tok
tok 1.0 "Building a House"

pool project "Building a House"
  lane you "You"
    start s "Start building a tree house"
    task collect "Collect requirements"
    task send_req "Send requirements to architect"
    task refine "Refine draft with additional requirements"
    task create_materials "Create list of needed materials from plan"
    and fork "Parallel order and building"
    task order_materials "Order materials from online stores"
    task send_messages "Send messages to friends to build the house"
    and join "Parallel join"
    task send_invitations "Send invitations for a tree house party"
    task create_attendance "Create list of people that attend the party"
    end e "Process completed"

  lane architect "Architect"
    task receive_requirements "Receive requirements"
    task send_draft "Send draft to you"

  lane friends "Friends"
    task build_house "Build the house"

flow s -> collect
flow collect -> send_req
flow send_req -> receive_requirements
flow receive_requirements -> send_draft
flow send_draft -> refine
flow refine -> create_materials
flow create_materials -> fork
flow fork -> order_materials
flow fork -> send_messages
flow send_messages -> build_house
flow order_materials -> join
flow build_house -> join
flow join -> send_invitations
flow send_invitations -> create_attendance
flow create_attendance -> e
```
