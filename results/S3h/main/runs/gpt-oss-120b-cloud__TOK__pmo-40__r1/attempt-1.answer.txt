```tok
tok 1.0 "OEE"

pool org "Organization"
  lane machine "Machine"
    start s "Start OEE Process"
    task collect "Collect relevant information from machine"
      # boundary event for disconnection
    boundary disc "Machine disconnected" on=collect
    task stop "Stop the machine"
  lane system "System"
    task calc "Calculate the Overall Equipment Efficiency (OEE)"
    xor dec "OEE below threshold?"
    and g1 "Parallel actions"
    and g2 "Join actions"
    task restart "Restart the machine"
  lane engineer "Engineer"
    task email "Send email to engineer"

  end end_ok "OEE acceptable"
  end end_cont "Measurement continued after restart"
  end end_disc "Process stopped due to machine disconnection"

flow s -> collect
flow disc -> end_disc
flow collect -> calc
flow calc -> dec
flow dec -> end_ok "no"
flow dec -> g1 "yes"
flow g1 -> stop
flow g1 -> email
flow stop -> g2
flow email -> g2
flow g2 -> restart
flow restart -> end_cont
```
