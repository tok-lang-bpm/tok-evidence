tok 1.0 "Inspection of an Energy Drink Bottling Machine" [executable]
    [process-name="InspectionProcess"]
    pool org "Organization"
      lane inspector "Inspector"
        start s "Start inspection"
        task enter "Enter machine type and serial number"
        task answer_questions "Answer inspection questions"
        and split1
        task press_buttons "Press inspection buttons"
        xor more_questions "Additional questions needed?"
        task answer_additional "Answer additional questions"
        end e_complete "Inspection complete"

        flow s -> enter -> answer_questions -> split1
        flow split1 -> press_buttons
        flow split1 -> collect_values
        flow press_buttons -> join1
        flow collect_values -> join1
        flow join1 -> more_questions
        flow more_questions -> answer_additional "yes"
        flow answer_additional -> e_complete
        flow more_questions -> e_complete "no"

      lane machine "Machine"
        task collect_values "Collect machine values"
        and join1
