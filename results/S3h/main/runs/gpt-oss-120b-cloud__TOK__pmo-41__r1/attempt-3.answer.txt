```tok
tok 1.0 "Inspection of an Energy Drink Bottling Machine" executable
    pool org "Organisation"
        lane inspector "Inspector"
            start s "Start inspection"
            task t_enter "Enter machine type and serial number"
            task t_questions "Answer inspection questions"
            and split_ps "Parallel split"
            task t_press "Press inspection buttons"
            xor decision_more "Additional questions needed?"
            task t_additional "Answer additional questions"
            and join_pj "Parallel join"
            end e "Inspection complete"

        lane machine "Machine"
            task t_collect "Collect machine values"

        flow s -> t_enter -> t_questions -> split_ps
        flow split_ps -> t_press
        flow split_ps -> t_collect
        flow t_press -> join_pj
        flow t_collect -> join_pj
        flow join_pj -> decision_more
        flow decision_more -> t_additional "yes"
        flow t_additional -> e
        flow decision_more -> e "no"
```
