```tok
tok 1.0 "Car Service"

pool org "Car Service Organisation"
  lane system "System"
    start start1 "Service completed"
    task checkReg "Check car registration"
    xor regCheck "Car registered?"
    task notify "Notify customer"
    xor goService "Customer goes to service?"
    task fine "Fine customer"
    task statusUpdate "Send status updates via email"
    end fineEnd "Process ended with fine"
  lane customer "Customer"
    task goToFacility "Go to service facility"
    task payApp "Pay through app"
  lane mechanic "Mechanic"
    task enterProblems "Enter car problems"
    and parallelSplit "Parallel split"
    and parallelJoin "Parallel join"
    task repairCar "Repair car"
    task enterSuccess "Enter repair successful and issue Pickerl"
    task enterNextService "Enter next service time"
    end end1 "Process completed"

  flow start1 -> checkReg -> regCheck
  flow regCheck -> notify "yes"
  flow regCheck -> fine "no"
  flow notify -> goService
  flow goService -> fine "no"
  flow goService -> goToFacility "yes"
  flow goToFacility -> enterProblems
  flow enterProblems -> parallelSplit
  flow parallelSplit -> statusUpdate
  flow parallelSplit -> repairCar
  flow statusUpdate -> parallelJoin
  flow repairCar -> parallelJoin
  flow parallelJoin -> enterSuccess
  flow enterSuccess -> enterNextService
  flow enterNextService -> payApp
  flow payApp -> end1
  flow fine -> fineEnd
```
