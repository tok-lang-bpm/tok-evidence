tok 1.0 "Robotic Burger Seller near the University of Vienna"

pool seller "Robotic Burger Seller"
  lane robot "Robot"
    start s "Receive order"
    xor dec1 "Menu or just burger?"
    and split "Menu parallel"
    task drink "Prepare the drink"
    task ask "Ask fries or wedges"
    xor dec2 "Fries or wedges?"
    task fries "Prepare the fries"
    task wedges "Prepare the wedges"
    and join "Menu join"
    task burger "Prepare the burger"
    task deliver "Deliver order via conveyor belt"
    end e "Order completed"

flow s -> dec1
flow dec1 -> split "menu"
flow dec1 -> burger "just burger"
flow split -> drink
flow split -> ask
flow drink -> join
flow ask -> dec2
flow dec2 -> fries "fries"
flow dec2 -> wedges "wedges"
flow fries -> join
flow wedges -> join
flow join -> burger
flow burger -> deliver
flow deliver -> e
