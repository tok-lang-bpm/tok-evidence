```tok
tok 1.0 "App For Participating at the Vienna Night Run"

pool org "Organisation"
  lane participant "Participant"
    start start "Start participation"
    task select_block "Select starting block"
    task run_test "Run 5km and measure time"
    xor time_check "Time less than 25 minutes?"
    task train_check "Train and check"
    task get_number "Get starting number"
    task determine_travel "Determine travel option"
    xor travel_choice "More than one hour left?"
    task go_home "Go to event from home"
    task leave_work "Leave directly from work"
    xor travel_join "Travel option join"
    and parallel_split "Run and drink split"
    task run_event "Run at Night Run"
    task drink "Drink during run"
    and parallel_join "Run and drink join"
    task receive_time "Receive final running time"
    end end "Process completed"

flow start -> select_block -> run_test -> time_check
flow time_check -> get_number "yes"
flow time_check -> train_check "no"
flow train_check -> run_test
flow get_number -> determine_travel -> travel_choice
flow travel_choice -> go_home "yes"
flow travel_choice -> leave_work "no"
flow go_home -> travel_join
flow leave_work -> travel_join
flow travel_join -> parallel_split
flow parallel_split -> run_event
flow parallel_split -> drink
flow run_event -> parallel_join
flow drink -> parallel_join
flow parallel_join -> receive_time -> end
```
