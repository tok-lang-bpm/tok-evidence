```tok
tok 1.0 "New hire process"

pool org "Organisation"
  lane dept "Department"
    start needed "New hire need identified"
    task identify "Identify the need for a new hire"
  lane hr "HR"
    task describe "Create the job description"
    task post "Post the job description on job boards"
    task collect "Collect the resumes"
    task screen "Screen the resumes"
    task phone "Conduct the initial phone interviews"
    xor preference "Candidate's preference?"
    task inperson "Conduct the in-person interviews"
    task virtual "Conduct the virtual interviews"
    xor interviewed "Interview done"
    xor chosen "Candidate chosen?"
    task offer "Extend the offer"
    xor negotiate "Salary negotiation needed?"
    task negotiate_salary "Negotiate the salary"
    xor accepted "Offer accepted?"
    task onboard "Begin the onboarding process"
    task paperwork "Complete the paperwork"
    task orient "Conduct the orientation"
    task train "Conduct the training"
    and onboarding_done "Onboarding steps joined"
    task integrate "Integrate the new hire into the team"
    end done "New hire fully integrated"
    end declined "Offer declined"

flow needed -> identify -> describe -> post -> collect -> screen -> phone -> preference
flow preference -> inperson "in person"
flow preference -> virtual "virtual"
flow inperson -> interviewed
flow virtual -> interviewed
flow interviewed -> chosen
flow chosen -> offer "yes"
flow chosen -> declined "no"
flow offer -> negotiate
flow negotiate -> negotiate_salary "yes"
flow negotiate -> accepted "no"
flow negotiate_salary -> accepted
flow accepted -> onboard "yes"
flow accepted -> declined "no"
flow onboard -> paperwork
flow onboard -> orient
flow onboard -> train
flow paperwork -> onboarding_done
flow orient -> onboarding_done
flow train -> onboarding_done
flow onboarding_done -> integrate -> done
```
