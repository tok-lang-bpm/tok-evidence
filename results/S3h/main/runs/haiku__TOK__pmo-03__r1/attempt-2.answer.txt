```tok
tok 1.0 "Inventory restock"

pool store "Warehouse or store"
  lane staff "Warehouse staff"
    start begin "Inventory monitoring begins"
    task monitor "Monitor inventory levels"
    xor threshold "Stock reached threshold?"
    xor trigger "Automated alert or manual check?"
    task check "Perform manual stock check"
    xor reorder "Reorder signalled"
    task receive "Receive the stock"
    task inspect "Inspect stock quality"
    task record "Record stock in the system"
    task shelve "Place stock on shelves or in storage"
  lane system "Inventory system"
    task alert "Generate automated alert"
    task dates "Update inventory system with expected delivery dates"
    task levels "Update inventory levels after restock"
    end done "Inventory levels updated"
  lane procurement "Procurement team"
    task order "Place order with suppliers considering cost, delivery time and supplier reliability"

flow begin -> monitor -> threshold
flow threshold -> trigger "yes" when="stock_reached = true"
flow threshold -> monitor "no" default
flow trigger -> alert "automated alert"
flow trigger -> check "manual check"
flow alert -> reorder
flow check -> reorder
flow reorder -> order -> dates -> receive -> inspect -> record -> shelve -> levels -> done
```
