```tok
tok 1.0 "Marketing campaign"

pool org "Organisation"
  lane marketing "Marketing team"
    start begin "Campaign process started"
    task objectives "Define the campaign objectives"
    task content "Create the content"
    task visuals "Design the visuals"
    task channels "Select the promotion channels"
    task launch "Launch the campaign"
    and fork "Split"
    task track "Track the campaign performance in real time"
    catch period "Campaign period ends" timer=P30D
    task analyze "Analyze the campaign performance"
    end done "Campaign concluded"
  lane sales "Sales team"
    task follow "Follow up on the leads"
  lane crm "CRM system"
    task collect "Collect the leads in the CRM system"
    and join "Join"

flow begin -> objectives -> content -> visuals -> channels -> launch -> fork
flow fork -> track
flow fork -> collect
flow collect -> follow
flow track -> join
flow follow -> join
flow join -> period -> analyze -> done
```
