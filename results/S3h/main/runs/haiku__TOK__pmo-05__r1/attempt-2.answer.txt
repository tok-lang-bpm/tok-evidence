tok 1.0 "Supplier selection"

pool org "Organisation"
  lane company "Company"
    start need "Need for a new supplier identified"
    task identify "Identify the need for a new supplier"
    and split
    task onboard "Onboard the supplier"
    task execute "Execute the contract"
    and joined
    end done "Supplier onboarded and contract executed"
  lane procurement "Procurement team"
    task rfp "Issue a request for proposals"
    task receive "Receive responses from suppliers"
    task evaluate "Evaluate each proposal on cost, quality and delivery time"
    xor offers "Site visits or interviews needed?"
    task visit "Conduct site visits"
    task interview "Conduct interviews with suppliers"
    xor checked
    task select "Select a supplier"
    task negotiate "Negotiate contract terms"
    xor agreed "Both parties agree on the terms?"
    task sign "Sign the contract"
  lane suppliers "Suppliers"
    task respond "Respond to the request for proposals"

flow need -> identify -> rfp -> respond -> receive -> evaluate -> offers
flow offers -> visit "site visit" when="needsVisit = true"
flow offers -> interview "interview" when="needsInterview = true"
flow offers -> checked "none" default
flow visit -> checked
flow interview -> checked
flow checked -> select -> negotiate -> agreed
flow agreed -> sign "yes"
flow agreed -> negotiate "no"
flow sign -> split
flow split -> onboard -> joined
flow split -> execute -> joined
flow joined -> done
