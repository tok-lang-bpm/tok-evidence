```tok
tok 1.0 "Order fulfilment"

pool company "Company"
  lane customer "Customer"
    start placed "Order placed"
    task place "Place the order via online platform or phone"
    task receiveConfirm "Receive the order confirmation"
    task receiveTracking "Receive the tracking information"
    task confirmDelivery "Confirm delivery at the address"
    xor feedbackAsk "Feedback or return?"
    end feedback "Customer feedback or return initiated"
  lane system "System"
    task generateConfirm "Generate the order confirmation"
    task sendConfirm "Send the order confirmation to the customer"
    task sendTracking "Send the tracking information to the customer"
    task monitor "Monitor the shipment until delivery"
    xor delivered "Shipment reached the customer's address?"
    end completed "Delivery completed"
  lane warehouse "Warehouse team"
    task pick "Pick the items"
    task pack "Pack the items"
    task label "Generate the shipping label"
  lane logistics "Logistics provider"
    task handover "Hand over the order to the logistics provider"
    and dispatch
    task deliver "Deliver the shipment"
    and tracked

flow placed -> place -> generateConfirm -> sendConfirm -> receiveConfirm -> pick -> pack -> label -> handover -> dispatch
flow dispatch -> sendTracking -> receiveTracking -> tracked
flow dispatch -> deliver -> tracked
flow tracked -> monitor -> delivered
flow delivered -> confirmDelivery "yes"
flow delivered -> monitor "no"
flow confirmDelivery -> feedbackAsk
flow feedbackAsk -> feedback "yes"
flow feedbackAsk -> completed "no"
```
