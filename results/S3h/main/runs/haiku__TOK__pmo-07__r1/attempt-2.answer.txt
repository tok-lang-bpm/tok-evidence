```tok
tok 1.0 "Ticket booking and travel"

pool company "Travel company"
  lane customer "Customer"
    start begin "Customer needs a ticket"
    task search "Search for a flight, train, or bus ticket"
    task select "Select the route, date, and time"
    task provide "Provide personal information and payment details"
    task journey "Complete the journey"
    xor feedback_q "Feedback or services after travel?"
    task feedback "Give post-travel feedback or use services"
    end done "Process concluded"
  lane booking "Booking system"
    task generate "Generate the ticket"
    and dispatch "Split"
    task send "Send the ticket by email or SMS"
    and joined "Join"
  lane travel "Travel company"
    task inventory "Update the seat inventory"
    xor reminder_q "Reminders and instructions to send?"
    task remind "Send reminders and instructions for check-in or boarding"
    task checkin "Check in and board"

flow begin -> search -> select -> provide -> generate -> dispatch
flow dispatch -> send -> joined
flow dispatch -> inventory -> joined
flow joined -> reminder_q
flow reminder_q -> remind "yes"
flow reminder_q -> checkin "no" default
flow remind -> checkin -> journey -> feedback_q
flow feedback_q -> feedback "yes"
flow feedback_q -> done "no" default
flow feedback -> done
```
