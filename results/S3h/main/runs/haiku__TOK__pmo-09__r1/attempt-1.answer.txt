```tok
tok 1.0 "New product development"

pool org "Organisation"
  lane rd "R&D team"
    start idea "Idea for a new or improved product identified"
    task research "Conduct initial research and feasibility studies"
    task draft "Draft design concepts"
    task select "Select a promising design"
    task build "Build a prototype using available materials and resources"
    task test "Test the prototype for functionality, safety and market potential"
    task collect "Collect feedback from the testing phase"
    xor refine "Refinement needed?"
    task refine_proto "Refine the prototype"
    xor approved "Prototype approved or discarded?"
    end approvedEnd "Prototype approved for further development"
    end discardedEnd "Prototype discarded"

flow idea -> research -> draft -> select -> build -> test -> collect -> refine
flow refine -> refine_proto "yes"
flow refine -> approved "no"
flow refine_proto -> test
flow approved -> approvedEnd "approved"
flow approved -> discardedEnd "discarded"
```
