```tok
tok 1.0 "IT solution request handling"

pool organisation "Organisation"
  lane requester "Requesting employee or department"
    start submitted "Request needed"
    task submit "Submit a request for an IT solution"
    task receive "Receive the new solution"
    xor training "Training needed?"
    task train "Provide training"
    task receive_support "Use IT support for troubleshooting"
  lane it "IT"
    task assess "Assess the request"
    task check_compat "Check compatibility with existing systems"
    task evaluate "Evaluate cost and resource needs"
    xor approved "Request approved?"
    task procure "Procure the necessary tools or licenses"
    task install "Install the solution"
    task test "Test the solution in a controlled environment"
    xor passed "Testing successful?"
    task rollout "Roll out the new solution to the requesting department"
    task provide_support "Provide IT support"
    end rejected "Request rejected"
    end finished "Solution in use"

flow submitted -> submit -> assess -> check_compat -> evaluate -> approved
flow approved -> procure "yes"
flow approved -> rejected "no"
flow procure -> install -> test -> passed
flow passed -> rollout "yes"
flow passed -> install "no"
flow rollout -> receive
flow receive -> training
flow training -> train "yes" when="trainingNeeded = true"
flow training -> provide_support "no" default
flow train -> provide_support
flow provide_support -> receive_support
flow receive_support -> finished
```
