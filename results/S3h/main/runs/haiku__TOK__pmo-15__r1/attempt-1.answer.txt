```tok
tok 1.0 "Compliance audit"

pool company "Company"
  lane management "Management"
    start scheduled "Compliance audit scheduled"
    task schedule "Schedule the compliance audit"
  lane internal "Internal teams"
    task prepare "Prepare the documentation"
    task gather "Gather the evidence"
    task selfassess "Conduct the self-assessment"
  lane auditor "External auditors"
    task review "Review the company's processes"
    xor gaps "Gaps or issues found?"
    task identify "Identify the gaps or issues"
  lane company2 "Company"
    task correct "Make the necessary corrections or improvements"
    task finalaudit "Undergo the final audit"
    xor criteria "All criteria met?"
    task certify "Award the certification"
    task issue "Issue the official documents"
    end concluded "Process concluded"
    end notcertified "Certification not awarded"

flow scheduled -> schedule -> prepare -> gather -> selfassess -> review -> gaps
flow gaps -> identify "yes"
flow gaps -> finalaudit "no"
flow identify -> correct -> finalaudit -> criteria
flow criteria -> certify "yes"
flow criteria -> notcertified "no"
flow certify -> issue -> concluded
```
