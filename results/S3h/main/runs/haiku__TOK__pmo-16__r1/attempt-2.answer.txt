```tok
tok 1.0 "Multinational company annual audit"

pool org "Multinational company"
  lane coord "Audit coordinator"
    start initiate "Audit process needed"
    task notify "Send notification to the regional office"
    and forkprep
  lane regional "Regional office"
    task prepare "Prepare financial statements"
    task gather "Gather necessary documents"
    and joinprep
    task submit "Submit the documents to the central audit team"
    task clarify "Provide the necessary clarifications"
  lane compliance "Compliance team"
    task checkreg "Check for regulatory updates affecting the audit"
  lane central "Central audit team"
    task review "Review the submission"
    xor discrepancy "Discrepancies found?"
    task request "Request clarifications from the regional office"
    xor reviewed "All documents reviewed without issues?"
    task assess "Conduct risk assessment"
    and assessstart
    task finrisk "Evaluate financial risks"
    task oprisk "Evaluate operational risks"
    task comprisk "Evaluate compliance risks"
    and assessend
    xor highrisk "High risk identified?"
    task investigate "Launch detailed investigation"
    task analyze "Perform data analysis"
    task interview "Conduct interviews"
    xor sitevisit "Site visits required?"
    task visit "Conduct site visits"
    xor risksaddressed "All risks addressed?"
    task compile "Compile the audit report"
  lane director "Audit director"
    task reviewreport "Review the audit report"
    xor decision "Report approved?"
    task revise "Update the report with revisions"
    task distribute "Distribute the final report to the executive board"
    task archive "Archive the final report"
    end closed "Audit process closed"

flow initiate -> notify -> forkprep
flow forkprep -> prepare -> gather -> joinprep
flow forkprep -> checkreg -> joinprep
flow joinprep -> submit -> review -> discrepancy
flow discrepancy -> request "yes"
flow discrepancy -> reviewed "no"
flow request -> clarify -> reviewed
flow reviewed -> assess "yes"
flow reviewed -> review "no"
flow assess -> assessstart
flow assessstart -> finrisk
flow assessstart -> oprisk
flow assessstart -> comprisk
flow finrisk -> assessend
flow oprisk -> assessend
flow comprisk -> assessend
flow assessend -> highrisk
flow highrisk -> investigate "yes"
flow highrisk -> risksaddressed "no"
flow investigate -> analyze -> interview -> sitevisit
flow sitevisit -> visit "yes"
flow sitevisit -> risksaddressed "no"
flow visit -> risksaddressed
flow risksaddressed -> compile "yes"
flow risksaddressed -> investigate "no"
flow compile -> reviewreport -> decision
flow decision -> distribute "approved"
flow decision -> revise "revisions required"
flow revise -> reviewreport
flow distribute -> archive -> closed
```
