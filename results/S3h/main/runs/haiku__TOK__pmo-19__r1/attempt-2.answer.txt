```tok
tok 1.0 "Online shop purchase"

pool onlineshop "Online shop"
  lane customer "Customer"
    start started "Order started"
    task login "Log in to the account"
    and split1
    task select "Select the items to purchase"
    task method "Set the payment method"
    and join1
    and split2
    xor payment "Pay or complete an installment agreement?"
    task pay "Pay for the order"
    task installment "Complete the installment agreement"
    xor paid
    task reward "Choose a free reward option"
    and join2
    xor returned "Items returned?"
    task give_back "Return the items for exchange"
    end done "Order completed"
  lane shop "Shop"
    task deliver "Deliver the items"

flow started -> login -> split1
flow split1 -> select
flow split1 -> method
flow select -> join1
flow method -> join1
flow join1 -> split2
flow split2 -> payment
flow split2 -> reward
flow payment -> pay "Pay"
flow payment -> installment "Installment"
flow pay -> paid
flow installment -> paid
flow paid -> join2
flow reward -> join2
flow join2 -> deliver -> returned
flow returned -> give_back "Yes"
flow returned -> done "No"
flow give_back -> deliver
```
