```tok
tok 1.0 "Goods shipping"

pool company "Company"
  lane secretary "Secretary"
    start needed "Shipping required"
    xor amount "Large amounts?"
    task invite "Invite three logistic companies to make offers"
    task select "Select one logistic company"
    task clarify "Clarify who will do the shipping"
    task label "Write package label"
    xor insured "Parcel insurance necessary?"
    and ready
    xor prepared "Everything ready?"
    task postship "Use normal post shipment"
    xor merged "Shipping method chosen"
    end shipped "Goods shipped"
  lane logistics "Logistics department head"
    task insure "Take out parcel insurance"
  lane warehouse "Warehousemen"
    task pack "Package the goods"
    task preparepickup "Prepare packaged goods for pickup by logistic company"

flow needed -> clarify -> amount
flow amount -> invite "yes"
flow amount -> postship "no"
flow invite -> select -> merged
flow postship -> merged
flow merged -> label
flow label -> insured
flow insured -> insure "yes"
flow insured -> ready "no"
flow insure -> ready
flow pack -> ready
flow ready -> prepared
flow prepared -> preparepickup "yes"
flow prepared -> prepared "no"
flow preparepickup -> shipped
```
