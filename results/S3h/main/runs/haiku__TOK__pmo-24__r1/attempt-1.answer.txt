```tok
tok 1.0 "Restaurant service"

pool restaurant "Restaurant"
  lane guest "Guest"
    start hungry "Guest feels hungry"
    task enter "Enter the restaurant"
    task choose "Choose a dish from the changing meal range"
    task wait "Wait until it is his turn"
    task order "Place the order with the employee"
    task findout "Find out that the meal is ready"
    task appear "Appear at the service hatch"
    task pickup "Pick up the meal"
    task eat "Eat the meal"
    end eaten "Meal eaten"
  lane employee "Employee"
    task enterpos "Enter the order into the POS system"
    task collect "Collect the money from the guest"
    task setbuzzer "Set up a buzzer"
    task passbuzzer "Pass the buzzer to the guest with the information: when the buzzer rings, your dinner is ready"
    task informchef "Inform the chef of the new meal order"
    task setoff "Set off the guest's buzzer"
    xor reacts "Guest reacts to the buzzer?"
    catch wait5 "Wait 5 minutes" timer=PT5M
    task callguest "Call for the guest"
    task handover "Hand over the meal"
  lane chef "Chef"
    task prepare "Prepare the meal"
    task hatch "Place the finished meal in the service hatch"
    task informready "Inform the employee that the meal is in the service hatch"

flow hungry -> enter -> choose -> wait -> order -> enterpos -> collect -> setbuzzer -> passbuzzer -> informchef -> prepare -> hatch -> informready -> setoff -> findout -> reacts
flow reacts -> appear "yes"
flow reacts -> wait5 "no"
flow wait5 -> callguest -> reacts
flow appear -> handover -> pickup -> eat -> eaten
```
