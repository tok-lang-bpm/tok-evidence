```tok
tok 1.0 "Becoming A Parent"

pool office "Maternity leave administration"
  lane parent "Parent"
    start planning "Parent plans maternity leave"
    task fetch "Fetch information about potential leave models"
    task select "Select leave model"
    task collect "Collect relevant information"
    and notify
    task notifySS "Notify Social Security in time"
    task notifyCo "Notify the company in time"
    and notified
    catch periodEnd "Leave period ends" timer=P12M   # duration depends on the chosen model
    xor extend "Extend leave?"
    task request "Request leave extension"
    end extended "Leave extended"
    end ended "Leave ends"
  lane ss "Social Security"
    task gather "Gather information from companies"

flow planning -> fetch -> select -> collect -> notify
flow notify -> notifySS -> notified
flow notify -> notifyCo -> notified
flow notified -> gather -> periodEnd -> extend
flow extend -> request "yes"
flow extend -> ended "no"
flow request -> extended
```
