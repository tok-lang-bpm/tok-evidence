```tok
tok 1.0 "LAN Party"

pool organiser "Organiser"
  lane host "Host"
    start planning "Planning the LAN party"
    task invite "Send invitations to the 10 friends"
    task askGames "Find out which games they want to play"
    and fork
    task appointDate "Appoint a date for the LAN party"
    xor enough "8 or more people agree on the date?"
    task findOtherDate "Appoint another date"
    xor games "Do you have all the games?"
    task download "Download the missing games"
    and joined
    task findBeer "Find out how much beer is required"
    xor beer "Enough beer?"
    task buyBeer "Buy the missing beer"
    and beerReady
    task enjoy "Enjoy the LAN party"
    end done "LAN party finished"

flow planning -> invite -> askGames -> fork
flow fork -> appointDate -> enough
flow fork -> games
flow enough -> joined "yes"
flow enough -> findOtherDate "no"
flow findOtherDate -> enough
flow games -> joined "yes"
flow games -> download "no"
flow download -> joined
flow joined -> findBeer -> beer
flow beer -> beerReady "yes"
flow beer -> buyBeer "no"
flow buyBeer -> beerReady
flow beerReady -> enjoy -> done
```
