```tok
tok 1.0 "Instruct an artist to create a 3D model and print it"

pool maker "Maker"
  lane customer "Customer"
    start craving "Custom 3D model craved"
    task instruct "Instruct the artist to create a project"
    task sendSketches "Send several sketches to the artist"
    xor projectReady
    xor satisfied "Satisfied with the result?"
    task tellChanges "Tell the artist what to change"
    task chooseColor "Choose a plastic color"
    xor inStock "Color in stock at home?"
    task checkLeft "Check how much color is left"
    xor under100 "Less than 100 grams left?"
    task shoppingList "Put the color on the shopping list"
    task orderPlastic "Order the plastic"
    xor havePlastic
    and startPrep
    task genGcode "Generate the gcode file from the STL"
    and readyPrint
    end done "Model printed"
  lane artist "Artist"
    task createProject "Create the project"
    task reviseProject "Change the project"
    task sendSTL "Send the STL file"
  lane printer "3D printer"
    task turnOn "Turn the printer on"
    task heatUp "Heat up the bed and the extruder"
    task reportStatus "Send its status back to you"
    task printModel "Print the model"

flow craving -> instruct -> sendSketches -> createProject -> projectReady -> satisfied
flow satisfied -> tellChanges "no"
flow tellChanges -> reviseProject -> projectReady
flow satisfied -> sendSTL "yes"
flow sendSTL -> chooseColor -> inStock
flow inStock -> checkLeft "yes" -> under100
flow inStock -> orderPlastic "no" -> havePlastic
flow under100 -> shoppingList "yes" -> havePlastic
flow under100 -> havePlastic "no"
flow havePlastic -> startPrep
flow startPrep -> turnOn -> heatUp -> reportStatus -> readyPrint
flow startPrep -> genGcode -> readyPrint
flow readyPrint -> printModel -> done
```
