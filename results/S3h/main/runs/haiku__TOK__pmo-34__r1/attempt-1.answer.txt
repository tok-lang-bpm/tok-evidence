```tok
tok 1.0 "Building a custom machine out of Lego bricks"

pool maker "Machine builder"
  lane builder "Builder"
    start begin "Machine project started"
    task design "Develop the basic design"
    task order "Order the Lego brick sets"
    xor ready
    xor stock "Parts left in the container?"
    task reorder "Reorder individual parts"
    task build "Build the subcomponent"
    task verify "Test the subcomponent individually"
    xor testok "Test successful?"
    task redesign "Redesign the subcomponent"
    task rebuild "Rebuild the subcomponent"
    task assemble "Assemble the subcomponent"
    xor more "Subcomponents remaining?"
    end done "Machine built"
  lane children "Group of children"
    task sort "Sort the parts into containers"

flow begin -> design -> order -> sort -> ready -> stock
flow stock -> build "yes"
flow stock -> reorder "no"
flow reorder -> ready
flow build -> verify -> testok
flow testok -> assemble "yes"
flow testok -> redesign "no"
flow redesign -> rebuild -> verify
flow assemble -> more
flow more -> ready "yes"
flow more -> done "no"
```
