```tok
tok 1.0 "Chainsaw"

pool shop "Chainsaw workshop"
  lane customer "Customer"
    start needed "Chainsaw order needed"
    task specify "Specify the chainsaw properties"
    task evaluate "Evaluate the first saw"
    xor liked "Customer likes the first saw?"
    end notcontinued "Rest of order not produced"
    end completed "Order completed"
  lane workshop "Workshop"
    task order "Order the parts from various online sources"
    and fork
    task updates "Send regular production updates to the customer"
    catch arrived "Parts arrived" message="parts-delivered"
    task inspect "Inspect all parts manually"
    task assemble "Assemble the parts into the first saw"
    and joined
    task sendfirst "Send the first saw to the customer"
    task produce "Produce the rest of the order"

flow needed -> specify -> order -> fork
flow fork -> updates -> joined
flow fork -> arrived -> inspect -> assemble -> joined
flow joined -> sendfirst -> evaluate -> liked
flow liked -> produce "yes"
flow liked -> notcontinued "no"
flow produce -> completed
```
