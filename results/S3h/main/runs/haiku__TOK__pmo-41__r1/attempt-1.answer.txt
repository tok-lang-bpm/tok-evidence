```tok
tok 1.0 "Inspection of an Energy Drink Bottling Machine"

pool org "Inspection organisation"
  lane inspector "Inspector"
    start begin "Inspection requested"
    task enterType "Enter the type of the machine"
    task enterSerial "Enter the serial number"
    xor needQuestions "Additional questions needed?"
    task answerQuestions "Input the values for the questions"
    task pressButtons "Press the buttons"
    task reviewResults "Review the results"
    xor more "Depending on the results, additional questions asked?"
    end done "Inspection completed"
  lane app "Inspection application"
    task collectValues "Collect the values from the machine"
    task showValues "Show the collected values"

flow begin -> enterType -> enterSerial -> needQuestions
flow needQuestions -> answerQuestions "yes"
flow needQuestions -> pressButtons "no"
flow answerQuestions -> pressButtons
flow pressButtons -> collectValues -> showValues -> reviewResults
flow reviewResults -> more
flow more -> answerQuestions "additional questions" when="true"
flow more -> done "no further questions" default
```
