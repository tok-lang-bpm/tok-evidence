```tok
tok 1.0 "Service for Your Fridge"

pool service "Fridge service"
  lane customer "Customer"
    start noise "Fridge produces strange noises"
    task describe "Describe the symptoms"
    task send "Send the symptoms and the fridge type to the central fridge service center"
    task confirm "Confirm that the fridge is working again"
    xor rateq "Rate the service facility?"
    task rate "Rate the service facility"
    end rated "Service rated"
    end finished "Service finished"
  lane central "Central fridge service center"
    task select "Select a local service facility"
  lane facility "Local service facility"
    task appoint "Make an appointment"
    task arrive "Arrive at a random time"
    xor parts "Additional parts needed?"
    task leave "Leave without repairing the fridge"
    task repair "Repair the fridge"

flow noise -> describe -> send -> select -> appoint -> arrive -> parts
flow parts -> leave "yes" when="parts_needed = true"
flow leave -> appoint
flow parts -> repair "no" default
flow repair -> confirm -> rateq
flow rateq -> rate "yes" when="wants_rating = true"
flow rateq -> finished "no" default
flow rate -> rated
```
