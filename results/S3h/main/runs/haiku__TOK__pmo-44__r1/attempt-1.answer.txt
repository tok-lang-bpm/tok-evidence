```tok
tok 1.0 "Robotic Burger Seller near the University of Vienna"

pool seller "Robotic Burger Seller"
  lane customer "Customer"
    start ordered "Order placed"
    xor menuq "Customer wants a menu?"
    task orderburger "Order the burger only"
    task chooseside "Choose fries or wedges"
    end delivered "Order delivered"
  lane robot "Robot"
    task receive "Receive the order"
    task askmenu "Ask whether the customer wants a menu or just the burger"
    and split
    task prepdrink "Prepare the drink"
    task askside "Ask whether the customer wants fries or wedges"
    xor side "Fries or wedges?"
    task prepfries "Prepare the fries"
    task prepwedges "Prepare the wedges"
    xor sidedone "Side prepared"
    task prepburger "Prepare the burger"
    task updatestatus "Give enthusiastic status updates every 30 seconds"
    task conveyor "Deliver the order via the conveyor belt"
    and joined
    end finished "Order finished"

flow ordered -> receive -> askmenu -> menuq
flow menuq -> split "yes"
flow menuq -> orderburger "no"
flow orderburger -> prepburger
flow split -> prepdrink
flow split -> askside
flow askside -> side
flow side -> prepfries "fries"
flow side -> prepwedges "wedges"
flow prepfries -> sidedone
flow prepwedges -> sidedone
flow prepdrink -> joined
flow sidedone -> joined
flow joined -> prepburger
flow prepburger -> updatestatus -> conveyor -> finished
flow delivered -> finished
```
