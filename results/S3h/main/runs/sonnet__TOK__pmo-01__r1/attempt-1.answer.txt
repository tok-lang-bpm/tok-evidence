```tok
tok 1.0 "Order handling"

pool org "Organisation"
  lane sales "Sales staff or customer support"
    start inquiry "Customer inquires about a product or service"
    task collect "Collect relevant information"
    task address "Address concerns and questions"
    xor interested "Customer interested?"
    task guide "Guide the customer through selecting the product or service"
    end notInterested "Customer not interested"
  lane rep "Sales representative"
    task quote "Provide a quote"
    xor approved "Customer approves the quote?"
    task place "Place the order"
    end declined "Quote declined"
  lane system "System"
    task record "Record the order in the system"
    task confirm "Send order confirmation to the customer"
    end done "Order placed and confirmed"

flow inquiry -> collect -> address -> interested
flow interested -> guide "yes"
flow interested -> notInterested "no" default
flow guide -> quote -> approved
flow approved -> place "yes"
flow approved -> declined "no" default
flow place -> record -> confirm -> done
```
