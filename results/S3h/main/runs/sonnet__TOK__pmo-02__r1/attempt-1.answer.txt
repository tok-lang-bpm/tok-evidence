```tok
tok 1.0 "Hiring process"

pool company "Company"
  lane department "Department"
    start need "Need for a new hire identified"
    task identify "Identify the need for a new hire"
  lane hr "HR"
    task create "Create a job description"
    task post "Post the job description on job boards"
    task collect "Collect resumes"
    task screen "Screen resumes"
    task phone "Conduct initial phone interviews"
    task invite "Invite selected candidates for interviews"
    xor preference "Candidate's preference?"
    task inperson "Conduct in-person interview"
    task virtual "Conduct virtual interview"
    xor interviewed
    task choose "Choose a candidate"
    task offer "Extend an offer"
    xor negotiate "Salary negotiation needed?"
    task negotiation "Negotiate the salary"
    xor agreed
    task paperwork "Complete the paperwork"
    task orientation "Conduct the orientation"
    task training "Conduct the training"
    task integrate "Integrate the new hire into the team"
    end done "New hire fully integrated"

flow need -> identify -> create -> post -> collect -> screen -> phone -> invite -> preference
flow preference -> inperson "in-person"
flow preference -> virtual "virtual"
flow inperson -> interviewed
flow virtual -> interviewed
flow interviewed -> choose -> offer -> negotiate
flow negotiate -> negotiation "yes"
flow negotiate -> agreed "no" default
flow negotiation -> agreed
flow agreed -> paperwork -> orientation -> training -> integrate -> done
```
