```tok
tok 1.0 "Order fulfilment"

pool org "Organisation"
  lane customer "Customer"
    start placed "Order placed"
  lane system "System"
    service confirm "Generate the order confirmation"
    send sendconf "Send the order confirmation to the customer"
    task label "Generate the shipping label"
    send track "Send the tracking information to the customer"
  lane warehouse "Warehouse team"
    and prep
    task pick "Pick the items"
    task pack "Pack the items"
    and prepared
  lane logistics "Logistics provider"
    task handover "Hand over the order to the logistics provider"
    task monitor "Monitor the shipment"
    task deliver "Deliver the shipment to the customer's address"
    end delivered "Order delivered"

flow placed -> confirm -> sendconf -> prep
flow prep -> pick -> pack -> prepared
flow prep -> label
flow label -> prepared
flow prepared -> handover -> track -> monitor -> deliver -> delivered
```
