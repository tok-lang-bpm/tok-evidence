tok 1.0 "Subscription lifecycle"

pool org "Subscription organisation"
  lane customer "Customer"
    start signedup "Customer signs up"
    task signup "Enter personal and payment information"
    task cancelreq "Submit the cancellation request"
  lane system "System"
    task generate "Generate the account"
    task assign "Assign access"
    task billing "Trigger automated billing cycles"
    task updates "Send updates, product enhancements and renewal notifications"
    xor cancels "Does the customer cancel?"
  lane support "Support team"
    task process "Process the cancellation request"
    xor terms "Is a refund due under the terms?"
    task refund "Apply the refund"
    task charges "Apply the charges"
    xor applied
    task deactivate "Deactivate the subscription"
    task settle "Settle the final account balance"
    end done "Subscription closed"
