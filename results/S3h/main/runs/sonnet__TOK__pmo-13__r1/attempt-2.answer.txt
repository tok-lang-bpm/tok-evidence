```tok
tok 1.0 "Complaint handling"

pool org "Organisation"
  lane customer "Customer"
    start filed "Complaint filed"
    task file "File the complaint"
    task feedback "Provide feedback on satisfaction"
    end done "Case closed"
  lane service "Customer service"
    task log "Log the complaint"
    task assign "Assign the complaint to the relevant department"
    task notify "Notify the customer with the decision"
    xor approved "Refund approved?"
    xor joined
    task resolve "Mark the complaint as resolved"
  lane dept "Relevant department"
    task review "Review the details"
    task decide "Determine whether the complaint is valid and a refund is justified"
  lane finance "Financial team"
    task reimburse "Process the reimbursement"

flow filed -> file -> log -> assign -> review -> decide -> notify -> approved
flow approved -> reimburse "yes" when="refundApproved = true"
flow approved -> joined "no" default
flow reimburse -> joined
flow joined -> resolve -> feedback -> done
```
