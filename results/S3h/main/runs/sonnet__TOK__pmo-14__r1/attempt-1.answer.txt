```tok
tok 1.0 "Budget planning"

pool org "Organisation"
  lane dept "Department"
    start begin "Planning period starts"
    task outline "Outline the objectives"
    task draft "Draft the plan"
    task adjust "Adjust the plan"
    task document "Document the adjustment"
    task implement "Begin implementing the plan"
    end done "Plan implementation begun"
  lane mgmt "Management and stakeholders"
    xor review "Review again"
    task align "Review the plan in the strategic alignment meeting"
    task approveadj "Approve the adjustment"
    task approvefinal "Approve the final budget"
  lane fin "Finance department"
    task finreview "Perform the detailed review"
    xor adjneeded "Adjustments needed?"
    task allocate "Allocate the budget"
    task distribute "Distribute the budget"

flow begin -> outline -> draft -> review -> align -> finreview -> adjneeded
flow adjneeded -> adjust "yes"
flow adjneeded -> approvefinal "no" default
flow adjust -> document -> approveadj -> review
flow approvefinal -> allocate -> distribute -> implement -> done
```
