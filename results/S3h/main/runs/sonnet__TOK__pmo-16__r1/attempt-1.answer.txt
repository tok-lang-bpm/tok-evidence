```tok
tok 1.0 "Annual audit process"

pool audit "Annual audit"
  lane coordinator "Audit coordinator"
    start begin "Audit needed"
    task notify "Send a notification to the regional office"
  lane regional "Regional office"
    and split1
    task prepare "Prepare financial statements"
    task gather "Gather necessary documents"
    and join1
    task submit "Submit the documents to the central audit team"
    task clarify "Provide the necessary clarifications"
  lane compliance "Compliance team"
    task checkreg "Check regulatory updates"
  lane team "Central audit team"
    task review "Review the submission"
    xor disc "Discrepancies found?"
    task request "Request clarifications from the regional office"
    adhoc risk "Conduct a risk assessment"
      task finrisk "Evaluate financial risks"
      task oprisk "Evaluate operational risks"
      task comprisk "Evaluate compliance risks"
    xor high "High risk identified?"
    task investigate "Launch a detailed investigation"
    and invsplit
    task analyze "Perform data analysis"
    task interview "Conduct interviews"
    xor sitegw "Site visit needed?"
    task visit "Conduct site visits"
    xor sitemerge
    and invjoin
    xor merge
    task compile "Compile the audit report"
    task update "Update the report"
    and split2
    task distribute "Distribute the final report to the executive board"
    task archive "Archive the final report"
    and join2
    task close "Close the audit process"
    end done "Audit process closed"
  lane director "Audit director"
    task reviewrep "Review the report"
    xor approved "Report approved?"

flow begin -> notify -> split1
flow split1 -> prepare -> gather -> join1
flow split1 -> checkreg -> join1
flow join1 -> submit -> review -> disc
flow disc -> request "yes"
flow request -> clarify -> review
flow disc -> risk "no"
flow risk -> high
flow high -> investigate "yes"
flow investigate -> invsplit
flow invsplit -> analyze -> invjoin
flow invsplit -> interview -> invjoin
flow invsplit -> sitegw
flow sitegw -> visit "yes"
flow visit -> sitemerge
flow sitegw -> sitemerge "no"
flow sitemerge -> invjoin
flow invjoin -> merge
flow high -> merge "no"
flow merge -> compile -> reviewrep -> approved
flow approved -> split2 "yes"
flow approved -> update "no"
flow update -> reviewrep
flow split2 -> distribute -> join2
flow split2 -> archive -> join2
flow join2 -> close -> done
```
