```tok
tok 1.0 "Online shop purchase"

pool shop "Online shop purchase"
  lane user "User"
    start begin "Order started"
    task login "Log in to the account"
    and split
    task select "Select the items to purchase"
    task reward "Choose a free reward"
    task payment "Set a payment method"
    xor how "Pay or installment?"
    task pay "Pay"
    task installment "Complete an installment agreement"
    xor paid
    and ready
    task return "Return the items for exchange"
  lane seller "Online shop"
    xor again
    task deliver "Deliver the items"
    xor returned "Items returned?"
    end done "Order completed"

flow begin -> login -> split
flow split -> select -> reward -> ready
flow split -> payment -> how
flow how -> pay "pay"
flow how -> installment "installment"
flow pay -> paid
flow installment -> paid
flow paid -> ready
flow ready -> again -> deliver -> returned
flow returned -> return "yes"
flow returned -> done "no" default
flow return -> again
```
