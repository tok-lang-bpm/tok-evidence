```tok
tok 1.0 "Shipping of goods"

pool company "Company"
  lane secretary "Secretary"
    start needed "Goods shall be shipped"
    task clarify "Clarify who will do the shipping"
    xor large "Large amounts?"
    task invite "Invite three logistic companies to make offers"
    task select "Select one logistic company"
    and split
    task label "Write package label"
    and ready
    xor shipped
  lane head "Logistics department head"
    xor insurance "Insurance necessary?"
    task insure "Take parcel insurance"
    xor insured
  lane warehouse "Warehousemen"
    task package "Package the goods"
    task prepare "Prepare the packaged goods for pick-up"
    end done "Goods ready for pick-up"

flow needed -> clarify -> large
flow large -> invite "yes"
flow large -> split "no"
flow invite -> select -> shipped
flow split -> label
flow split -> insurance
flow split -> package
flow insurance -> insure "yes"
flow insurance -> insured "no"
flow insure -> insured
flow label -> ready
flow insured -> ready
flow package -> ready
flow ready -> shipped
flow shipped -> prepare -> done
```
