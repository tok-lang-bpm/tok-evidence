```tok
tok 1.0 "LAN Party"

pool org "LAN party organisation"
  lane host "Host"
    start planned "LAN party planned"
    task invite "Send invitations to the 10 friends"
    task findgames "Find out which games the friends want to play"
    and split
    xor dateMerge
    task appoint "Appoint a date for the LAN party"
    xor enough "8 or more people agree on the date?"
    task checkgames "Find out if you have all the games the guests want to play"
    xor missing "Some games missing?"
    task download "Download the missing games"
    xor gamesMerge
    and join
    task findbeer "Find out how much beer the friends will require"
    xor beerEnough "Enough beer?"
    task buybeer "Buy the missing beer"
    xor beerMerge
    task enjoy "Enjoy the LAN party"
    end done "LAN party held"

flow planned -> invite -> findgames -> split
flow split -> dateMerge
flow split -> checkgames
flow dateMerge -> appoint -> enough
flow enough -> join "yes" when="agreed >= 8"
flow enough -> dateMerge "no" default
flow checkgames -> missing
flow missing -> download "yes" when="gamesMissing = true"
flow missing -> gamesMerge "no" default
flow download -> gamesMerge
flow gamesMerge -> join
flow join -> findbeer -> beerEnough
flow beerEnough -> buybeer "no" when="beerMissing = true"
flow beerEnough -> beerMerge "yes" default
flow buybeer -> beerMerge
flow beerMerge -> enjoy -> done
```
