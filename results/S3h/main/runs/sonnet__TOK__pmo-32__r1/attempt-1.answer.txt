```tok
tok 1.0 "Farming Bot"

pool game "Farming bot game"
  lane player "Player"
    start wanted "Player wants a farming bot"
    task find "Find out the wanted resources"
    and prefs
    task prioritise "Set the urgency of the resources"
    task depend "Define the dependencies of the resources"
    and prefsdone
    task update "Update the list of resources"
    end updated "List updated"
    or share "Share the result?"
    task brag "Brag to selected friends"
    task send "Send materials to friends"
    or shared
    end finished "Farming finished"
  lane bot "Farming bot"
    task farm "Farm the resources"
    boundary milestone on=farm signal="Milestone reached" noninterrupting
    task notify "Notify the player"
    end notified "Player notified"
    boundary disaster on=farm signal="Natural disaster strikes" noninterrupting
    task recover "Recover from the disaster"
    end recovered "Bot recovered"
    boundary change on=farm message="List update requested" noninterrupting

flow wanted -> find -> prefs
flow prefs -> prioritise
flow prefs -> depend
flow prioritise -> prefsdone
flow depend -> prefsdone
flow prefsdone -> farm
flow milestone -> notify -> notified
flow disaster -> recover -> recovered
flow change -> update -> updated
flow farm -> share
flow share -> brag "brag" when="brag = true"
flow share -> send "send materials" when="sendMaterials = true"
flow share -> shared "neither" default
flow brag -> shared
flow send -> shared
flow shared -> finished
```
