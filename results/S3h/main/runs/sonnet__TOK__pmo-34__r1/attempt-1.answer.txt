```tok
tok 1.0 "Building a custom machine out of Lego bricks"

pool project "Lego machine project"
  lane builder "Builder"
    start needed "Custom machine needed"
    task design "Develop the basic design"
    task order "Order Lego brick sets"
    task give "Give the Lego sets to the children"
    task build "Build the subcomponents"
    xor missing "Parts missing in a container?"
    task reorder "Reorder individual parts"
    task test "Test the subcomponents"
    xor passed "All tests successful?"
    task assemble "Assemble the machine"
    task redesign "Redesign the subcomponents"
    end done "Machine built"
  lane children "Children"
    task sort "Sort the parts into containers"

flow needed -> design -> order -> give -> sort -> build -> missing
flow missing -> reorder "yes"
flow reorder -> build
flow missing -> test "no" default
flow test -> passed
flow passed -> assemble "yes"
flow passed -> redesign "no" default
flow redesign -> build
flow assemble -> done
```
