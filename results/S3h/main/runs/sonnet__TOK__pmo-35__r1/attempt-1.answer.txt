```tok
tok 1.0 "Chainsaw"

pool maker "Chainsaw manufacturer"
  lane producer "Producer"
    start ordered "Custom chainsaw ordered"
    task receive_props "Receive the chainsaw properties from the customer"
    and order_split
    task order_a "Order parts from online source 1"
    task order_b "Order parts from online source 2"
    task order_c "Order parts from online source 3"
    and order_join
    task receive_parts "Receive the parts"
    and prod_split
    task inspect "Inspect all parts manually"
    task assemble "Assemble the parts"
    task updates "Send updates to the customer"
    and prod_join
    task send_saw "Send the first saw to the customer"
    xor liked "Does the customer like the saw?"
    task produce_rest "Produce the rest of the order"
    end done "Order completed"
    end cancelled "Order ended"

flow ordered -> receive_props -> order_split
flow order_split -> order_a -> order_join
flow order_split -> order_b -> order_join
flow order_split -> order_c -> order_join
flow order_join -> receive_parts -> prod_split
flow prod_split -> inspect -> assemble -> prod_join
flow prod_split -> updates -> prod_join
flow prod_join -> send_saw -> liked
flow liked -> produce_rest "yes"
flow liked -> cancelled "no"
flow produce_rest -> done
```
