```tok
tok 1.0 "Luxury Airplane"

pool company "Airplane producer"
  lane customer "Customer"
    start wish "Customer wants a luxury airplane"
    task specify "Specify the interior"
    task confirm "Confirm the delivery"
    end done "Airplane delivered"
  lane producer "Producer"
    task receive "Receive the specifications"
    task send_req "Send the requirements to the teams"
    task assemble "Assemble the interior"
    task test_flight "Send the plane on a test flight"
    task protocol "Create the test protocol"
    task send_protocol "Send the test protocol to the customer"
    task deliver "Deliver the plane to the customer"
  lane teams "Manufacturing teams"
    and split
    task vodka "Manufacture the vodka bar"
    task whiskey "Manufacture the whiskey bar"
    task seats "Manufacture the seats"
    task toilets "Manufacture the toilets"
    and join

flow wish -> specify -> receive -> send_req -> split
flow split -> vodka
flow split -> whiskey
flow split -> seats
flow split -> toilets
flow vodka -> join
flow whiskey -> join
flow seats -> join
flow toilets -> join
flow join -> assemble -> test_flight -> protocol -> send_protocol -> deliver -> confirm -> done
```
