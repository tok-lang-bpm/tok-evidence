```tok
tok 1.0 "Contract"

pool contract "Contract project"
  lane builder "Contractor"
    start needed "Parts needed for the contract project"
    task query "Query all web shops for quantity, price and delivery time"
    task list "Create the list of parts to order from each shop"
    xor enough "One shop has enough parts?"
    task orderOne "Order the parts from one shop"
    task orderMany "Order the parts from several shops"
    xor ordered
    receive first "Receive the first parts" message="Parts batch"
    task build "Start building"
    task check "Check the stock of a part"
    xor level "Stock level?"
    end enoughStock "Stock sufficient"
    task reorderCheapA "Reorder the cheapest parts"
    end reorderedA "Parts reordered"
    task reorderCheapB "Reorder the cheapest parts"
    task reorderFast "Reorder the fastest parts"
    end reorderedB "Parts reordered twice"
    task mail "Write a mail to friends complaining about the web shops"
    end complained "Complaint sent"

flow needed -> query -> list -> enough
flow enough -> orderOne "yes"
flow enough -> orderMany "no"
flow orderOne -> ordered
flow orderMany -> ordered
flow ordered -> first -> build -> check -> level
flow level -> enoughStock "5 or more"
flow level -> reorderCheapA "3 or 4"
flow level -> reorderCheapB "1 or 2"
flow level -> mail "0"
flow reorderCheapA -> reorderedA
flow reorderCheapB -> reorderFast -> reorderedB
flow mail -> complained
```
