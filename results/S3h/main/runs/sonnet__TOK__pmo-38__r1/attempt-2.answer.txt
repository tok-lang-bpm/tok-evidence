```tok
tok 1.0 "Building a House"

pool house "Building a House"
  lane you "You"
    start needed "Tree house wanted"
    task collect "Collect the requirements"
    task sendreq "Send the requirements to the architect"
    xor again
    task refine "Refine the draft with additional requirements"
    xor more "Refine again?"
    task list "Create the list of needed materials"
    and split
    task order "Order the materials from online stores" parallel
    task message "Send messages to friends to build the house"
    and joined
    task build "Build the tree house"
    task invite "Send invitations for the tree house party"
    task attendees "Create the list of attending people"
    task snacks "Buy the snacks"
    end done "Party prepared"
  lane architect "Tree house architect"
    task draft "Send the draft"

flow needed -> collect -> sendreq -> draft -> again -> refine -> more
flow more -> again "yes" when="refineAgain = true"
flow more -> list "no" default
flow list -> split
flow split -> order
flow split -> message
flow order -> joined
flow message -> joined
flow joined -> build -> invite -> attendees -> snacks -> done
```
