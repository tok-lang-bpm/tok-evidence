```tok
tok 1.0 "Inspection of an Energy Drink Bottling Machine"

pool inspection "Machine inspection"
  lane inspector "Inspector"
    start begin "Inspection needed"
    task enter "Enter the type and serial number of the machine"
    task input "Input values"
    task press "Press buttons"
    task inputmore "Input values for additional questions"
    end done "Inspection completed"
  lane app "Application"
    task ask "Ask questions"
    task collect "Collect values from the machine"
    task show "Show values"
    xor more "Additional questions needed?"
    task askmore "Ask additional questions"

flow begin -> enter -> ask -> input -> press -> collect -> show -> more
flow more -> askmore "yes"
flow more -> done "no" default
flow askmore -> inputmore -> done
```
