```tok
tok 1.0 "Service for Your Fridge"

pool fridgeservice "Fridge service"
  lane customer "Customer"
    start noises "Fridge produces strange noises"
    task describe "Describe the symptoms"
    task send "Send the symptoms and the type of fridge to the service center"
    task confirm "Confirm that the fridge is OK"
    task rate "Rate the service facility"
    end done "Fridge service completed"
  lane center "Central fridge service center"
    task select "Select a local service facility"
  lane facility "Local service facility"
    xor appointmentjoin
    task appoint "Make an appointment"
    task repair "Repair the fridge"
    xor repaired "Fridge repaired?"
    task leave "Leave without repairing the fridge"

flow noises -> describe -> send -> select -> appointmentjoin -> appoint -> repair -> repaired
flow repaired -> confirm "yes"
flow repaired -> leave "no, additional parts needed"
flow leave -> appointmentjoin
flow confirm -> rate -> done
```
