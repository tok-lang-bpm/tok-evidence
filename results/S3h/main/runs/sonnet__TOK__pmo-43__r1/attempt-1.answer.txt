```tok
tok 1.0 "Car Service"

pool police "Police"
  lane system "System"
    start serviced "Service successfully completed"
    task check "Check if the car is still registered"
    xor registered "Car still registered?"
    task notify "Notify the car owner"
    task fine "Fine the car owner"
    task status "Send status updates via email"
    end unregistered "Process ended, car not registered"
    end fined "Car owner fined"
  lane owner "Car owner"
    task go "Go to the car service"
    boundary late "30 days passed" on=go timer=P30D
    task pay "Pay through the app"
  lane mechanic "Mechanic"
    task enter "Enter the problems of the car"
    and split
    task repair "Repair the car"
    and join
    task confirm "Enter that the repair was successful and the Pickerl was issued"
    task next "Enter the time for the next service"
    end done "Car service completed"

flow serviced -> check -> registered
flow registered -> notify "yes"
flow registered -> unregistered "no" default
flow notify -> go -> enter -> split
flow late -> fine -> fined
flow split -> repair
flow split -> status
flow repair -> join
flow status -> join
flow join -> pay -> confirm -> next -> done
```
