```tok
tok 1.0 "App For Participating at the Vienna Night Run"

pool org "Vienna Night Run"
  lane runner "Runner"
    start begin "Wish to participate"
    task select "Select a starting block"
    xor again
    task run5 "Run 5km"
    xor fast "Time under 25 minutes?"
    task train "Train and check"
    task getnumber "Get the starting number"
    xor early "More than one hour left between starting time and end of workday?"
    task fromhome "Go there from home"
    task fromwork "Leave directly from work"
    xor arrived
    and race
    task runnight "Run at the Night Run"
    task drink "Drink"
    and raceend
    task receive "Receive the final running time"
    end finished "Night Run completed"
  lane gadget "Fitness gadget"
    task measure "Measure the time"

flow begin -> select -> again -> run5 -> measure -> fast
flow fast -> getnumber "yes" when="time < 25"
flow fast -> train "no" default
flow train -> again
flow getnumber -> early
flow early -> fromhome "yes" when="timeLeft > 1"
flow early -> fromwork "no" default
flow fromhome -> arrived
flow fromwork -> arrived
flow arrived -> race
flow race -> runnight
flow race -> drink
flow runnight -> raceend
flow drink -> raceend
flow raceend -> receive -> finished
```
