```tok
tok 1.0 "Ultrasound-guided catheter placement"

pool hospital "Hospital"
  lane doctor "Doctor"
    start begin "Procedure needed"
    task implements "Prepare the implements"
    task wash "Wash the hands"
    task clothes "Get in sterile clothes"
    task clean "Clean the puncture area"
    task drape "Drape the puncture zone"
    task configure "Configure the ultrasound"
    task gelprobe "Put gel in the probe"
    task cover "Cover the probe"
    task sterilegel "Put the sterile gel"
    task posprobe "Position the probe"
    task pospatient "Position the patient"
    xor veinsplit "How is the vein identified?"
    task anatomic "Identify the vein using an anatomic identification"
    task doppler "Identify the vein using a doppler identification"
    task compression "Identify the vein using a compression identification"
    xor veinjoin
    xor retry
    task anesthetize "Anesthetize the patient"
    task puncture "Puncture the patient"
    task checkblood "Check the blood return"
    xor bloodok "Blood return correct?"
    task dropprobe "Drop the probe"
    task syringe "Remove the syringe"
    task guidewire "Install the guidewire"
    task trocar "Remove the trocar"
    xor wiresplit "How is the wire checked?"
    task longaxis "Check the wire in the long axis"
    task shortaxis "Check the wire in the short axis"
    xor wirejoin
    task checkpos "Check if the wire is in good position"
    xor posok "Wire in good position?"
    task widen "Widen the pathway"
    task advance "Advance the catheter"
    task removewire "Remove the guidewire"
    task flow "Verify the flow and the reflow"
    task catheterpos "Check the catheter position"
    end done "Catheter placed"

flow begin -> implements -> wash -> clothes -> clean -> drape -> configure -> gelprobe -> cover -> sterilegel -> posprobe -> pospatient -> veinsplit
flow veinsplit -> anatomic "anatomic"
flow veinsplit -> doppler "doppler"
flow veinsplit -> compression "compression"
flow anatomic -> veinjoin
flow doppler -> veinjoin
flow compression -> veinjoin
flow veinjoin -> retry -> anesthetize -> puncture -> checkblood -> bloodok
flow bloodok -> retry "no"
flow bloodok -> dropprobe "yes"
flow dropprobe -> syringe -> guidewire -> trocar -> wiresplit
flow wiresplit -> longaxis "long axis"
flow wiresplit -> shortaxis "short axis"
flow longaxis -> wirejoin
flow shortaxis -> wirejoin
flow wirejoin -> checkpos -> posok
flow posok -> retry "no"
flow posok -> widen "yes"
flow widen -> advance -> removewire -> flow -> catheterpos -> done
```
