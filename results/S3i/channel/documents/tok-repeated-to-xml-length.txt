tok 1.0 "Supplier selection and onboarding"

pool org "Company"
  lane company "Company"
    start needed "Need for a new supplier identified"
    task identify "Identify the need for a new supplier"
  lane procurement "Procurement team"
    task issue "Issue the request for proposals"
    task receive "Receive responses from suppliers"
    task evaluate "Evaluate the proposals"
    xor visitq "Site visits or interviews needed?"
    task visit "Conduct site visits or interviews with suppliers"
    xor visitm
    task select "Select the supplier"
    task negotiate "Negotiate the contract"
    xor agreed "Terms agreed?"
    task sign "Sign the contract"
    task onboard "Onboard the supplier"
    task execute "Execute the contract"
    end done "Supplier onboarded and contract executed"

flow needed -> identify -> issue -> receive -> evaluate -> visitq
flow visitq -> visit "yes"
flow visitq -> visitm "no"
flow visit -> visitm
flow visitm -> select -> negotiate -> agreed
flow agreed -> sign "yes"
flow agreed -> negotiate "no"
flow sign -> onboard -> execute -> done
